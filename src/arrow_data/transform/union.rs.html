18use super::{_MutableArrayData, Extend};
19use crate::ArrayData;
20use arrow_schema::{ArrowError, DataType};

22fn nullable_union_field(fields: &arrow_schema::UnionFields) -> Result<(usize, i8), ArrowError> {
23    fields
24        .iter()
25        .enumerate()
26        .find_map(|(index, (type_id, field))| field.is_nullable().then_some((index, type_id)))
27        .ok_or_else(|| {
28            ArrowError::InvalidArgumentError(if fields.is_empty() {
29                "Cannot extend a union with no fields with nulls".to_owned()
30            } else {
31                "Cannot extend a union with no nullable fields with nulls".to_owned()
32            })
33        })
34}

36pub(super) fn build_extend_sparse(array: &ArrayData) -> Extend<'_> {
37    // `buffer` already applies the parent offset to the type ids
38    let type_ids = array.buffer::<i8>(0);
39    // but a sparse union's children are not sliced by the parent offset
40    let offset = array.offset();
41
42    Box::new(
43        move |mutable: &mut _MutableArrayData, index: usize, start: usize, len: usize| {
44            // extends type_ids
45            mutable
46                .buffer1
47                .extend_from_slice(&type_ids[start..start + len]);
48
49            let child_start = start.checked_add(offset).ok_or_else(|| {
50                ArrowError::InvalidArgumentError(format!(
51                    "sparse union offset {offset} with start {start} overflows usize"
52                ))
53            })?;
54            let child_end = child_start.checked_add(len).ok_or_else(|| {
55                ArrowError::InvalidArgumentError(format!(
56                    "sparse union start {child_start} with length {len} overflows usize"
57                ))
58            })?;
59            for child in &mut mutable.child_data {
60                child.try_extend(index, child_start, child_end)?;
61            }
62            Ok(())
63        },
64    )
65}

67pub(super) fn build_extend_dense(array: &ArrayData) -> Extend<'_> {
68    let type_ids = array.buffer::<i8>(0);
69    let offsets = array.buffer::<i32>(1);
70    let arrow_schema::DataType::Union(src_fields, _) = array.data_type() else {
71        unreachable!();
72    };
73
74    Box::new(
75        move |mutable: &mut _MutableArrayData, index: usize, start: usize, len: usize| {
76            // extends type_ids
77            mutable
78                .buffer1
79                .extend_from_slice(&type_ids[start..start + len]);
80
81            for i in start..start + len {
82                let type_id = type_ids[i];
83                let child_index = src_fields
84                    .iter()
85                    .position(|(r, _)| r == type_id)
86                    .expect("invalid union type ID");
87                let src_offset = offsets[i] as usize;
88                let child_data = &mut mutable.child_data[child_index];
89                let dst_offset = child_data.len();
90
91                // Extend offsets
92                mutable.buffer2.push(dst_offset as i32);
93                mutable.child_data[child_index].try_extend(index, src_offset, src_offset + 1)?;
94            }
95            Ok(())
96        },
97    )
98}

100pub(super) fn extend_nulls_dense(
101    mutable: &mut _MutableArrayData,
102    len: usize,
103) -> Result<(), ArrowError> {
104    let DataType::Union(fields, _) = &mutable.data_type else {
105        unreachable!()
106    };
107    let (child_index, null_type_id) = nullable_union_field(fields)?;
108
109    // Extend type_ids buffer
110    mutable
111        .buffer1
112        .try_extend_from_slice(&vec![null_type_id; len])
113        .map_err(|e| ArrowError::MemoryError(e.to_string()))?;
114
115    // Dense: point into a nullable child and extend its values with nulls.
116    let child_offset = mutable.child_data[child_index].len();
117    let (start, end) = (child_offset as i32, (child_offset + len) as i32);
118    mutable.buffer2.extend(start..end);
119    mutable.child_data[child_index].try_extend_nulls(len)?;
120    Ok(())
121}

123pub(super) fn extend_nulls_sparse(
124    mutable: &mut _MutableArrayData,
125    len: usize,
126) -> Result<(), ArrowError> {
127    let DataType::Union(fields, _) = &mutable.data_type else {
128        unreachable!()
129    };
130    let (_, null_type_id) = nullable_union_field(fields)?;
131
132    // Extend type_ids buffer
133    mutable
134        .buffer1
135        .try_extend_from_slice(&vec![null_type_id; len])
136        .map_err(|e| ArrowError::MemoryError(e.to_string()))?;
137
138    // Sparse: extend nulls in ALL children
139    for child in &mut mutable.child_data {
140        child.try_extend_nulls(len)?;
141    }
142    Ok(())
143}
