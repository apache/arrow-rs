18//! Cast support for numeric and boolean arrays.
19
20use crate::cast::*;
21
22/// Convert Array into a PrimitiveArray of type, and apply numeric cast
23pub(crate) fn cast_numeric_arrays<FROM, TO>(
24    from: &dyn Array,
25    cast_options: &CastOptions,
26) -> Result<ArrayRef, ArrowError>
27where
28    FROM: ArrowPrimitiveType,
29    TO: ArrowPrimitiveType,
30    FROM::Native: NumCast + NumericNative,
31    TO::Native: NumCast + NumericNative,
32{
33    let from = from.as_primitive::<FROM>();
34    let array: PrimitiveArray<TO> = if const {
35        is_infallible_numeric_cast(
36            <FROM::Native as NumericNative>::KIND,
37            <TO::Native as NumericNative>::KIND,
38        )
39    } {
40        // This cast cannot fail, so the fastest kernel, `unary`, can be used.
41        from.unary(|v| num_cast(v).expect("numeric cast is infallible"))
42    } else if cast_options.safe {
43        // If the value can't be cast to the `TO::Native`, return null
44        from.unary_opt(num_cast)
45    } else {
46        // If the value can't be cast to the `TO::Native`, return error
47        from.try_unary(|v| {
48            num_cast(v).ok_or_else(|| {
49                ArrowError::CastError(format!("Can't cast value {v:?} to type {}", TO::DATA_TYPE))
50            })
51        })?
52    };
53    Ok(Arc::new(array))
54}

56/// Properties of a numeric native type that decide whether [`num_cast`] to
57/// another numeric type can fail.
58#[derive(Clone, Copy)]
59pub(crate) struct NumericKind {
60    float: bool,
61    signed: bool,
62    bits: u32,
63}
64
65/// A native type with a [`NumericKind`].
66pub(crate) trait NumericNative {
67    const KIND: NumericKind;
68}
69
70macro_rules! numeric_native {
71    ($($t:ty => $float:literal, $signed:literal;)*) => {$(
72        impl NumericNative for $t {
73            const KIND: NumericKind = NumericKind {
74                float: $float,
75                signed: $signed,
76                bits: 8 * std::mem::size_of::<$t>() as u32,
77            };
78        }
79    )*};
80}
81
82numeric_native! {
83    u8 => false, false;
84    u16 => false, false;
85    u32 => false, false;
86    u64 => false, false;
87    i8 => false, true;
88    i16 => false, true;
89    i32 => false, true;
90    i64 => false, true;
91    half::f16 => true, true;
92    f32 => true, true;
93    f64 => true, true;
94}
95
96/// Returns true if [`num_cast`] succeeds for every value of `from` when casting
97/// to `to`: every cast to a float, and integer casts whose target holds every
98/// source value.
99///
100/// This must hold for every bit pattern of the source type, not only for the
101/// valid values of a particular array, because [`PrimitiveArray::unary`] applies
102/// the conversion to null slots as well, and their contents are arbitrary.
103pub(crate) const fn is_infallible_numeric_cast(from: NumericKind, to: NumericKind) -> bool {
104    if to.float {
105        true
106    } else if from.float {
107        false
108    } else if from.signed == to.signed {
109        to.bits >= from.bits
110    } else {
111        // Only an unsigned source fits a strictly wider signed target.
112        !from.signed && to.bits > from.bits
113    }
114}

116/// Natural cast between numeric types
117/// Return None if the input `value` can't be casted to type `O`.
118#[inline]
119pub fn num_cast<I, O>(value: I) -> Option<O>
120where
121    I: NumCast,
122    O: NumCast,
123{
124    num_traits::cast::cast::<I, O>(value)
125}

127/// Cast numeric types to Boolean
128///
129/// Any zero value returns `false` while non-zero returns `true`
130pub(crate) fn cast_numeric_to_bool<FROM>(from: &dyn Array) -> Result<ArrayRef, ArrowError>
131where
132    FROM: ArrowPrimitiveType,
133{
134    Ok(Arc::new(BooleanArray::from_unary(
135        from.as_primitive::<FROM>(),
136        cast_num_to_bool,
137    )))
138}
139
140/// Cast numeric types to boolean
141#[inline]
142pub fn cast_num_to_bool<I>(value: I) -> bool
143where
144    I: Default + PartialEq,
145{
146    value != I::default()
147}

149/// Cast Boolean types to numeric
150///
151/// `false` returns 0 while `true` returns 1
152pub(crate) fn cast_bool_to_numeric<TO>(
153    from: &dyn Array,
154    cast_options: &CastOptions,
155) -> Result<ArrayRef, ArrowError>
156where
157    TO: ArrowPrimitiveType,
158{
159    Ok(Arc::new(bool_to_numeric_cast::<TO>(
160        from.as_any().downcast_ref::<BooleanArray>().unwrap(),
161        cast_options,
162    )))
163}
164
165fn bool_to_numeric_cast<T>(from: &BooleanArray, _cast_options: &CastOptions) -> PrimitiveArray<T>
166where
167    T: ArrowPrimitiveType,
168{
169    let to_numeric = |bits: u64, i: usize| {
170        if bits & (1 << i) != 0 {
171            T::Native::ONE
172        } else {
173            T::Native::ZERO
174        }
175    };
176    // Unpack a 64-bit word at a time: the fixed-size inner loop helps the
177    // compiler vectorize the conversion.
178    let chunks = from.values().bit_chunks();
179    let mut values = Vec::with_capacity(from.len());
180    for bits in &chunks {
181        values.extend((0..64).map(|i| to_numeric(bits, i)));
182    }
183    let bits = chunks.remainder_bits();
184    values.extend((0..chunks.remainder_len()).map(|i| to_numeric(bits, i)));
185    PrimitiveArray::new(values.into(), from.nulls().cloned())
186}

188/// Cast single bool value to numeric value.
189#[inline]
190pub fn single_bool_to_numeric<O>(value: bool) -> Option<O>
191where
192    O: num_traits::NumCast + Default,
193{
194    if value {
195        // a workaround to cast a primitive to type O, infallible
196        num_traits::cast::cast(1)
197    } else {
198        Some(O::default())
199    }
