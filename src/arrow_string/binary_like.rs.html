5//! Provide SQL's CONTAINS, STARTS_WITH, ENDS_WITH operators for Arrow's binary arrays
6
7use crate::binary_predicate::BinaryPredicate;
8
9use arrow_array::cast::AsArray;
10use arrow_array::*;
11use arrow_schema::*;
12use arrow_select::take::take;

14#[derive(Debug)]
15pub(crate) enum Op {
16    Contains,
17    StartsWith,
18    EndsWith,
19}
20
21impl TryFrom<crate::like::Op> for Op {
22    type Error = ArrowError;
23
24    fn try_from(value: crate::like::Op) -> Result<Self, Self::Error> {
25        match value {
26            crate::like::Op::Contains => Ok(Op::Contains),
27            crate::like::Op::StartsWith => Ok(Op::StartsWith),
28            crate::like::Op::EndsWith => Ok(Op::EndsWith),
29            _ => Err(ArrowError::InvalidArgumentError(format!(
30                "Invalid binary operation: {value}"
31            ))),
32        }
33    }
34}
35
36impl std::fmt::Display for Op {
37    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
38        match self {
39            Op::Contains => write!(f, "CONTAINS"),
40            Op::StartsWith => write!(f, "STARTS_WITH"),
41            Op::EndsWith => write!(f, "ENDS_WITH"),
42        }
43    }
44}

46pub(crate) fn binary_apply<'a, T: BinaryArrayType<'a> + 'a>(
47    op: Op,
48    l: T,
49    l_s: bool,
50    l_v: Option<&'a dyn AnyDictionaryArray>,
51    r: T,
52    r_s: bool,
53    r_v: Option<&'a dyn AnyDictionaryArray>,
54) -> Result<BooleanArray, ArrowError> {
55    let l_len = l_v.map_or_else(|| l.len(), |l| l.len());
56    if r_s {
57        let idx = match r_v {
58            Some(dict) if dict.null_count() != 0 => return Ok(BooleanArray::new_null(l_len)),
59            Some(dict) => dict.normalized_keys()[0],
60            None => 0,
61        };
62        if r.is_null(idx) {
63            return Ok(BooleanArray::new_null(l_len));
64        }
65        op_scalar::<T>(op, l, l_v, r.value(idx))
66    } else {
67        match (l_s, l_v, r_v) {
68            (true, None, None) => {
69                let v = l.is_valid(0).then(|| l.value(0));
70                op_binary(op, std::iter::repeat(v), r.iter())
71            }
72            (true, Some(l_v), None) => {
73                let idx = l_v.is_valid(0).then(|| l_v.normalized_keys()[0]);
74                let v = idx.and_then(|idx| l.is_valid(idx).then(|| l.value(idx)));
75                op_binary(op, std::iter::repeat(v), r.iter())
76            }
77            (true, None, Some(r_v)) => {
78                let v = l.is_valid(0).then(|| l.value(0));
79                op_binary(op, std::iter::repeat(v), vectored_iter(r, r_v))
80            }
81            (true, Some(l_v), Some(r_v)) => {
82                let idx = l_v.is_valid(0).then(|| l_v.normalized_keys()[0]);
83                let v = idx.and_then(|idx| l.is_valid(idx).then(|| l.value(idx)));
84                op_binary(op, std::iter::repeat(v), vectored_iter(r, r_v))
85            }
86            (false, None, None) => op_binary(op, l.iter(), r.iter()),
87            (false, Some(l_v), None) => op_binary(op, vectored_iter(l, l_v), r.iter()),
88            (false, None, Some(r_v)) => op_binary(op, l.iter(), vectored_iter(r, r_v)),
89            (false, Some(l_v), Some(r_v)) => {
90                op_binary(op, vectored_iter(l, l_v), vectored_iter(r, r_v))
91            }
92        }
93    }
94}

96#[inline(never)]
97fn op_scalar<'a, T: BinaryArrayType<'a>>(
98    op: Op,
99    l: T,
100    l_v: Option<&dyn AnyDictionaryArray>,
101    r: &[u8],
102) -> Result<BooleanArray, ArrowError> {
103    let r = match op {
104        Op::Contains => BinaryPredicate::contains(r).evaluate_array(l, false),
105        Op::StartsWith => BinaryPredicate::StartsWith(r).evaluate_array(l, false),
106        Op::EndsWith => BinaryPredicate::EndsWith(r).evaluate_array(l, false),
107    };
108
109    Ok(match l_v {
110        Some(v) => take(&r, v.keys(), None)?.as_boolean().clone(),
111        None => r,
112    })
113}
114
115fn vectored_iter<'a, T: BinaryArrayType<'a> + 'a>(
116    a: T,
117    a_v: &'a dyn AnyDictionaryArray,
118) -> impl Iterator<Item = Option<&'a [u8]>> + 'a {
119    let nulls = a_v.nulls();
120    let keys = a_v.normalized_keys();
121    keys.into_iter().enumerate().map(move |(idx, key)| {
122        if nulls.is_some_and(|n| n.is_null(idx)) || a.is_null(key) {
123            return None;
124        }
125        Some(a.value(key))
126    })
127}

129#[inline(never)]
130fn op_binary<'a>(
131    op: Op,
132    l: impl Iterator<Item = Option<&'a [u8]>>,
133    r: impl Iterator<Item = Option<&'a [u8]>>,
134) -> Result<BooleanArray, ArrowError> {
135    match op {
136        Op::Contains => Ok(l
137            .zip(r)
138            .map(|(l, r)| Some(bytes_contains(l?, r?)))
139            .collect()),
140        Op::StartsWith => Ok(l
141            .zip(r)
142            .map(|(l, r)| Some(BinaryPredicate::StartsWith(r?).evaluate(l?)))
143            .collect()),
144        Op::EndsWith => Ok(l
145            .zip(r)
146            .map(|(l, r)| Some(BinaryPredicate::EndsWith(r?).evaluate(l?)))
147            .collect()),
148    }
149}
150
151fn bytes_contains(haystack: &[u8], needle: &[u8]) -> bool {
152    memchr::memmem::find(haystack, needle).is_some()
153}
