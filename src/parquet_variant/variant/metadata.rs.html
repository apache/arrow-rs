13/// Header structure for [`VariantMetadata`]
14#[derive(Debug, Clone, Copy, PartialEq)]
15pub(crate) struct VariantMetadataHeader {
16    version: u8,
17    is_sorted: bool,
18    /// Note: This is `offset_size_minus_one` + 1
19    offset_size: OffsetSizeBytes,
20}
21
22/// The version of the Variant specification implemented by this crate.
23///
24/// The spec currently defines only version 1. It appears in two places on
25/// disk: the 4-bit version field of every metadata header, and the
26/// `specification_version` of the Parquet `VARIANT` logical type annotation.
27/// Both must carry this value.
28pub const VARIANT_SPEC_VERSION: u8 = 1;
29
30// The metadata header occupies one byte; use a named constant for readability
31const NUM_HEADER_BYTES: u32 = 1;
32
33impl VariantMetadataHeader {
34    // Hide the cast
35    const fn offset_size(&self) -> u32 {
36        self.offset_size as u32
37    }
38
39    // Avoid materializing this offset, since it's cheaply and safely computable
40    const fn first_offset_byte(&self) -> u32 {
41        NUM_HEADER_BYTES + self.offset_size()
42    }
43
44    /// Tries to construct the variant metadata header, which has the form
45    ///
46    /// ```text
47    ///              7     6  5   4  3             0
48    ///             +-------+---+---+---------------+
49    /// header      |       |   |   |    version    |
50    ///             +-------+---+---+---------------+
51    ///                 ^         ^
52    ///                 |         +-- sorted_strings
53    ///                 +-- offset_size_minus_one
54    /// ```
55    ///
56    /// The version is a 4-bit value that must always contain the value 1.
57    /// - sorted_strings is a 1-bit value indicating whether dictionary strings are sorted and unique.
58    /// - offset_size_minus_one is a 2-bit value providing the number of bytes per dictionary size and offset field.
59    /// - The actual number of bytes, offset_size, is offset_size_minus_one + 1
60    pub(crate) fn try_new(header_byte: u8) -> Result<Self, ArrowError> {
61        let version = header_byte & 0x0F; // First four bits
62        if version != VARIANT_SPEC_VERSION {
63            let err_msg = format!(
64                "The version bytes in the header is not {VARIANT_SPEC_VERSION}, got {version:b}",
65            );
66            return Err(ArrowError::InvalidArgumentError(err_msg));
67        }
68        let is_sorted = (header_byte & 0x10) != 0; // Fifth bit
69        let offset_size_minus_one = header_byte >> 6; // Last two bits
70        Ok(Self {
71            version,
72            is_sorted,
73            offset_size: OffsetSizeBytes::try_new(offset_size_minus_one)?,
74        })
75    }
76}

78/// [`Variant`] Metadata
79///
80/// See the [Variant Spec] file for more information
81///
82/// # Validation
83///
84/// Every instance of variant metadata is either _valid_ or _invalid_. depending on whether the
85/// underlying bytes are a valid encoding of variant metadata (see below).
86///
87/// Instances produced by [`Self::try_new`] or [`Self::with_full_validation`] are fully _validated_. They always
88/// contain _valid_ data, and infallible accesses such as iteration and indexing are panic-free. The
89/// validation cost is linear in the number of underlying bytes.
90///
91/// Instances produced by [`Self::new`] are _unvalidated_ and so they may contain either _valid_ or
92/// _invalid_ data. Infallible accesses such as iteration and indexing will panic if the underlying
93/// bytes are _invalid_, and fallible alternatives such as [`Self::iter_try`] and [`Self::get`] are
94/// provided as panic-free alternatives. [`Self::with_full_validation`] can also be used to _validate_ an
95/// _unvalidated_ instance, if desired.
96///
97/// _Unvalidated_ instances can be constructed in constant time. This can be useful if the caller
98/// knows the underlying bytes were already validated previously, or if the caller intends to
99/// perform a small number of (fallible) accesses to a large dictionary.
100///
101/// A _validated_ variant [metadata instance guarantees that:
102///
103/// - header byte is valid
104/// - dictionary size is in bounds
105/// - offset array content is in-bounds
106/// - first offset is zero
107/// - last offset is in-bounds
108/// - all other offsets are in-bounds (*)
109/// - all offsets are non-decreasing (*)
110/// - all values are valid utf-8 (*)
111///
112/// NOTE: [`Self::new`] only skips expensive (non-constant cost) validation checks (marked by `(*)`
113/// in the list above); it panics any of the other checks fails.
114///
115/// # Safety
116///
117/// Even an _invalid_ variant metadata instance is still _safe_ to use in the Rust sense. Accessing
118/// it with infallible methods may cause panics but will never lead to undefined behavior.
119///
120/// [`Variant`]: crate::Variant
121/// [Variant Spec]: https://github.com/apache/parquet-format/blob/master/VariantEncoding.md#metadata-encoding
122#[derive(Debug, Clone, PartialEq)]
123pub struct VariantMetadata<'m> {
124    /// (Only) the bytes that make up this metadata instance.
125    pub(crate) bytes: &'m [u8],
126    header: VariantMetadataHeader,
127    dictionary_size: u32,
128    first_value_byte: u32,
129    validated: bool,
130}
131

174    /// Attempts to interpret `bytes` as a variant metadata instance, with full [validation] of all
175    /// dictionary entries.
176    ///
177    /// [validation]: Self#Validation
178    pub fn try_new(bytes: &'m [u8]) -> Result<Self, ArrowError> {
179        Self::try_new_with_shallow_validation(bytes)?.with_full_validation()
180    }

182    /// Interprets `bytes` as a variant metadata instance, without attempting to [validate] dictionary
183    /// entries. Panics if basic sanity checking fails, and subsequent infallible accesses such as
184    /// indexing and iteration could also panic if the underlying bytes are invalid.
185    ///
186    /// This constructor can be a useful lightweight alternative to [`Self::try_new`] if the bytes
187    /// were already validated previously by other means, or if the caller expects a small number of
188    /// accesses to a large dictionary (preferring to use a small number of fallible accesses as
189    /// needed, instead of paying expensive full validation up front).
190    ///
191    /// [validate]: Self#Validation
192    ///
193    /// # Panics
194    ///
195    /// Panics if basic sanity checking fails. Use [`Self::try_new`] for a fallible version.
196    pub fn new(bytes: &'m [u8]) -> Self {
197        Self::try_new_with_shallow_validation(bytes).expect("Invalid variant metadata")
198    }

200    // The actual constructor, which performs only basic (constant-const) validation.
201    pub(crate) fn try_new_with_shallow_validation(bytes: &'m [u8]) -> Result<Self, ArrowError> {
202        let header_byte = first_byte_from_slice(bytes)?;
203        let header = VariantMetadataHeader::try_new(header_byte)?;
204
205        // First element after header is dictionary size; the offset array immediately follows.
206        let dictionary_size =
207            header
208                .offset_size
209                .unpack_u32_at_offset(bytes, NUM_HEADER_BYTES as usize, 0)?;
210
211        // Calculate the starting offset of the dictionary string bytes.
212        //
213        // There are dict_size + 1 offsets, and the value bytes immediately follow
214        // = (dict_size + 1) * offset_size + header.first_offset_byte()
215        let first_value_byte = dictionary_size
216            .checked_add(1)
217            .and_then(|n| n.checked_mul(header.offset_size()))
218            .and_then(|n| n.checked_add(header.first_offset_byte()))
219            .ok_or_else(|| overflow_error("offset of variant metadata dictionary"))?;
220
221        let mut new_self = Self {
222            bytes,
223            header,
224            dictionary_size,
225            first_value_byte,
226            validated: false,
227        };
228
229        // Validate just the first and last offset, ignoring the other offsets and all value bytes.
230        let first_offset = new_self.get_offset(0)?;
231        if first_offset != 0 {
232            return Err(ArrowError::InvalidArgumentError(format!(
233                "First offset is not zero: {first_offset}"
234            )));
235        }
236
237        // Use the last offset to upper-bound the byte slice
238        let last_offset = new_self
239            .get_offset(dictionary_size as _)?
240            .checked_add(first_value_byte)
241            .ok_or_else(|| overflow_error("variant metadata size"))?;
242        new_self.bytes = slice_from_slice(bytes, ..last_offset as _)?;
243        Ok(new_self)
244    }

263    /// Performs a full [validation] of this metadata dictionary and returns the result.
264    ///
265    /// [validation]: Self#Validation
266    pub fn with_full_validation(mut self) -> Result<Self, ArrowError> {
267        if !self.validated {
268            let offset_bytes = slice_from_slice(
269                self.bytes,
270                self.header.first_offset_byte() as _..self.first_value_byte as _,
271            )?;
272
273            // Verify the string values in the dictionary are UTF-8 encoded strings.
274            let value_buffer =
275                string_from_slice(self.bytes, 0, self.first_value_byte as _..self.bytes.len())?;
276
277            let mut offsets = map_bytes_to_offsets(offset_bytes, self.header.offset_size);
278            let mut current_offset = offsets.next().unwrap_or(0);
279
280            if self.header.is_sorted {
281                // Validate the dictionary values are unique and lexicographically sorted
282                //
283                // Since we use the offsets to access dictionary values, this also validates
284                // offsets are in-bounds and monotonically increasing
285                let mut prev_value: Option<&str> = None;
286                for next_offset in offsets {
287                    let current_value = value_buffer.get(current_offset..next_offset).ok_or_else(
288                        || {
289                            ArrowError::InvalidArgumentError(format!(
290                                "range {current_offset}..{next_offset} is invalid or out of bounds"
291                            ))
292                        },
293                    )?;
294
295                    if let Some(prev_val) = prev_value
296                        && current_value <= prev_val
297                    {
298                        return Err(ArrowError::InvalidArgumentError(
299                            "dictionary values are not unique and ordered".to_string(),
300                        ));
301                    }
302
303                    prev_value = Some(current_value);
304                    current_offset = next_offset;
305                }
306            } else {
307                // Slicing each dictionary value validates that offsets are in-bounds, non-decreasing,
308                // and land on UTF-8 character boundaries. Equal offsets are legal: they encode an
309                // empty dictionary entry.
310                for next_offset in offsets {
311                    value_buffer
312                        .get(current_offset..next_offset)
313                        .ok_or_else(|| {
314                            ArrowError::InvalidArgumentError(format!(
315                                "range {current_offset}..{next_offset} is invalid or out of bounds"
316                            ))
317                        })?;
318                    current_offset = next_offset;
319                }
320            }
321
322            self.validated = true;
323        }
324        Ok(self)
325    }

337    /// Gets an offset into the dictionary entry by index.
338    ///
339    /// This offset is an index into the dictionary, at the boundary between string `i-1` and string
340    /// `i`. See [`Self::get`] to retrieve a specific dictionary entry.
341    fn get_offset(&self, i: usize) -> Result<u32, ArrowError> {
342        let offset_byte_range = self.header.first_offset_byte() as _..self.first_value_byte as _;
343        let bytes = slice_from_slice(self.bytes, offset_byte_range)?;
344        self.header.offset_size.unpack_u32(bytes, i)
345    }

356    /// Attempts to retrieve a dictionary entry by index, failing if out of bounds or if the
357    /// underlying bytes are [invalid].
358    ///
359    /// [invalid]: Self#Validation
360    pub fn get(&self, i: usize) -> Result<&'m str, ArrowError> {
361        let byte_range = self.get_offset(i)? as _..self.get_offset(i + 1)? as _;
362        if !self.validated {
363            return string_from_slice(self.bytes, self.first_value_byte as _, byte_range);
364        }
365
366        // Full validation already proved every dictionary entry is valid UTF-8, so validating
367        // again here would charge that cost once per field access instead of once per buffer.
368        let value_bytes =
369            slice_from_slice_at_offset(self.bytes, self.first_value_byte as _, byte_range)?;
370
371        debug_assert!(std::str::from_utf8(value_bytes).is_ok());
372
373        // SAFETY: `validated` is set only by `with_full_validation`, which proved that every
374        // dictionary entry, including this one, is valid UTF-8.
375        Ok(unsafe { str::from_utf8_unchecked(value_bytes) })
376    }

378    // Helper method used by our `impl Index` and also by `get_entry`. Panics if the underlying
379    // bytes are invalid. Needed because the `Index` trait forces the returned result to have the
380    // lifetime of `self` instead of the string's own (longer) lifetime `'m`.
381    fn get_impl(&self, i: usize) -> &'m str {
382        self.get(i).expect("Invalid metadata dictionary entry")
383    }

385    /// Attempts to resolve `field_name` to its field id under the assumption that `field_name` is
386    /// itself a slice of this dictionary's value region, as is the case for every field name
387    /// obtained from [`VariantObject::field_name`] or [`Self::get`] on the same metadata instance.
388    /// Returns `None` when that assumption does not hold, so callers must be prepared to fall back
389    /// to a name-based search such as [`Self::get_entry`].
390    ///
391    /// This is much cheaper than searching by name, because a borrowed field name already encodes
392    /// its own field id: it belongs to the entry whose dictionary offset equals its distance from
393    /// the start of the value region. Finding that entry is a binary search over the
394    /// (non-decreasing) offset array, comparing integers rather than decoding and comparing
395    /// dictionary strings at every step, and it needs no string comparison to confirm the hit.
396    ///
397    /// # Correctness
398    ///
399    /// This is only attempted for a [sorted] dictionary. The spec requires dictionary entries to
400    /// be unique only when `sorted_strings` is set, so an unsorted dictionary may legally hold the
401    /// same string at more than one field id. Resolving a borrowed name by the id it came from
402    /// would then disagree with a resolution by name, and callers rely on a name mapping to a
403    /// single id: [`ObjectBuilder`] detects duplicate fields by comparing field ids, so two ids
404    /// naming the same string would build an object whose field names are not unique. Requiring
405    /// sortedness makes the two resolutions agree, because validation rejects a sorted dictionary
406    /// with duplicate entries.
407    ///
408    /// A returned field id is also always verified, so this never reports an id whose entry is not
409    /// `field_name` itself, even for [invalid] metadata whose offsets are arbitrary. The candidate
410    /// entry is accepted only when it starts at `field_name`'s address and has exactly
411    /// `field_name`'s length, which makes the entry's bytes and `field_name`'s bytes the same
412    /// bytes. (Two live allocations cannot overlap, so an address inside our own byte range
413    /// belongs to our own bytes. In the degenerate zero-length case the two are both empty and
414    /// therefore still equal.)
415    ///
416    /// [`ObjectBuilder`]: crate::ObjectBuilder
417    /// [`VariantObject::field_name`]: crate::VariantObject::field_name
418    /// [invalid]: Self#Validation
419    /// [sorted]: Self::is_sorted
420    pub(crate) fn borrowed_field_id(&self, field_name: &str) -> Option<u32> {
421        // An unsorted dictionary may hold duplicate entries, which would make the id a name was
422        // borrowed from differ from the id a search by name returns; see "Correctness" above.
423        if !self.is_sorted() {
424            return None;
425        }
426
427        // Addresses are compared as integers and never dereferenced, so this stays safe even when
428        // `field_name` borrows from an unrelated allocation.
429        let value_region_start = (self.bytes.as_ptr() as usize) + self.first_value_byte as usize;
430        let value_region_end = (self.bytes.as_ptr() as usize) + self.bytes.len();
431        let field_name_start = field_name.as_ptr() as usize;
432        if field_name_start < value_region_start || field_name_start >= value_region_end {
433            return None;
434        }
435        let field_name_offset = u32::try_from(field_name_start - value_region_start).ok()?;
436
437        // Hoist the offset array out of the search, so each step is just an unaligned load.
438        let offset_byte_range = self.header.first_offset_byte() as _..self.first_value_byte as _;
439        let offsets = slice_from_slice(self.bytes, offset_byte_range).ok()?;
440        let offset_size = self.header.offset_size;
441        let cmp = |i| {
442            Some(
443                offset_size
444                    .unpack_u32(offsets, i)
445                    .ok()?
446                    .cmp(&field_name_offset),
447            )
448        };
449        let field_id = try_binary_search_range_by(0..self.len(), cmp)?.ok()?;
450
451        // Verify that this entry has exactly `field_name`'s bytes; see "Correctness" above.
452        let entry_end = offset_size.unpack_u32(offsets, field_id + 1).ok()?;
453        let entry_len = entry_end.checked_sub(field_name_offset)?;
454        (entry_len as usize == field_name.len()).then_some(field_id as u32)
455    }

457    /// Attempts to retrieve a dictionary entry and its field id, returning None if the requested field
458    /// name is not present. The search cost is logarithmic if [`Self::is_sorted`] and linear
459    /// otherwise.
460    ///
461    /// # Panics
462    ///
463    /// Panics if the underlying bytes are [invalid].
464    ///
465    /// [invalid]: Self#Validation
466    pub fn get_entry(&self, field_name: &str) -> Option<(u32, &'m str)> {
467        // A field name borrowed from this dictionary's (sorted) value region resolves without
468        // any string comparisons.
469        if let Some(field_id) = self.borrowed_field_id(field_name) {
470            return Some((field_id, self.get_impl(field_id as _)));
471        }
472
473        let field_id = if self.is_sorted() && self.len() > 10 {
474            // Binary search is faster for a not-tiny sorted metadata dictionary
475            let cmp = |i| Some(self.get_impl(i).cmp(field_name));
476            try_binary_search_range_by(0..self.len(), cmp)?.ok()?
477        } else {
478            // Fall back to Linear search for tiny or unsorted dictionary
479            (0..self.len()).find(|i| self.get_impl(*i) == field_name)?
480        };
481        Some((field_id as u32, self.get_impl(field_id)))
482    }

484    /// Returns an iterator that attempts to visit all dictionary entries, producing `Err` if the
485    /// iterator encounters [invalid] data.
486    ///
487    /// [invalid]: Self#Validation
488    pub fn iter_try(&self) -> impl Iterator<Item = Result<&'m str, ArrowError>> + '_ {
489        (0..self.len()).map(|i| self.get(i))
490    }

492    /// Iterates over all dictionary entries. When working with [unvalidated] input, consider
493    /// [`Self::iter_try`] to avoid panics due to invalid data.
494    ///
495    /// [unvalidated]: Self#Validation
496    ///
497    /// # Panics
498    ///
499    /// Panics if the underlying bytes are invalid. Use [`Self::iter_try`] for a
500    /// fallible version.
501    pub fn iter(&self) -> impl Iterator<Item = &'m str> + '_ {
502        self.iter_try()
503            .map(|result| result.expect("Invalid metadata dictionary entry"))
504    }

530            0b0000_0001, // header, offset_size_minus_one=0 and version=1
531            0x02,        // dictionary_size (2 strings)
532            0x00,
533            0x03,

733    /// Every entry of `metadata` must resolve to its own field id through the general name search
734    /// and through an owned (non-borrowed) copy of the name. The borrowed fast path must agree
735    /// whenever it applies, which is only for a sorted dictionary.
736    fn assert_lookups_agree(metadata: &VariantMetadata<'_>) {
737        for i in 0..metadata.len() {
738            let borrowed = metadata.get(i).unwrap();
739            let owned = borrowed.to_string();
740
741            assert_eq!(
742                metadata.borrowed_field_id(borrowed),
743                metadata.is_sorted().then_some(i as u32),
744                "borrowed lookup of {borrowed:?} (field id {i})"
745            );
746            assert_eq!(
747                metadata.get_entry(borrowed),
748                Some((i as u32, borrowed)),
749                "get_entry of borrowed {borrowed:?} (field id {i})"
750            );
751            assert_eq!(
752                metadata.get_entry(owned.as_str()),
753                Some((i as u32, borrowed)),
754                "get_entry of owned {borrowed:?} (field id {i})"
755            );
756            // An owned copy does not borrow from the dictionary, so it cannot take the fast path.
757            assert_eq!(metadata.borrowed_field_id(owned.as_str()), None);
758        }
759    }

761    /// Like [`assert_lookups_agree`], but tolerates the fast path declining an entry, as it may
762    /// for a dictionary whose offset array does not increase strictly.
763    fn assert_lookups_agree_or_decline(metadata: &VariantMetadata<'_>) {
764        for i in 0..metadata.len() {
765            let borrowed = metadata.get(i).unwrap();
766            if let Some(field_id) = metadata.borrowed_field_id(borrowed) {
767                assert_eq!(field_id, i as u32, "borrowed lookup of {borrowed:?}");
768            }
769            assert_eq!(
770                metadata.get_entry(borrowed),
771                Some((i as u32, borrowed)),
772                "get_entry of borrowed {borrowed:?} (field id {i})"
773            );
774        }
775    }

1013    /// exercised. `values` is the raw dictionary value region.
1014    fn raw_metadata(is_sorted: bool, offsets: &[u8], values: &[u8]) -> Vec<u8> {
1015        let mut bytes = vec![0x01 | (u8::from(is_sorted) << 4)];
1016        bytes.push(offsets.len() as u8 - 1); // dictionary_size
1017        bytes.extend_from_slice(offsets);
1018        bytes.extend_from_slice(values);
1019        bytes
