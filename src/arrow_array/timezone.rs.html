5//! Timezone for timestamp arrays
6//!
7//! Without additional features, [`Tz`] only supports fixed offsets such as `+09:00`. The
8//! `chrono-tz` and `jiff` features add named timezones from the [IANA time zone database], such
9//! as `America/Los_Angeles`:
10//!
11//! * `chrono-tz` resolves names with [chrono-tz], which precomputes each zone's transitions up to
12//!   2099 and keeps the last offset for any later instant, so daylight saving no longer changes it.
13//! * `jiff` resolves names with [jiff], which applies each zone's daylight saving rules to any
14//!   instant, as `java.time` and libc do.
15//!
16//! When both are enabled, `jiff` is used. Cargo unifies features across a build, so enabling
17//! `jiff` for any crate changes how named timezones are resolved for every crate in the build.
18//!
19//! The `jiff` feature is experimental. It is a step in moving arrow-rs from chrono to jiff, so what
20//! it covers and how it behaves may change, even between minor releases.
21//!
22//! Each feature bundles its own copy of the database, which may be a different release, so the two
23//! can also disagree wherever the database changed between releases. [`tzdb_version`] returns the
24//! release in use.
25//!
26//! [IANA time zone database]: https://www.iana.org/time-zones
27//! [chrono-tz]: https://docs.rs/chrono-tz
28//! [jiff]: https://docs.rs/jiff
29

34/// Returns the release of the [IANA time zone database] that resolves named timezones, such as
35/// `"2026c"`.
36///
37/// Returns `None` when neither the `chrono-tz` nor the `jiff` feature is enabled, as only fixed
38/// offsets are then supported.
39///
40/// [IANA time zone database]: https://www.iana.org/time-zones
41pub fn tzdb_version() -> Option<&'static str> {
42    #[cfg(feature = "jiff")]
43    {
44        Some(jiff_tzdb::VERSION)
45    }
46    #[cfg(all(feature = "chrono-tz", not(feature = "jiff")))]
47    {
48        Some(chrono_tz::IANA_TZDB_VERSION)
49    }
50    #[cfg(not(any(feature = "chrono-tz", feature = "jiff")))]
51    {
52        None
53    }
54}

56/// Parses a fixed offset of the form "+09:00", "-09" or "+0930"
57fn parse_fixed_offset(tz: &str) -> Option<FixedOffset> {
58    let bytes = tz.as_bytes();
59
60    let mut values = match bytes.len() {
61        // [+-]XX:XX
62        6 if bytes[3] == b':' => [bytes[1], bytes[2], bytes[4], bytes[5]],
63        // [+-]XXXX
64        5 => [bytes[1], bytes[2], bytes[3], bytes[4]],
65        // [+-]XX
66        3 => [bytes[1], bytes[2], b'0', b'0'],
67        _ => return None,
68    };
69    values.iter_mut().for_each(|x| *x = x.wrapping_sub(b'0'));
70    if values.iter().any(|x| *x > 9) {
71        return None;
72    }
73    let secs =
74        (values[0] * 10 + values[1]) as i32 * 60 * 60 + (values[2] * 10 + values[3]) as i32 * 60;
75
76    match bytes[0] {
77        b'+' => FixedOffset::east_opt(secs),
78        b'-' => FixedOffset::west_opt(secs),
79        _ => None,
80    }
81}

128    /// Builds a timezone from jiff-tzdb. `jiff::tz::db()` would read the host's zoneinfo files
129    /// instead if any crate in the build enabled jiff's `tzdb-zoneinfo` feature.
130    fn bundled_tz(name: &str) -> Option<JiffTimeZone> {
131        // As jiff's own bundled database does, so that UTC is not backed by TZif data
132        if name == "UTC" {
133            return Some(JiffTimeZone::UTC);
134        }
135        let (canonical_name, tzif) = jiff_tzdb::get(name)?;
136        JiffTimeZone::tzif(canonical_name, tzif).ok()
137    }

139    /// Resolves a name in the bundled database. As with chrono-tz, and unlike jiff, names are case
140    /// sensitive.
141    ///
142    /// [`Tz`] is `Copy` but [`JiffTimeZone`] is reference counted, so `Tz` holds a `'static`
143    /// reference into a table with one slot per bundled name, each resolved on first use.
144    fn named_tz(name: &str) -> Result<&'static NamedTz, ArrowError> {
145        type Slots = HashMap<&'static str, OnceLock<Option<NamedTz>>, ahash::RandomState>;
146        static SLOTS: OnceLock<Slots> = OnceLock::new();
147        let slots = SLOTS.get_or_init(|| {
148            jiff_tzdb::available()
149                // A placeholder for an unknown local timezone, which chrono-tz does not accept
150                .filter(|name| *name != "Factory")
151                .map(|name| (name, OnceLock::new()))
152                .collect()
153        });
154        slots
155            .get_key_value(name)
156            .and_then(|(name, slot)| {
157                slot.get_or_init(|| {
158                    let tz = bundled_tz(name)?;
159                    Some(NamedTz { name, tz })
160                })
161                .as_ref()
162            })
163            // The same message as with chrono-tz
164            .ok_or_else(|| {
165                ArrowError::ParseError(format!(
166                    "Invalid timezone \"{name}\": failed to parse timezone"
167                ))
168            })
169    }

208    fn fixed_offset(offset: jiff::tz::Offset) -> FixedOffset {
209        // The bundled database never holds an offset of a day or more, so this cannot fail
210        FixedOffset::east_opt(offset.seconds()).expect("timezone offset within a day")
211    }

213    /// The instant jiff should consult for a UTC datetime. jiff covers years -9999 to 9999;
214    /// anything outside that range takes the offset in force at the nearest covered instant.
215    fn jiff_timestamp(utc: &NaiveDateTime) -> Timestamp {
216        let seconds = utc.and_utc().timestamp();
217        Timestamp::from_second(seconds).unwrap_or(if seconds < 0 {
218            Timestamp::MIN
219        } else {
220            Timestamp::MAX
221        })
222    }

224    /// The civil datetime jiff should resolve for a local datetime, or `None` when the year is
225    /// outside jiff's range. Sub-second precision is irrelevant to the offset in force.
226    fn jiff_civil(local: &NaiveDateTime) -> Option<JiffDateTime> {
227        JiffDateTime::new(
228            i16::try_from(local.year()).ok()?,
229            local.month() as i8,
230            local.day() as i8,
231            local.hour() as i8,
232            local.minute() as i8,
233            local.second() as i8,
234            0,
235        )
236        .ok()
237    }

245    impl TimeZone for Tz {
246        type Offset = TzOffset;
247
248        fn from_offset(offset: &Self::Offset) -> Self {
249            offset.tz
250        }
251
252        fn offset_from_local_date(&self, local: &NaiveDate) -> LocalResult<Self::Offset> {
253            match self.0 {
254                TzInner::Offset(tz) => tz
255                    .offset_from_local_date(local)
256                    .map(|x| self.offset(x.fix())),
257                TzInner::Timezone(_) => {
258                    // As chrono-tz does, take an offset in force at the start or end of the day,
259                    // so that a day with a transition still resolves
260                    let earliest = self.offset_from_local_datetime(&local.and_time(NaiveTime::MIN));
261                    let latest =
262                        self.offset_from_local_datetime(&local.and_hms_opt(23, 59, 59).unwrap());
263                    match (earliest, latest) {
264                        (result @ LocalResult::Single(_), _) => result,
265                        (_, result @ LocalResult::Single(_)) => result,
266                        (LocalResult::Ambiguous(offset, _), _) => LocalResult::Single(offset),
267                        (_, LocalResult::Ambiguous(offset, _)) => LocalResult::Single(offset),
268                        (LocalResult::None, LocalResult::None) => LocalResult::None,
269                    }
270                }
271            }
272        }
273
274        fn offset_from_local_datetime(&self, local: &NaiveDateTime) -> LocalResult<Self::Offset> {
275            match self.0 {
276                TzInner::Offset(tz) => tz
277                    .offset_from_local_datetime(local)
278                    .map(|x| self.offset(x.fix())),
279                TzInner::Timezone(tz) => match jiff_civil(local) {
280                    // Outside jiff's range, treat the local datetime as if it were UTC. Either
281                    // way it takes the offset in force at the nearest covered instant.
282                    None => LocalResult::Single(self.offset_from_utc_datetime(local)),
283                    Some(dt) => match tz.tz.to_ambiguous_timestamp(dt).offset() {
284                        AmbiguousOffset::Unambiguous { offset } => {
285                            LocalResult::Single(self.offset(fixed_offset(offset)))
286                        }
287                        AmbiguousOffset::Gap { .. } => LocalResult::None,
288                        AmbiguousOffset::Fold { before, after } => LocalResult::Ambiguous(
289                            self.offset(fixed_offset(before)),
290                            self.offset(fixed_offset(after)),
291                        ),
292                    },
293                },
294            }
295        }
296
297        fn offset_from_utc_date(&self, utc: &NaiveDate) -> Self::Offset {
298            self.offset_from_utc_datetime(&utc.and_time(NaiveTime::MIN))
299        }
300
301        fn offset_from_utc_datetime(&self, utc: &NaiveDateTime) -> Self::Offset {
302            match self.0 {
303                TzInner::Offset(tz) => self.offset(tz.offset_from_utc_datetime(utc).fix()),
304                TzInner::Timezone(tz) => {
305                    self.offset(fixed_offset(tz.tz.to_offset(jiff_timestamp(utc))))
306                }
307            }
308        }
309    }

311    pub(super) fn new_gap_info(local: &NaiveDateTime, tz: &Tz) -> Option<GapInfo> {
312        let TzInner::Timezone(named) = tz.0 else {
313            return None;
314        };
315        // A local datetime outside jiff's range resolves to a single offset, so it is not in a gap
316        let civil = jiff_civil(local)?;
317        let AmbiguousOffset::Gap { before, after } =
318            named.tz.to_ambiguous_timestamp(civil).offset()
319        else {
320            return None;
321        };
322        // Reading a local datetime in the gap with the later offset gives an instant before the
323        // transition, so the next transition is the one that opened the gap
324        let from = after.to_timestamp(civil).ok()?;
325        let transition = named.tz.following(from).next()?;
326        let end = DateTime::from_timestamp(transition.timestamp().as_second(), 0)?;
327        let begin = end.naive_utc() + chrono::TimeDelta::seconds(before.seconds().into());
328        Some(GapInfo {
329            begin: Some((begin, tz.offset(fixed_offset(before)))),
330            end: Some(end.with_timezone(tz)),
331        })
332    }

691            #[test]
692            fn test_same_offsets() {
693                // 1900-01-01 to 2099-01-01, before chrono-tz stops applying daylight saving
694                const START: i64 = -2_208_988_800;
695                const END: i64 = 4_070_908_800;
696                const WEEK: usize = 7 * 24 * 60 * 60;
697
698                let mut different = BTreeSet::new();
699                for variant in chrono_tz::TZ_VARIANTS {
700                    let name = variant.name();
701                    let jiff: Tz = name.parse().unwrap();
702                    let TzInner::Timezone(named) = jiff.0 else {
703                        unreachable!("{name} is not an offset");
704                    };
705
706                    // Weekly samples find transitions that only one engine knows about
707                    let mut samples: Vec<_> = (START..END).step_by(WEEK).map(at).collect();
708
709                    // Probing each side of every transition covers gaps and folds
710                    let start = Timestamp::from_second(START).unwrap();
711                    let transitions = named.tz.following(start);
712                    for t in transitions.take_while(|t| t.timestamp().as_second() < END) {
713                        let instant = t.timestamp().as_second();
714                        let after = i64::from(t.offset().seconds());
715                        let before = i64::from(
716                            named
717                                .tz
718                                .to_offset(Timestamp::from_second(instant - 1).unwrap())
719                                .seconds(),
720                        );
721                        samples.extend(
722                            [
723                                instant - 1,
724                                instant,
725                                instant + before - 1,
726                                instant + before,
727                                instant + after - 1,
728                                instant + after,
729                            ]
730                            .map(at),
731                        );
732                    }
733
734                    // Years in which the two releases give different UTC offsets
735                    let changed: BTreeSet<_> = samples
736                        .iter()
737                        .filter(|dt| {
738                            variant.offset_from_utc_datetime(dt).fix()
739                                != jiff.offset_from_utc_datetime(dt).fix()
740                        })
741                        .map(|dt| dt.year())
742                        .collect();
743                    if !changed.is_empty() {
744                        different.insert(name);
745                    }
746
747                    // Every other year must also agree on local times and gaps. Neighbouring
748                    // years are skipped too, as a local reading can fall in them.
749                    for dt in &samples {
750                        if !(dt.year() - 1..=dt.year() + 1).any(|y| changed.contains(&y)) {
751                            assert!(agree(variant, jiff, dt), "{name} at {dt}, {}", releases());
752                        }
753                    }
754                }
755                let known = KNOWN_DIFFERENCES.iter().copied().collect();
756                assert_eq!(different, known, "{}", releases());
757            }
758        }
759    }

1054/// Represents the information of a gap.
1055///
1056/// This returns useful information that can be used when converting a local [`NaiveDateTime`]
1057/// to a timezone-aware [`DateTime`] with [`chrono::TimeZone::from_local_datetime`] and a gap
1058/// ([`chrono::LocalResult::None`]) is found.
1059///
1060/// # Examples
1061///
1062/// In the `America/New_York` timezone the clocks went forward one hour at `2026-03-08T02:00:00`.
1063/// The local time `2026-03-08T01:30:00` exists, but `2026-03-08T02:30:00` falls in the gap and
1064/// does not.
1065///
1066/// ```
1067/// # #[cfg(any(feature = "chrono-tz", feature = "jiff"))]
1068/// # {
1069/// use arrow_array::timezone::{GapInfo, Tz};
1070/// use chrono::{DateTime, FixedOffset, LocalResult, NaiveDateTime, Offset, TimeZone};
1071///
1072/// let tz: Tz = "America/New_York".parse().unwrap();
1073/// let expect_offset: FixedOffset = "-0500".parse().unwrap();
1074///
1075/// // A local time before the gap has a single offset, and is not in a gap.
1076/// let exists: NaiveDateTime = "2026-03-08T01:30:00".parse().unwrap();
1077/// let offset = tz.offset_from_local_datetime(&exists).single().unwrap();
1078/// assert_eq!(offset.fix(), expect_offset);
1079/// assert!(GapInfo::new(&exists, &tz).is_none());
1080///
1081/// // A local time inside the gap has no offset...
1082/// let not_exist: NaiveDateTime = "2026-03-08T02:30:00".parse().unwrap();
1083/// let offset = tz.offset_from_local_datetime(&not_exist);
1084/// assert!(matches!(offset, LocalResult::None));
1085///
1086/// // ...but `GapInfo` describes the gap it falls in.
1087/// let GapInfo { begin, end } = GapInfo::new(&not_exist, &tz).unwrap();
1088///
1089/// // The gap begins at 02:00 local time. The offset is the one in effect before the gap.
1090/// let (begin_datetime, begin_offset) = begin.unwrap();
1091/// let expect_begin: NaiveDateTime = "2026-03-08T02:00:00".parse().unwrap();
1092/// assert_eq!(begin_datetime, expect_begin);
1093/// assert_eq!(begin_offset.fix(), expect_offset);
1094///
1095/// // The gap ends at 03:00 local time, with the offset in effect after the gap.
1096/// let expect_end: DateTime<FixedOffset> = "2026-03-08T03:00:00-0400".parse().unwrap();
1097/// assert_eq!(end.unwrap(), expect_end.with_timezone(&tz));
1098/// # }
1099/// ```
1100#[derive(Debug, Clone, Copy)]
1101pub struct GapInfo {
1102    /// When available it contains information about the beginning of the gap.
1103    ///
1104    /// The time represents the first instant in which the gap starts.
1105    /// This means that it is the first instant that when used with
1106    /// [`chrono::TimeZone::from_local_datetime`] it will return
1107    /// [`chrono::LocalResult::None`].
1108    ///
1109    /// The offset represents the offset of the first instant before the gap.
1110    pub begin: Option<(NaiveDateTime, TzOffset)>,
1111
1112    /// When available it contains the first instant after the gap.
1113    pub end: Option<DateTime<Tz>>,
1114}
1115
1116impl GapInfo {
1117    /// Return information about a gap.
1118    ///
1119    /// It returns `None` if `local` is not in a gap for the current timezone. Timezones
1120    /// that represent fixed offsets contain no gaps and therefore always return `None`.
1121    ///
1122    /// If `local` is at the limits of the known timestamps the fields `begin` or `end` in
1123    /// [`GapInfo`] will be `None`.
1124    pub fn new(local: &NaiveDateTime, tz: &Tz) -> Option<Self> {
1125        private::new_gap_info(local, tz)
1126    }
1127}

1206        }

1209            Utc.with_ymd_and_hms(y, mo, d, h, mi, 0).unwrap()
1210        }
