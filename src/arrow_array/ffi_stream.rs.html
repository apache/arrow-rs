5//! Contains declarations to bind to the [C Stream Interface](https://arrow.apache.org/docs/format/CStreamInterface.html).
6//!
7//! This module has two main interfaces:
8//! One interface maps C ABI to native Rust types, i.e. convert c-pointers, c_char, to native rust.
9//! This is handled by [FFI_ArrowArrayStream].
10//!
11//! The second interface is used to import `FFI_ArrowArrayStream` as a Rust reader. A stream
12//! whose top-level type is `Struct` can be read as `RecordBatch` by [`RecordBatchStreamReader`];
13//! a stream of any type can be read as `ArrayRef` by [`ArrayStreamReader`].
14//!
15//! ```ignore
16//! # use std::fs::File;
17//! # use std::sync::Arc;
18//! # use arrow::error::Result;
19//! # use arrow::ffi_stream::{export_reader_into_raw, RecordBatchStreamReader, FFI_ArrowArrayStream};
20//! # use arrow::ipc::reader::FileReader;
21//! # use arrow::record_batch::RecordBatchReader;
22//! # fn main() -> Result<()> {
23//! // create an record batch reader natively
24//! let file = File::open("arrow_file").unwrap();
25//! let reader = Box::new(FileReader::try_new(file).unwrap());
26//!
27//! // export it
28//! let mut stream = FFI_ArrowArrayStream::empty();
29//! unsafe { export_reader_into_raw(reader, &mut stream) };
30//!
31//! // consumed and used by something else...
32//!
33//! // import it
34//! let stream_reader = unsafe { RecordBatchStreamReader::from_raw(&mut stream).unwrap() };
35//! let imported_schema = stream_reader.schema();
36//!
37//! let mut produced_batches = vec![];
38//! for batch in stream_reader {
39//!      produced_batches.push(batch.unwrap());
40//! }
41//! Ok(())
42//! }
43//! ```
44

83/// ABI-compatible struct for `ArrayStream` from C Stream Interface
84/// See <https://arrow.apache.org/docs/format/CStreamInterface.html#structure-definitions>
85/// This was created by bindgen
86#[repr(C)]
87#[derive(Debug)]
88pub struct FFI_ArrowArrayStream {
89    // Fields are intentionally private so safety guarantees can be upheld via
90    // explicit unsafe functions.
91    /// C function to get schema from the stream
92    get_schema: Option<unsafe extern "C" fn(arg1: *mut Self, out: *mut FFI_ArrowSchema) -> c_int>,
93    /// C function to get next array from the stream
94    get_next: Option<unsafe extern "C" fn(arg1: *mut Self, out: *mut FFI_ArrowArray) -> c_int>,
95    /// C function to get the error from last operation on the stream
96    get_last_error: Option<unsafe extern "C" fn(arg1: *mut Self) -> *const c_char>,
97    /// C function to release the stream
98    release: Option<unsafe extern "C" fn(arg1: *mut Self)>,
99    /// Private data used by the stream, owned by the release callback.
100    private_data: *mut c_void,
101}
102
103unsafe impl Send for FFI_ArrowArrayStream {}
104
105// callback used to drop [FFI_ArrowArrayStream] when it is exported.
106unsafe extern "C" fn release_stream(stream: *mut FFI_ArrowArrayStream) {
107    if stream.is_null() {
108        return;
109    }
110    let stream = unsafe { &mut *stream };
111
112    stream.get_schema = None;
113    stream.get_next = None;
114    stream.get_last_error = None;
115
116    let private_data = unsafe { Box::from_raw(stream.private_data.cast::<StreamPrivateData>()) };
117    drop(private_data);
118
119    stream.release = None;
120}

122struct StreamPrivateData {

124    last_error: Option<CString>,

127// The callback used to get array schema
128unsafe extern "C" fn get_schema(
129    stream: *mut FFI_ArrowArrayStream,
130    schema: *mut FFI_ArrowSchema,
131) -> c_int {
132    ExportedArrayStream { stream }.get_schema(schema)
133}
134
135// The callback used to get next array
136unsafe extern "C" fn get_next(
137    stream: *mut FFI_ArrowArrayStream,
138    array: *mut FFI_ArrowArray,
139) -> c_int {
140    ExportedArrayStream { stream }.get_next(array)
141}

143// The callback used to get the error from last operation on the `FFI_ArrowArrayStream`
144unsafe extern "C" fn get_last_error(stream: *mut FFI_ArrowArrayStream) -> *const c_char {
145    let mut ffi_stream = ExportedArrayStream { stream };
146    // The consumer should not take ownership of this string, we should return
147    // a const pointer to it.
148    match ffi_stream.get_last_error() {
149        Some(err_string) => err_string.as_ptr(),
150        None => std::ptr::null(),
151    }
152}

154impl Drop for FFI_ArrowArrayStream {
155    fn drop(&mut self) {
156        match self.release {
157            None => (),
158            Some(release) => unsafe { release(self) },
159        }
160    }
161}

163impl FFI_ArrowArrayStream {
164    /// Creates a new [`FFI_ArrowArrayStream`].
165    pub fn new(batch_reader: Box<dyn RecordBatchReader + Send>) -> Self {
166        let private_data = Box::new(StreamPrivateData {
167            batch_reader,
168            last_error: None,
169        });
170
171        Self {
172            get_schema: Some(get_schema),
173            get_next: Some(get_next),
174            get_last_error: Some(get_last_error),
175            release: Some(release_stream),
176            private_data: Box::into_raw(private_data).cast::<c_void>(),
177        }
178    }
179
180    /// Creates a new [`FFI_ArrowArrayStream`] from its raw parts.
181    ///
182    /// # Safety
183    ///
184    /// The caller takes responsibility for the [C stream interface] contract:
185    ///
186    /// * Each callback that is `Some` must be sound to invoke with a pointer to
187    ///   this struct, and must implement the semantics the C stream interface
188    ///   specifies for it.
189    /// * `private_data` must own everything the callbacks rely on, and must
190    ///   remain valid until `release` is called.
191    /// * `release`, if `Some`, is invoked by [`Drop`] with a pointer to this
192    ///   struct. It must free `private_data` exactly once and then set the
193    ///   release callback to `None` to mark the stream released.
194    /// * If `release` is `None` the stream is already released, so dropping it
195    ///   must not leak: nothing may be left for the callback to free.
196    ///
197    /// [C stream interface]: https://arrow.apache.org/docs/format/CStreamInterface.html
198    pub unsafe fn new_unchecked(
199        get_schema: Option<
200            unsafe extern "C" fn(arg1: *mut Self, out: *mut FFI_ArrowSchema) -> c_int,
201        >,
202        get_next: Option<unsafe extern "C" fn(arg1: *mut Self, out: *mut FFI_ArrowArray) -> c_int>,
203        get_last_error: Option<unsafe extern "C" fn(arg1: *mut Self) -> *const c_char>,
204        release: Option<unsafe extern "C" fn(arg1: *mut Self)>,
205        private_data: *mut c_void,
206    ) -> Self {
207        Self {
208            get_schema,
209            get_next,
210            get_last_error,
211            release,
212            private_data,
213        }
214    }
215
216    /// Takes ownership of the pointed to [`FFI_ArrowArrayStream`]
217    ///
218    /// This acts to [move] the data out of `raw_stream`, setting the release callback to NULL
219    ///
220    /// # Safety
221    ///
222    /// * `raw_stream` must be [valid] for reads and writes
223    /// * `raw_stream` must be properly aligned
224    /// * `raw_stream` must point to a properly initialized value of [`FFI_ArrowArrayStream`]
225    ///
226    /// [move]: https://arrow.apache.org/docs/format/CDataInterface.html#moving-an-array
227    /// [valid]: https://doc.rust-lang.org/std/ptr/index.html#safety
228    pub unsafe fn from_raw(raw_stream: *mut FFI_ArrowArrayStream) -> Self {
229        unsafe { std::ptr::replace(raw_stream, Self::empty()) }
230    }
231
232    /// Creates a new empty [FFI_ArrowArrayStream]. Used to import from the C Stream Interface.
233    pub fn empty() -> Self {
234        Self {
235            get_schema: None,
236            get_next: None,
237            get_last_error: None,
238            release: None,
239            private_data: std::ptr::null_mut(),
240        }
241    }
242
243    #[expect(clippy::unnecessary_safety_doc)]
244    /// Returns the producer-provided callback that writes this stream's schema, if any.
245    ///
246    /// # Safety
247    ///
248    /// The callback must be invoked with a pointer to the stream it was read from.
249    pub fn get_schema(
250        &self,
251    ) -> Option<unsafe extern "C" fn(arg1: *mut Self, out: *mut FFI_ArrowSchema) -> c_int> {
252        self.get_schema
253    }
254
255    #[expect(clippy::unnecessary_safety_doc)]
256    /// Returns the producer-provided callback that yields the next array, if any.
257    ///
258    /// # Safety
259    ///
260    /// The callback must be invoked with a pointer to the stream it was read from.
261    pub fn get_next(
262        &self,
263    ) -> Option<unsafe extern "C" fn(arg1: *mut Self, out: *mut FFI_ArrowArray) -> c_int> {
264        self.get_next
265    }
266
267    #[expect(clippy::unnecessary_safety_doc)]
268    /// Returns the producer-provided callback that describes the last error, if any.
269    ///
270    /// # Safety
271    ///
272    /// The callback must be invoked with a pointer to the stream it was read from.
273    /// The string it returns is owned by the producer and valid only until the next
274    /// call on the stream.
275    pub fn get_last_error(&self) -> Option<unsafe extern "C" fn(arg1: *mut Self) -> *const c_char> {
276        self.get_last_error
277    }
278
279    /// Returns the producer-provided release callback, if any.
280    pub fn release(&self) -> Option<unsafe extern "C" fn(arg1: *mut Self)> {
281        self.release
282    }
283
284    /// Returns the opaque producer-provided private data pointer.
285    pub fn private_data(&self) -> *mut c_void {
286        self.private_data
287    }
288
289    /// Replaces the release callback, returning the previous one.
290    ///
291    /// Lets a consumer wrap release: save the old callback, install its own, and
292    /// chain back on drop. See <https://github.com/apache/arrow-rs/issues/9771>.
293    ///
294    /// # Safety
295    ///
296    /// [`Drop`] calls this callback with a pointer to `self`. The new callback
297    /// must correctly release this stream (usually by chaining to the returned
298    /// one) and must match the [`FFI_ArrowArrayStream::private_data`] it reads.
299    /// A wrong callback is undefined behavior on drop.
300    pub unsafe fn set_release(
301        &mut self,
302        release: Option<unsafe extern "C" fn(arg1: *mut Self)>,
303    ) -> Option<unsafe extern "C" fn(arg1: *mut Self)> {
304        std::mem::replace(&mut self.release, release)
305    }
306
307    /// Replaces the private data pointer, returning the previous one.
308    ///
309    /// # Safety
310    ///
311    /// The old pointer is returned without being freed; the caller owns it from
312    /// here. The new pointer must match what the current
313    /// [`FFI_ArrowArrayStream::release`] callback expects.
314    pub unsafe fn set_private_data(&mut self, private_data: *mut c_void) -> *mut c_void {
315        std::mem::replace(&mut self.private_data, private_data)
316    }
317}

319struct ExportedArrayStream {
320    stream: *mut FFI_ArrowArrayStream,
321}
322
323impl ExportedArrayStream {
324    fn get_private_data(&mut self) -> &mut StreamPrivateData {
325        unsafe { &mut *(*self.stream).private_data.cast::<StreamPrivateData>() }
326    }
327
328    pub fn get_schema(&mut self, out: *mut FFI_ArrowSchema) -> i32 {
329        let private_data = self.get_private_data();
330        let reader = &private_data.batch_reader;
331
332        let schema = FFI_ArrowSchema::try_from(reader.schema().as_ref());
333
334        match schema {
335            Ok(schema) => {
336                unsafe { std::ptr::copy(addr_of!(schema), out, 1) };
337                std::mem::forget(schema);
338                0
339            }
340            Err(ref err) => {
341                private_data.last_error = Some(
342                    CString::new(err.to_string()).expect("Error string has a null byte in it."),
343                );
344                get_error_code(err)
345            }
346        }
347    }
348
349    pub fn get_next(&mut self, out: *mut FFI_ArrowArray) -> i32 {
350        let private_data = self.get_private_data();
351        let reader = &mut private_data.batch_reader;
352
353        match reader.next() {
354            None => {
355                // Marks ArrowArray released to indicate reaching the end of stream.
356                unsafe { std::ptr::write(out, FFI_ArrowArray::empty()) }
357                0
358            }
359            Some(next_batch) => {
360                if let Ok(batch) = next_batch {
361                    let struct_array = StructArray::from(batch);
362                    let array = FFI_ArrowArray::new(&struct_array.to_data());
363
364                    unsafe { std::ptr::write_unaligned(out, array) };
365                    0
366                } else {
367                    let err = &next_batch.unwrap_err();
368                    private_data.last_error = Some(
369                        CString::new(err.to_string()).expect("Error string has a null byte in it."),
370                    );
371                    get_error_code(err)
372                }
373            }
374        }
375    }
376
377    pub fn get_last_error(&mut self) -> Option<&CString> {
378        self.get_private_data().last_error.as_ref()
379    }
380}
381
382fn get_error_code(err: &ArrowError) -> i32 {
383    match err {
384        ArrowError::NotYetImplemented(_) => ENOSYS,
385        ArrowError::MemoryError(_) => ENOMEM,
386        ArrowError::IoError(_, _) => EIO,
387        _ => EINVAL,
388    }
389}

408/// Returns the producer's message for the last failed call on a `FFI_ArrowArrayStream`.
409///
410/// Returns `None` when the producer supplies no message, either because it installs no
411/// `get_last_error` callback or because that callback returns NULL: the C Stream Interface
412/// lets `get_last_error` return NULL when no detailed description is available.
413///
414/// # Safety
415///
416/// `stream_ptr` must point to a valid, not yet released [`FFI_ArrowArrayStream`], and the last
417/// operation on it must have returned an error: the C Stream Interface forbids calling
418/// `get_last_error` in any other case.
419unsafe fn producer_error(stream_ptr: *mut FFI_ArrowArrayStream) -> Option<String> {
420    let get_last_error = unsafe { (*stream_ptr).get_last_error }?;
421
422    let error_str = unsafe { get_last_error(stream_ptr) };
423    if error_str.is_null() {
424        return None;
425    }
426
427    Some(
428        unsafe { CStr::from_ptr(error_str) }
429            .to_string_lossy()
430            .into_owned(),
431    )
432}

434/// Gets schema from a raw pointer of `FFI_ArrowArrayStream`. This is used when constructing
435/// `RecordBatchStreamReader` to cache schema.
436fn get_stream_schema(stream_ptr: *mut FFI_ArrowArrayStream) -> Result<SchemaRef> {
437    let field = get_stream_field(stream_ptr)?;
438    match field.data_type() {
439        DataType::Struct(fields) => Ok(Arc::new(
440            Schema::new(fields.clone()).with_metadata(field.metadata().clone()),
441        )),
442        _ => Err(ArrowError::CDataInterface(
443            "Unable to interpret C data struct as a Schema".to_string(),
444        )),
445    }
446}
447
448/// Gets the field from a raw pointer of `FFI_ArrowArrayStream`. This is used when constructing
449/// `ArrayStreamReader` to cache the field.
450fn get_stream_field(stream_ptr: *mut FFI_ArrowArrayStream) -> Result<FieldRef> {
451    let mut schema = FFI_ArrowSchema::empty();
452
453    let Some(get_schema) = (unsafe { (*stream_ptr).get_schema }) else {
454        return Err(ArrowError::CDataInterface(
455            "input stream does not provide the required get_schema callback".to_string(),
456        ));
457    };
458
459    let ret_code = unsafe { get_schema(stream_ptr, &raw mut schema) };
460
461    if ret_code == 0 {
462        Ok(Arc::new(Field::try_from(&schema)?))
463    } else {
464        let message = format!("Cannot get schema from input stream. Error code: {ret_code}");
465        // SAFETY: `stream_ptr` is valid and unreleased, and the `get_schema` call above
466        // returned a non-zero code.
467        let message = match unsafe { producer_error(stream_ptr) } {
468            Some(producer_message) => format!("{message}. Producer error: {producer_message}"),
469            None => message,
470        };
471        Err(ArrowError::CDataInterface(message))
472    }
473}

475impl RecordBatchStreamReader {
476    /// Creates a new `RecordBatchStreamReader` from a `FFI_ArrowArrayStream`.
477    /// This is used to import from the C Stream Interface.
478    pub fn try_new(mut stream: FFI_ArrowArrayStream) -> Result<Self> {
479        if stream.release.is_none() {
480            return Err(ArrowError::CDataInterface(
481                "input stream is already released".to_string(),
482            ));
483        }
484
485        let schema = get_stream_schema(&raw mut stream)?;
486
487        Ok(Self { stream, schema })
488    }
489
490    /// Creates a new `RecordBatchStreamReader` from a raw pointer of `FFI_ArrowArrayStream`.
491    ///
492    /// Assumes that the pointer represents valid C Stream Interfaces.
493    /// This function copies the content from the raw pointer and cleans up it to prevent
494    /// double-dropping. The caller is responsible for freeing up the memory allocated for
495    /// the pointer.
496    ///
497    /// # Safety
498    ///
499    /// See [`FFI_ArrowArrayStream::from_raw`]
500    pub unsafe fn from_raw(raw_stream: *mut FFI_ArrowArrayStream) -> Result<Self> {
501        Self::try_new(unsafe { FFI_ArrowArrayStream::from_raw(raw_stream) })
502    }
503}

505impl Iterator for RecordBatchStreamReader {
506    type Item = Result<RecordBatch>;
507
508    fn next(&mut self) -> Option<Self::Item> {
509        let mut array = FFI_ArrowArray::empty();
510
511        let Some(get_next) = self.stream.get_next else {
512            return Some(Err(ArrowError::CDataInterface(
513                "input stream does not provide the required get_next callback".to_string(),
514            )));
515        };
516
517        let ret_code = unsafe { get_next(&raw mut self.stream, &raw mut array) };
518
519        if ret_code == 0 {
520            // The end of stream has been reached
521            if array.is_released() {
522                return None;
523            }
524
525            let result = unsafe {
526                from_ffi_and_data_type(array, DataType::Struct(self.schema().fields().clone()))
527            };
528            Some(result.and_then(|data| {
529                let len = data.len();
530                RecordBatch::try_new_with_options(
531                    self.schema.clone(),
532                    StructArray::from(data).into_parts().1,
533                    &RecordBatchOptions::new().with_row_count(Some(len)),
534                )
535            }))
536        } else {
537            let message =
538                format!("Cannot get next batch from input stream. Error code: {ret_code}");
539            // SAFETY: `self.stream` is valid and unreleased by construction, and the
540            // `get_next` call above returned a non-zero code.
541            let message = match unsafe { producer_error(&raw mut self.stream) } {
542                Some(producer_message) => format!("{message}. Producer error: {producer_message}"),
543                None => message,
544            };
545            Some(Err(ArrowError::CDataInterface(message)))
546        }
547    }
548}

556/// An [`ArrayReader`] which imports arrays from an [`FFI_ArrowArrayStream`].
557///
558/// [`RecordBatchStreamReader`] requires the stream's top-level type to be
559/// [`DataType::Struct`], which it unpacks into a [`RecordBatch`]. This reader has no such
560/// requirement: the stream's arrays may be of any data type.
561#[derive(Debug)]
562pub struct ArrayStreamReader {
563    stream: FFI_ArrowArrayStream,
564    field: FieldRef,
565}
566
567impl ArrayStreamReader {
568    /// Creates a new [`ArrayStreamReader`] from an [`FFI_ArrowArrayStream`].
569    /// This is used to import from the C Stream Interface.
570    pub fn try_new(mut stream: FFI_ArrowArrayStream) -> Result<Self> {
571        if stream.release.is_none() {
572            return Err(ArrowError::CDataInterface(
573                "input stream is already released".to_string(),
574            ));
575        }
576
577        let field = get_stream_field(&raw mut stream)?;
578
579        Ok(Self { stream, field })
580    }
581
582    /// Creates a new [`ArrayStreamReader`] from a raw pointer of [`FFI_ArrowArrayStream`].
583    ///
584    /// # Safety
585    ///
586    /// See [`FFI_ArrowArrayStream::from_raw`]
587    pub unsafe fn from_raw(raw_stream: *mut FFI_ArrowArrayStream) -> Result<Self> {
588        Self::try_new(unsafe { FFI_ArrowArrayStream::from_raw(raw_stream) })
589    }
590}
591
592impl Iterator for ArrayStreamReader {
593    type Item = Result<ArrayRef>;
594
595    fn next(&mut self) -> Option<Self::Item> {
596        let mut array = FFI_ArrowArray::empty();
597
598        let ret_code =
599            unsafe { self.stream.get_next.unwrap()(&raw mut self.stream, &raw mut array) };
600
601        if ret_code == 0 {
602            // The end of stream has been reached
603            if array.is_released() {
604                return None;
605            }
606
607            let result = unsafe { from_ffi_and_data_type(array, self.field.data_type().clone()) };
608            Some(result.map(make_array))
609        } else {
610            let message =
611                format!("Cannot get next array from input stream. Error code: {ret_code}");
612            // SAFETY: `self.stream` is valid and unreleased by construction, and the
613            // `get_next` call above returned a non-zero code.
614            let message = match unsafe { producer_error(&raw mut self.stream) } {
615                Some(producer_message) => format!("{message}. Producer error: {producer_message}"),
616                None => message,
617            };
618            Some(Err(ArrowError::CDataInterface(message)))
619        }
620    }
621}

625        &self.field
626    }

667    fn _test_round_trip_export(batch: RecordBatch, schema: Arc<Schema>) -> Result<()> {
668        let iter = Box::new(vec![batch.clone(), batch.clone()].into_iter().map(Ok)) as _;
669
670        let reader = Box::new(TestRecordBatchReader::new(schema.clone(), iter));
671
672        // Export a `RecordBatchReader` through `FFI_ArrowArrayStream`
673        let mut ffi_stream = FFI_ArrowArrayStream::new(reader);
674
675        // Get schema from `FFI_ArrowArrayStream`
676        let mut ffi_schema = FFI_ArrowSchema::empty();
677        let ret_code = unsafe { get_schema(&raw mut ffi_stream, &raw mut ffi_schema) };
678        assert_eq!(ret_code, 0);
679
680        let exported_schema = Schema::try_from(&ffi_schema).unwrap();
681        assert_eq!(&exported_schema, schema.as_ref());
682
683        // Get array from `FFI_ArrowArrayStream`
684        let mut produced_batches = vec![];
685        loop {
686            let mut ffi_array = FFI_ArrowArray::empty();
687            let ret_code = unsafe { get_next(&raw mut ffi_stream, &raw mut ffi_array) };
688            assert_eq!(ret_code, 0);
689
690            // The end of stream has been reached
691            if ffi_array.is_released() {
692                break;
693            }
694
695            let array = unsafe { from_ffi(ffi_array, &ffi_schema) }.unwrap();
696            let len = array.len();
697
698            let record_batch = RecordBatch::try_new_with_options(
699                SchemaRef::from(exported_schema.clone()),
700                StructArray::from(array).into_parts().1,
701                &RecordBatchOptions::new().with_row_count(Some(len)),
702            )
703            .unwrap();
704            produced_batches.push(record_batch);
705        }
706
707        assert_eq!(produced_batches, vec![batch.clone(), batch]);
708
709        Ok(())
710    }

712    fn _test_round_trip_import(batch: RecordBatch, schema: Arc<Schema>) -> Result<()> {
713        let iter = Box::new(vec![batch.clone(), batch.clone()].into_iter().map(Ok)) as _;
714
715        let reader = Box::new(TestRecordBatchReader::new(schema.clone(), iter));
716
717        // Import through `FFI_ArrowArrayStream` as `RecordBatchStreamReader`
718        let stream = FFI_ArrowArrayStream::new(reader);
719        let stream_reader = RecordBatchStreamReader::try_new(stream).unwrap();
720
721        let imported_schema = stream_reader.schema();
722        assert_eq!(imported_schema, schema);
723
724        let mut produced_batches = vec![];
725        for batch in stream_reader {
726            produced_batches.push(batch.unwrap());
727        }
728
729        assert_eq!(produced_batches, vec![batch.clone(), batch]);
730
731        Ok(())
732    }

949    // A consumer wraps the release callback with its own, then chains back to
950    // the original on drop. This is the same wrap-release pattern the
951    // release/private_data accessors exist for (#9771).
952    static STREAM_WRAPPER_RAN: std::sync::atomic::AtomicBool =
953        std::sync::atomic::AtomicBool::new(false);
954
955    struct StreamWrapperData {
956        original_release: Option<unsafe extern "C" fn(*mut FFI_ArrowArrayStream)>,
957        original_private_data: *mut c_void,
958    }
959
960    unsafe extern "C" fn wrapping_release(stream: *mut FFI_ArrowArrayStream) {
961        use std::sync::atomic::Ordering;
962        let stream = unsafe { &mut *stream };
963        let data = unsafe { Box::from_raw(stream.private_data().cast::<StreamWrapperData>()) };
964        STREAM_WRAPPER_RAN.store(true, Ordering::SeqCst);
965        unsafe { stream.set_release(data.original_release) };
966        unsafe { stream.set_private_data(data.original_private_data) };
967        if let Some(release) = stream.release() {
968            unsafe { release(stream) };
969        }
970    }

972    #[test]
973    fn test_wrap_release_callback() {
974        use std::sync::atomic::Ordering;
975
976        let batch_reader = Box::new(TestRecordBatchReader::new(
977            Arc::new(Schema::new(vec![Field::new("a", DataType::Int32, true)])),
978            Box::new(std::iter::empty()),
979        ));
980        let mut stream = FFI_ArrowArrayStream::new(batch_reader);
981
982        let data = Box::new(StreamWrapperData {
983            original_release: stream.release(),
984            original_private_data: stream.private_data(),
985        });
986        unsafe { stream.set_release(Some(wrapping_release)) };
987        unsafe { stream.set_private_data(Box::into_raw(data).cast::<c_void>()) };
988
989        drop(stream); // runs wrapping_release, which chains to the original
990        assert!(STREAM_WRAPPER_RAN.load(Ordering::SeqCst));
991    }

993    // A producer that exports a stream of plain arrays rather than record batches.
994    // `FFI_ArrowArrayStream::new` cannot express this, since it takes a
995    // `RecordBatchReader`. See <https://github.com/apache/arrow-rs/issues/6586>.
996    struct ArrayStreamPrivateData {
997        field: Field,
998        arrays: std::vec::IntoIter<Int32Array>,
999    }
1000
1001    unsafe extern "C" fn array_stream_get_schema(
1002        stream: *mut FFI_ArrowArrayStream,
1003        out: *mut FFI_ArrowSchema,
1004    ) -> c_int {
1005        let private_data = unsafe { &*(*stream).private_data().cast::<ArrayStreamPrivateData>() };
1006        let schema = FFI_ArrowSchema::try_from(&private_data.field).unwrap();
1007        unsafe { std::ptr::write(out, schema) };
1008        0
1009    }
1010
1011    unsafe extern "C" fn array_stream_get_next(
1012        stream: *mut FFI_ArrowArrayStream,
1013        out: *mut FFI_ArrowArray,
1014    ) -> c_int {
1015        let private_data =
1016            unsafe { &mut *(*stream).private_data().cast::<ArrayStreamPrivateData>() };
1017        match private_data.arrays.next() {
1018            // Marks ArrowArray released to indicate reaching the end of stream.
1019            None => unsafe { std::ptr::write(out, FFI_ArrowArray::empty()) },
1020            Some(array) => unsafe {
1021                std::ptr::write_unaligned(out, FFI_ArrowArray::new(&array.to_data()))
1022            },
1023        }
1024        0
1025    }
1026
1027    unsafe extern "C" fn array_stream_get_last_error(
1028        _stream: *mut FFI_ArrowArrayStream,
1029    ) -> *const c_char {
1030        std::ptr::null()
1031    }
1032
1033    unsafe extern "C" fn array_stream_release(stream: *mut FFI_ArrowArrayStream) {
1034        let private_data =
1035            unsafe { Box::from_raw((*stream).private_data().cast::<ArrayStreamPrivateData>()) };
1036        drop(private_data);
1037        // Clears every callback and marks the stream released, without dropping the
1038        // value being overwritten.
1039        unsafe { std::ptr::write(stream, FFI_ArrowArrayStream::empty()) };
1040    }
1041
1042    fn array_stream(field: Field, arrays: Vec<Int32Array>) -> FFI_ArrowArrayStream {
1043        let private_data = Box::new(ArrayStreamPrivateData {
1044            field,
1045            arrays: arrays.into_iter(),
1046        });
1047
1048        unsafe {
1049            FFI_ArrowArrayStream::new_unchecked(
1050                Some(array_stream_get_schema),
1051                Some(array_stream_get_next),
1052                Some(array_stream_get_last_error),
1053                Some(array_stream_release),
1054                Box::into_raw(private_data).cast::<c_void>(),
1055            )
1056        }
1057    }

1059    #[test]
1060    fn test_new_unchecked_exports_stream_of_arrays() {
1061        let field = Field::new("a", DataType::Int32, true);
1062        let arrays = vec![
1063            Int32Array::from(vec![1, 2, 3]),
1064            Int32Array::from(vec![4, 5]),
1065        ];
1066        let mut stream = array_stream(field.clone(), arrays.clone());
1067
1068        // Drive the stream through the callbacks the constructor stored, so a
1069        // field the constructor put in the wrong place would surface here.
1070        let get_schema_fn = stream.get_schema().unwrap();
1071        let get_next_fn = stream.get_next().unwrap();
1072        let get_last_error_fn = stream.get_last_error().unwrap();
1073
1074        let mut ffi_schema = FFI_ArrowSchema::empty();
1075        let ret_code = unsafe { get_schema_fn(&raw mut stream, &raw mut ffi_schema) };
1076        assert_eq!(ret_code, 0);
1077        assert_eq!(Field::try_from(&ffi_schema).unwrap(), field);
1078
1079        let mut produced = vec![];
1080        loop {
1081            let mut ffi_array = FFI_ArrowArray::empty();
1082            let ret_code = unsafe { get_next_fn(&raw mut stream, &raw mut ffi_array) };
1083            assert_eq!(ret_code, 0);
1084
1085            // The end of stream has been reached
1086            if ffi_array.is_released() {
1087                break;
1088            }
1089
1090            let data = unsafe { from_ffi(ffi_array, &ffi_schema) }.unwrap();
1091            produced.push(Int32Array::from(data));
1092        }
1093
1094        assert_eq!(produced, arrays);
1095
1096        // This producer never fails, so it reports no error message.
1097        assert!(unsafe { get_last_error_fn(&raw mut stream) }.is_null());
1098
1099        // Runs the stored release callback, freeing the private data exactly once.
1100        drop(stream);
1101    }
