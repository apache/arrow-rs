18use super::InProgressArray;
19use crate::concat::concat;
20use crate::filter::{FilterPredicate, FilterSelection};
21use crate::interleave::interleave;
22use arrow_array::{Array, ArrayRef, new_empty_array};
23use arrow_schema::{ArrowError, DataType};
24
25/// Bound retained input data while batching enough to amortize interleave.
26const MAX_PENDING_SOURCES: usize = 64;
27const MAX_PENDING_SOURCE_BYTES: usize = 64 * 1024 * 1024;
28
29#[derive(Debug)]
30pub(crate) struct InProgressListArray {
31    /// The current source for contiguous row copies.
32    source: Option<ArrayRef>,
33    /// The output array type, either List or LargeList.
34    data_type: DataType,
35    /// Materialized sparse groups and contiguous source slices in output order.
36    buffered_arrays: Vec<ArrayRef>,
37    /// Source arrays retained until the next sparse interleave.
38    sparse_sources: Vec<ArrayRef>,
39    /// Selected `(source index, row index)` pairs for `sparse_sources`.
40    sparse_indices: Vec<(usize, usize)>,
41    /// Total memory size reported by the retained sparse sources.
42    sparse_source_bytes: usize,
43    /// Whether buffered slices reference the current source's data.
44    source_data_referenced_in_buffers: bool,
45    /// Memory retained by buffered arrays that is not counted by `source`.
46    total_size_of_non_shared_buffers: usize,
47}
48
49impl InProgressListArray {
50    pub(crate) fn new(data_type: DataType) -> Self {
51        Self {
52            source: None,
53            data_type,
54            buffered_arrays: Vec::new(),
55            sparse_sources: Vec::new(),
56            sparse_indices: Vec::new(),
57            sparse_source_bytes: 0,
58            source_data_referenced_in_buffers: false,
59            total_size_of_non_shared_buffers: 0,
60        }
61    }
62
63    fn sparse_limit_reached(&self) -> bool {
64        self.sparse_sources.len() >= MAX_PENDING_SOURCES
65            || self.sparse_source_bytes >= MAX_PENDING_SOURCE_BYTES
66    }
67
68    fn flush_sparse(&mut self) -> Result<(), ArrowError> {
69        if self.sparse_indices.is_empty() {
70            self.sparse_sources.clear();
71            self.sparse_source_bytes = 0;
72            return Ok(());
73        }
74
75        let sources = self
76            .sparse_sources
77            .iter()
78            .map(|array| array.as_ref())
79            .collect::<Vec<_>>();
80        let array = interleave(&sources, &self.sparse_indices)?;
81        self.total_size_of_non_shared_buffers += array.get_array_memory_size();
82        self.buffered_arrays.push(array);
83        self.sparse_sources.clear();
84        self.sparse_indices.clear();
85        self.sparse_source_bytes = 0;
86        Ok(())
87    }
88}
89
90impl InProgressArray for InProgressListArray {
91    fn set_source(&mut self, source: Option<ArrayRef>) {
92        if let Some(old_source) = self.source.take()
93            && self.source_data_referenced_in_buffers
94        {
95            self.total_size_of_non_shared_buffers += old_source.get_array_memory_size();
96        }
97        self.source_data_referenced_in_buffers = false;
98        self.source = source;
99    }
100
101    fn copy_rows(&mut self, offset: usize, len: usize) -> Result<(), ArrowError> {
102        self.flush_sparse()?;
103        let source = self.source.as_ref().ok_or_else(|| {
104            ArrowError::InvalidArgumentError("InProgressListArray source not set".into())
105        })?;
106        self.source_data_referenced_in_buffers = true;
107        self.buffered_arrays.push(source.slice(offset, len));
108        Ok(())
109    }
110
111    fn copy_rows_by_filter_from(
112        &mut self,
113        source: ArrayRef,
114        filter: &FilterPredicate,
115    ) -> Result<(), ArrowError> {
116        let source_size = source.get_array_memory_size();
117        if !self.sparse_sources.is_empty()
118            && (self.sparse_limit_reached()
119                || self.sparse_source_bytes.saturating_add(source_size) > MAX_PENDING_SOURCE_BYTES)
120        {
121            self.flush_sparse()?;
122        }
123
124        let source_index = self.sparse_sources.len();
125        match filter.selection() {
126            FilterSelection::None => return Ok(()),
127            FilterSelection::All { len } => self
128                .sparse_indices
129                .extend((0..len).map(|index| (source_index, index))),
130            FilterSelection::Slices(slices) => slices.for_each(|(start, end)| {
131                self.sparse_indices
132                    .extend((start..end).map(|index| (source_index, index)));
133            }),
134            FilterSelection::Indices(indices) => indices.for_each(|index| {
135                self.sparse_indices.push((source_index, index));
136            }),
137        }
138        self.sparse_sources.push(source);
139        self.sparse_source_bytes += source_size;
140        if self.sparse_limit_reached() {
141            self.flush_sparse()?;
142        }
143        Ok(())
144    }
145
146    fn finish(&mut self) -> Result<ArrayRef, ArrowError> {
147        self.flush_sparse()?;
148        let result = match self.buffered_arrays.len() {
149            0 => new_empty_array(&self.data_type),
150            1 => self.buffered_arrays.pop().unwrap(),
151            _ => concat(
152                &self
153                    .buffered_arrays
154                    .iter()
155                    .map(|array| array.as_ref())
156                    .collect::<Vec<_>>(),
157            )?,
158        };
159        self.buffered_arrays.clear();
160        self.total_size_of_non_shared_buffers = 0;
161        self.source_data_referenced_in_buffers = false;
162        Ok(result)
163    }
164
165    fn size(&self) -> usize {
166        self.total_size_of_non_shared_buffers
167            + self.buffered_arrays.capacity() * size_of::<ArrayRef>()
168            + self.sparse_sources.capacity() * size_of::<ArrayRef>()
169            + self.sparse_indices.capacity() * size_of::<(usize, usize)>()
170            + self
171                .source
172                .as_ref()
173                .map_or(0, |source| source.get_array_memory_size())
174            + self.sparse_source_bytes
175    }
176}

178#[cfg(test)]
179mod tests {
180    use super::*;
181    use crate::filter::FilterBuilder;
182    use arrow_array::{BooleanArray, Int32Array, ListArray, StructArray, UInt32Array};
183    use arrow_buffer::{NullBuffer, OffsetBuffer};
184    use arrow_schema::Field;
185    use std::sync::Arc;
186
187    fn list_array() -> ArrayRef {
188        let field = Arc::new(Field::new("item", DataType::Int32, true));
189        let offsets = OffsetBuffer::from_lengths([2, 1, 0, 2, 0]);
190        let values = Arc::new(Int32Array::from(vec![
191            Some(1),
192            None,
193            Some(2),
194            Some(3),
195            Some(4),
196        ]));
197        let nulls = NullBuffer::from(vec![true, true, false, true, true]);
198        Arc::new(ListArray::new(field, offsets, values, Some(nulls)))
199    }
200
201    #[test]
202    fn sparse_copy_preserves_nulls_and_empty_lists() {
203        let source = list_array();
204        let filter = BooleanArray::from(vec![true, false, true, true, true]);
205        let predicate = FilterBuilder::new(&filter).optimize().build();
206        let expected = crate::filter::filter(source.as_ref(), &filter).unwrap();
207        let mut in_progress = InProgressListArray::new(source.data_type().clone());
208        in_progress
209            .copy_rows_by_filter_from(source, &predicate)
210            .unwrap();
211        assert_eq!(in_progress.sparse_sources.len(), 1);
212        assert!(in_progress.buffered_arrays.is_empty());
213        assert_eq!(in_progress.finish().unwrap().as_ref(), expected.as_ref());
214    }
215
216    #[test]
217    fn sparse_copy_nested_children() {
218        let child = StructArray::from(vec![(
219            Arc::new(Field::new("id", DataType::Int32, false)),
220            Arc::new(Int32Array::from_iter_values(0..6)) as ArrayRef,
221        )]);
222        let field = Arc::new(Field::new("item", child.data_type().clone(), true));
223        let source = Arc::new(ListArray::new(
224            field,
225            OffsetBuffer::from_lengths([2, 1, 3]),
226            Arc::new(child),
227            None,
228        )) as ArrayRef;
229        let filter = BooleanArray::from(vec![true, false, true]);
230        let predicate = FilterBuilder::new(&filter).optimize().build();
231        let expected = crate::take::take(
232            source.as_ref(),
233            &UInt32Array::from_iter_values([0, 2]),
234            None,
235        )
236        .unwrap();
237        let mut in_progress = InProgressListArray::new(source.data_type().clone());
238        in_progress
239            .copy_rows_by_filter_from(source, &predicate)
240            .unwrap();
241        let output = in_progress.finish().unwrap();
242        assert_eq!(output.as_ref(), expected.as_ref());
243    }
244
245    #[test]
246    fn preserves_order_when_sparse_and_contiguous_inputs_mix() {
247        let first = list_array();
248        let second = list_array();
249        let filter = BooleanArray::from(vec![true, false, false, true, false]);
250        let predicate = FilterBuilder::new(&filter).optimize().build();
251        let filtered = crate::filter::filter(first.as_ref(), &filter).unwrap();
252        let second_slice = second.slice(1, 2);
253        let expected = concat(&[filtered.as_ref(), second_slice.as_ref()]).unwrap();
254        let mut in_progress = InProgressListArray::new(first.data_type().clone());
255        in_progress
256            .copy_rows_by_filter_from(first, &predicate)
257            .unwrap();
258        in_progress.set_source(Some(second));
259        in_progress.copy_rows(1, 2).unwrap();
260
261        assert_eq!(in_progress.finish().unwrap().as_ref(), expected.as_ref());
262    }
263
264    #[test]
265    fn bounds_retained_sparse_sources() {
266        let source = list_array();
267        let filter = BooleanArray::from(vec![true, false, false, false, false]);
268        let predicate = FilterBuilder::new(&filter).optimize().build();
269        let mut in_progress = InProgressListArray::new(source.data_type().clone());
270
271        for _ in 0..MAX_PENDING_SOURCES {
272            in_progress
273                .copy_rows_by_filter_from(Arc::clone(&source), &predicate)
274                .unwrap();
275        }
276
277        assert!(in_progress.sparse_sources.is_empty());
278        assert_eq!(in_progress.sparse_source_bytes, 0);
279        assert_eq!(in_progress.buffered_arrays.len(), 1);
280        assert_eq!(in_progress.finish().unwrap().len(), MAX_PENDING_SOURCES);
281    }
