25/// Garbage collects a [DictionaryArray] by removing unreferenced values.
26///
27/// Returns a new [DictionaryArray] such that there are no values
28/// that are not referenced by at least one key. There may still be duplicate
29/// values.
30///
31/// See also [`garbage_collect_any_dictionary`] if you need to handle multiple dictionary types
32pub fn garbage_collect_dictionary<K: ArrowDictionaryKeyType>(
33    dictionary: &DictionaryArray<K>,
34) -> Result<DictionaryArray<K>, ArrowError> {
35    let keys = dictionary.keys();
36    let values = dictionary.values();
37
38    let mask = dictionary.occupancy();
39
40    // If no work to do, return the original dictionary
41    if mask.count_set_bits() == values.len() {
42        return Ok(dictionary.clone());
43    }
44
45    // Create a mapping from the old keys to the new keys, use a Vec for easy indexing
46    let mut key_remap = vec![K::Native::ZERO; values.len()];
47    for (new_idx, old_idx) in mask.set_indices().enumerate() {
48        key_remap[old_idx] =
49            K::Native::from_usize(new_idx).ok_or(ArrowError::DictionaryKeyOverflowError)?;
50    }
51
52    // ... and then build the new keys array
53    let new_keys = keys.unary(|key| {
54        key_remap
55            .get(key.as_usize())
56            .copied()
57            // nulls may be present in the keys, and they will have arbitrary value; we don't care
58            // and can safely return zero
59            .unwrap_or(K::Native::ZERO)
60    });
61
62    // Create a new values array by filtering using the mask
63    let values = filter(dictionary.values(), &BooleanArray::new(mask, None))?;
64
65    DictionaryArray::try_new(new_keys, values)
66}

68/// Equivalent to [`garbage_collect_dictionary`] but without requiring casting to a specific key type.
69pub fn garbage_collect_any_dictionary(
70    dictionary: &dyn AnyDictionaryArray,
71) -> Result<ArrayRef, ArrowError> {
72    let dictionary: &dyn Array = dictionary;
73    downcast_dictionary_array!(
74        dictionary => garbage_collect_dictionary(dictionary).map(|dict| Arc::new(dict) as ArrayRef),
75        _ => unreachable!("have a dictionary array")
76    )
77}

79/// A best effort interner that maintains a fixed number of buckets
80/// and interns keys based on their hash value
81///
82/// Hash collisions will result in replacement
83struct Interner<'a, V> {
84    state: RandomState,
85    buckets: Vec<Option<InternerBucket<'a, V>>>,
86    shift: u32,
87}
88
89/// A single bucket in [`Interner`].
90type InternerBucket<'a, V> = (Option<&'a [u8]>, V);
91
92impl<'a, V> Interner<'a, V> {
93    /// Capacity controls the number of unique buckets allocated within the Interner
94    ///
95    /// A larger capacity reduces the probability of hash collisions, and should be set
96    /// based on an approximation of the upper bound of unique values
97    fn new(capacity: usize) -> Self {
98        // Add additional buckets to help reduce collisions
99        let shift = (capacity as u64 + 128).leading_zeros();
100        let num_buckets = (u64::MAX >> shift) as usize;
101        let buckets = (0..num_buckets.saturating_add(1)).map(|_| None).collect();
102        Self {
103            // A fixed seed to ensure deterministic behaviour
104            state: RandomState::with_seeds(0, 0, 0, 0),
105            buckets,
106            shift,
107        }
108    }
109
110    fn intern<F: FnOnce() -> Result<V, E>, E>(
111        &mut self,
112        new: Option<&'a [u8]>,
113        f: F,
114    ) -> Result<&V, E> {
115        let hash = self.state.hash_one(new);
116        let bucket_idx = hash >> self.shift;
117        Ok(match &mut self.buckets[bucket_idx as usize] {
118            Some((current, v)) => {
119                if *current != new {
120                    *v = f()?;
121                    *current = new;
122                }
123                v
124            }
125            slot => &slot.insert((new, f()?)).1,
126        })
127    }
128}

137/// Performs a cheap, pointer-based comparison of two byte array
138///
139/// See [`ScalarBuffer::ptr_eq`]
140fn bytes_ptr_eq<T: ByteArrayType>(a: &dyn Array, b: &dyn Array) -> bool {
141    match (a.as_bytes_opt::<T>(), b.as_bytes_opt::<T>()) {
142        (Some(a), Some(b)) => {
143            let values_eq = a.values().ptr_eq(b.values()) && a.offsets().ptr_eq(b.offsets());
144            match (a.nulls(), b.nulls()) {
145                (Some(a), Some(b)) => values_eq && a.inner().ptr_eq(b.inner()),
146                (None, None) => values_eq,
147                _ => false,
148            }
149        }
150        _ => false,
151    }
152}

154/// Returns `true` if two child value arrays have the same layout and backing buffers.
155///
156/// This first checks whether the [`ArrayRef`]s point to the same allocation, then uses
157/// direct buffer comparisons for byte arrays and [`arrow_data::ArrayData::ptr_eq`] for
158/// other types.
159///
160/// This checks storage identity, not logical equality. Arrays with equal values in
161/// different buffers return `false`. Arrays with no allocated storage, such as two empty
162/// primitive arrays built from empty `Vec`s or two [`arrow_array::NullArray`]s of the same
163/// length, return `true`.
164pub(crate) fn child_values_ptr_eq(left: &ArrayRef, right: &ArrayRef) -> bool {
165    if Arc::ptr_eq(left, right) {
166        return true;
167    }
168
169    match left.data_type() {
170        DataType::Utf8 => bytes_ptr_eq::<Utf8Type>(left.as_ref(), right.as_ref()),
171        DataType::LargeUtf8 => bytes_ptr_eq::<LargeUtf8Type>(left.as_ref(), right.as_ref()),
172        DataType::Binary => bytes_ptr_eq::<BinaryType>(left.as_ref(), right.as_ref()),
173        DataType::LargeBinary => bytes_ptr_eq::<LargeBinaryType>(left.as_ref(), right.as_ref()),
174        _ => left.to_data().ptr_eq(&right.to_data()),
175    }
176}

181/// A weak heuristic of whether to merge dictionary values that aims to only
182/// perform the expensive merge computation when it is likely to yield at least
183/// some return over the naive approach used by MutableArrayData
184///
185/// `len` is the total length of the merged output
186///
187/// Returns `(should_merge, has_overflow)` where:
188/// - `should_merge`: whether dictionary values should be merged
189/// - `has_overflow`: whether the combined dictionary values would overflow the key type
190pub(crate) fn should_merge_dictionary_values<K: ArrowDictionaryKeyType>(
191    dictionaries: &[&DictionaryArray<K>],
192    len: usize,
193) -> (bool, bool) {
194    use DataType::*;
195    let first_values = dictionaries[0].values().as_ref();
196    let ptr_eq: PtrEq = match first_values.data_type() {
197        Utf8 => bytes_ptr_eq::<Utf8Type>,
198        LargeUtf8 => bytes_ptr_eq::<LargeUtf8Type>,
199        Binary => bytes_ptr_eq::<BinaryType>,
200        LargeBinary => bytes_ptr_eq::<LargeBinaryType>,
201        dt => {
202            if !dt.is_primitive() {
203                return (
204                    false,
205                    K::Native::from_usize(dictionaries.iter().map(|d| d.values().len()).sum())
206                        .is_none(),
207                );
208            }
209            |a, b| a.to_data().ptr_eq(&b.to_data())
210        }
211    };
212
213    let mut single_dictionary = true;
214    let mut total_values = first_values.len();
215    for dict in dictionaries.iter().skip(1) {
216        let values = dict.values().as_ref();
217        total_values += values.len();
218        if single_dictionary {
219            single_dictionary = ptr_eq(first_values, values)
220        }
221    }
222
223    let overflow = K::Native::from_usize(total_values).is_none();
224    let values_exceed_length = total_values >= len;
225
226    (
227        !single_dictionary && (overflow || values_exceed_length),
228        overflow,
229    )
230}

232/// Given an array of dictionaries and an optional key mask compute a values array
233/// containing referenced values, along with mappings from the [`DictionaryArray`]
234/// keys to the new keys within this values array. Best-effort will be made to ensure
235/// that the dictionary values are unique
236///
237/// This method is meant to be very fast and the output dictionary values
238/// may not be unique, unlike `GenericByteDictionaryBuilder` which is slower
239/// but produces unique values
240pub(crate) fn merge_dictionary_values<K: ArrowDictionaryKeyType>(
241    dictionaries: &[&DictionaryArray<K>],
242    masks: Option<&[BooleanBuffer]>,
243) -> Result<MergedDictionaries<K>, ArrowError> {
244    let mut num_values = 0;
245
246    let mut values_arrays = Vec::with_capacity(dictionaries.len());
247    let mut value_slices = Vec::with_capacity(dictionaries.len());
248
249    for (idx, dictionary) in dictionaries.iter().enumerate() {
250        let mask = masks.and_then(|m| m.get(idx));
251        let key_mask_owned;
252        let key_mask = match (dictionary.nulls(), mask) {
253            (Some(n), None) => Some(n.inner()),
254            (None, Some(n)) => Some(n),
255            (Some(n), Some(m)) => {
256                key_mask_owned = n.inner() & m;
257                Some(&key_mask_owned)
258            }
259            (None, None) => None,
260        };
261        let keys = dictionary.keys().values();
262        let values = dictionary.values().as_ref();
263        let values_mask = compute_values_mask(keys, key_mask, values.len());
264
265        let masked_values = get_masked_values(values, &values_mask);
266        num_values += masked_values.len();
267        value_slices.push(masked_values);
268        values_arrays.push(values)
269    }
270
271    // Map from value to new index
272    let mut interner = Interner::new(num_values);
273    // Interleave indices for new values array
274    let mut indices = Vec::with_capacity(num_values);
275
276    // Compute the mapping for each dictionary
277    let key_mappings = dictionaries
278        .iter()
279        .enumerate()
280        .zip(value_slices)
281        .map(|((dictionary_idx, dictionary), values)| {
282            let zero = K::Native::from_usize(0).unwrap();
283            let mut mapping = vec![zero; dictionary.values().len()];
284
285            for (value_idx, value) in values {
286                mapping[value_idx] =
287                    *interner.intern(value, || match K::Native::from_usize(indices.len()) {
288                        Some(idx) => {
289                            indices.push((dictionary_idx, value_idx));
290                            Ok(idx)
291                        }
292                        None => Err(ArrowError::DictionaryKeyOverflowError),
293                    })?;
294            }
295            Ok(mapping)
296        })
297        .collect::<Result<Vec<_>, ArrowError>>()?;
298
299    Ok(MergedDictionaries {
300        key_mappings,
301        values: interleave(&values_arrays, &indices)?,
302    })
303}

305/// Merges referenced entries from multiple dictionary values arrays.
306///
307/// `values_masks[i]` corresponds to `values_arrays[i]`, with each set bit identifying a
308/// value to retain. Returns the merged values and a mapping from each source value index
309/// to its index in the merged array. Mapping entries for unreferenced values are zero and
310/// must not be used.
311///
312/// Referenced values are gathered without hashing when their count fits in `K`'s key space,
313/// preserving duplicates across sources. Otherwise, values are deduplicated on a best-effort
314/// basis to avoid key overflow.
315pub(crate) fn merge_dictionary_values_with_masks<K: ArrowDictionaryKeyType>(
316    values_arrays: &[&dyn Array],
317    values_masks: &[BooleanBuffer],
318) -> Result<MergedDictionaries<K>, ArrowError> {
319    assert_eq!(values_arrays.len(), values_masks.len());
320    let num_values = values_masks.iter().map(BooleanBuffer::count_set_bits).sum();
321
322    // Dictionary values need not be unique. Avoid hashing when the referenced value count
323    // fits in `K`'s key space, and only deduplicate to avoid overflow.
324    if num_values == 0 || K::Native::from_usize(num_values - 1).is_some() {
325        let zero = K::Native::from_usize(0).unwrap();
326        let mut indices = Vec::with_capacity(num_values);
327        let key_mappings = values_arrays
328            .iter()
329            .zip(values_masks)
330            .enumerate()
331            .map(|(dictionary_idx, (values, mask))| {
332                let mut mapping = vec![zero; values.len()];
333                for value_idx in mask.set_indices() {
334                    mapping[value_idx] = K::Native::from_usize(indices.len()).unwrap();
335                    indices.push((dictionary_idx, value_idx));
336                }
337                mapping
338            })
339            .collect();
340        return Ok(MergedDictionaries {
341            key_mappings,
342            values: interleave(values_arrays, &indices)?,
343        });
344    }
345
346    let value_slices = values_arrays
347        .iter()
348        .zip(values_masks)
349        .map(|(values, mask)| get_masked_values(*values, mask))
350        .collect::<Vec<_>>();
351    let mut interner = Interner::new(num_values);
352    let mut indices = Vec::with_capacity(num_values);
353
354    let key_mappings = values_arrays
355        .iter()
356        .enumerate()
357        .zip(value_slices)
358        .map(|((dictionary_idx, value_array), values)| {
359            let zero = K::Native::ZERO;
360            let mut mapping = vec![zero; value_array.len()];
361
362            for (value_idx, value) in values {
363                mapping[value_idx] =
364                    *interner.intern(value, || match K::Native::from_usize(indices.len()) {
365                        Some(idx) => {
366                            indices.push((dictionary_idx, value_idx));
367                            Ok(idx)
368                        }
369                        None => Err(ArrowError::DictionaryKeyOverflowError),
370                    })?;
371            }
372            Ok(mapping)
373        })
374        .collect::<Result<Vec<_>, ArrowError>>()?;
375
376    Ok(MergedDictionaries {
377        key_mappings,
378        values: interleave(values_arrays, &indices)?,
379    })
380}

382/// Return a mask identifying the values that are referenced by keys in `dictionary`
383/// at the positions indicated by `selection`
384fn compute_values_mask<K: ArrowNativeType>(
385    keys: &ScalarBuffer<K>,
386    mask: Option<&BooleanBuffer>,
387    max_key: usize,
388) -> BooleanBuffer {
389    let mut builder = BooleanBufferBuilder::new(max_key);
390    builder.advance(max_key);
391
392    match mask {
393        Some(n) => n
394            .set_indices()
395            .for_each(|idx| builder.set_bit(keys[idx].as_usize(), true)),
396        None => keys
397            .iter()
398            .for_each(|k| builder.set_bit(k.as_usize(), true)),
399    }
400    builder.finish()
401}

403/// Process primitive array values to bytes
404fn masked_primitives_to_bytes<'a, T: ArrowPrimitiveType>(
405    array: &'a PrimitiveArray<T>,
406    mask: &BooleanBuffer,
407) -> Vec<(usize, Option<&'a [u8]>)>
408where
409    T::Native: ToByteSlice,
410{
411    let mut out = Vec::with_capacity(mask.count_set_bits());
412    let values = array.values();
413    for idx in mask.set_indices() {
414        out.push((
415            idx,
416            array.is_valid(idx).then_some(values[idx].to_byte_slice()),
417        ))
418    }
419    out
420}

422macro_rules! masked_primitive_to_bytes_helper {
423    ($t:ty, $array:expr, $mask:expr) => {
424        masked_primitives_to_bytes::<$t>($array.as_primitive(), $mask)
425    };
426}

428/// Return a Vec containing for each set index in `mask`, the index and byte value of that index
429fn get_masked_values<'a>(
430    array: &'a dyn Array,
431    mask: &BooleanBuffer,
432) -> Vec<(usize, Option<&'a [u8]>)> {
433    downcast_primitive! {
434        array.data_type() => (masked_primitive_to_bytes_helper, array, mask),
435        DataType::Utf8 => masked_bytes(array.as_string::<i32>(), mask),
436        DataType::LargeUtf8 => masked_bytes(array.as_string::<i64>(), mask),
437        DataType::Binary => masked_bytes(array.as_binary::<i32>(), mask),
438        DataType::LargeBinary => masked_bytes(array.as_binary::<i64>(), mask),
439        _ => unimplemented!("Dictionary merging for type {} is not implemented", array.data_type()),
440    }
441}

443/// Compute [`get_masked_values`] for a [`GenericByteArray`]
444///
445/// Note: this does not check the null mask and will return values contained in null slots
446fn masked_bytes<'a, T: ByteArrayType>(
447    array: &'a GenericByteArray<T>,
448    mask: &BooleanBuffer,
449) -> Vec<(usize, Option<&'a [u8]>)> {
450    let mut out = Vec::with_capacity(mask.count_set_bits());
451    for idx in mask.set_indices() {
452        out.push((
453            idx,
454            array.is_valid(idx).then_some(array.value(idx).as_ref()),
455        ))
456    }
457    out
458}

651    #[test]
652    fn test_merge_strings() {
653        let a = DictionaryArray::<Int32Type>::from_iter(["a", "b", "a", "b", "d", "c", "e"]);
654        let b = DictionaryArray::<Int32Type>::from_iter(["c", "f", "c", "d", "a", "d"]);
655        let merged = merge_dictionary_values(&[&a, &b], None).unwrap();
656
657        let values = as_string_array(merged.values.as_ref());
658        let actual: Vec<_> = values.iter().map(Option::unwrap).collect();
659        assert_eq!(&actual, &["a", "b", "d", "c", "e", "f"]);
660
661        assert_eq!(merged.key_mappings.len(), 2);
662        assert_eq!(&merged.key_mappings[0], &[0, 1, 2, 3, 4]);
663        assert_eq!(&merged.key_mappings[1], &[3, 5, 2, 0]);
664
665        let a_slice = a.slice(1, 4);
666        let merged = merge_dictionary_values(&[&a_slice, &b], None).unwrap();
667
668        let values = as_string_array(merged.values.as_ref());
669        let actual: Vec<_> = values.iter().map(Option::unwrap).collect();
670        assert_eq!(&actual, &["a", "b", "d", "c", "f"]);
671
672        assert_eq!(merged.key_mappings.len(), 2);
673        assert_eq!(&merged.key_mappings[0], &[0, 1, 2, 0, 0]);
674        assert_eq!(&merged.key_mappings[1], &[3, 4, 2, 0]);
675
676        // Mask out only ["b", "b", "d"] from a
677        let a_mask = BooleanBuffer::from_iter([false, true, false, true, true, false, false]);
678        let b_mask = BooleanBuffer::new_set(b.len());
679        let merged = merge_dictionary_values(&[&a, &b], Some(&[a_mask, b_mask])).unwrap();
680
681        let values = as_string_array(merged.values.as_ref());
682        let actual: Vec<_> = values.iter().map(Option::unwrap).collect();
683        assert_eq!(&actual, &["b", "d", "c", "f", "a"]);
684
685        assert_eq!(merged.key_mappings.len(), 2);
686        assert_eq!(&merged.key_mappings[0], &[0, 0, 1, 0, 0]);
687        assert_eq!(&merged.key_mappings[1], &[2, 3, 1, 4]);
688    }
