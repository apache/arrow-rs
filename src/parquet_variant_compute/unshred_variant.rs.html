32/// Removes all (nested) typed_value columns from a VariantArray by converting them back to binary
33/// variant and merging the resulting values back into the value column.
34///
35/// This function efficiently converts a shredded VariantArray back to an unshredded form where all
36/// data resides in the value column.
37///
38/// # Arguments
39/// * `array` - The VariantArray to unshred
40///
41/// # Returns
42/// A new VariantArray with all data in the value column and no typed_value column
43///
44/// # Errors
45/// - If the shredded data contains spec violations (e.g., field name conflicts)
46/// - If unsupported data types are encountered in typed_value columns
47pub fn unshred_variant(array: &VariantArray) -> Result<VariantArray> {
48    let nulls = array.nulls();
49    let metadata = array.metadata_column();
50    let value_col = array.value_column();
51    let typed_value_col = array.typed_value_column();
52
53    // Already unshredded: no data movement needed, but the output must annotate `value` as
54    // non-nullable per the spec. Inputs whose value-nulls are not all masked by the parent null
55    // buffer (spec-invalid "missing" rows) cannot be re-annotated and fall through to the row
56    // loop below, whose top-level builder materializes `Variant::Null` for such rows.
57    if typed_value_col.is_none() {
58        if value_field_is_non_nullable(array) {
59            return Ok(array.clone());
60        }
61        if value_nulls_are_masked(value_col, nulls) {
62            return Ok(VariantArray::from_parts_unshredded(
63                metadata.clone(),
64                value_col.clone(),
65                nulls.cloned(),
66            ));
67        }
68    }
69
70    // NOTE: None/None at top-level is technically invalid, but the shredding spec requires us to
71    // emit `Variant::Null` when a required value is missing.
72    let mut row_builder = UnshredVariantRowBuilder::try_new_opt(array.inner())?
73        .unwrap_or_else(UnshredVariantRowBuilder::null);
74
75    let mut value_builder = VariantValueArrayBuilder::new(array.len());
76    for i in 0..array.len() {
77        if array.is_null(i) {
78            value_builder.append_null();
79        } else {
80            let metadata_bytes = binary_array_value(metadata.as_ref(), i).ok_or_else(|| {
81                ArrowError::InvalidArgumentError(
82                    "metadata field must be a binary-like array".to_string(),
83                )
84            })?;
85            let metadata = VariantMetadata::try_new(metadata_bytes)?;
86            let mut value_builder = value_builder.builder_ext(&metadata);
87            row_builder.append_row(&mut value_builder, &metadata, i)?;
88        }
89    }
90
91    let value = value_builder.build()?;
92    Ok(VariantArray::from_parts_unshredded(
93        metadata.clone(),
94        Arc::new(value),
95        nulls.cloned(),
96    ))
97}

99fn value_field_is_non_nullable(array: &VariantArray) -> bool {
100    array
101        .inner()
102        .field_by_name("value")
103        .is_some_and(|field| !field.is_nullable())
104}

106/// Returns true if every null in `value` is masked by a parent null, i.e. the column may be
107/// annotated non-nullable.
108fn value_nulls_are_masked(value: &ArrayRef, parent_nulls: Option<&NullBuffer>) -> bool {
109    value.null_count() == 0
110        || parent_nulls
111            .zip(value.nulls())
112            .is_some_and(|(parent, value_nulls)| parent.contains(value_nulls))
113}

147impl<'a> UnshredVariantRowBuilder<'a> {
148    /// Creates an all-null row builder.
149    fn null() -> Self {
150        Self::Null(NullUnshredVariantBuilder)
151    }
152
153    /// Appends a single row at the given value index to the supplied builder.
154    fn append_row(
155        &mut self,
156        builder: &mut impl VariantBuilderExt,
157        metadata: &VariantMetadata,
158        index: usize,
159    ) -> Result<()> {
160        match self {
161            Self::PrimitiveInt8(b) => b.append_row(builder, metadata, index),
162            Self::PrimitiveInt16(b) => b.append_row(builder, metadata, index),
163            Self::PrimitiveInt32(b) => b.append_row(builder, metadata, index),
164            Self::PrimitiveInt64(b) => b.append_row(builder, metadata, index),
165            Self::PrimitiveFloat32(b) => b.append_row(builder, metadata, index),
166            Self::PrimitiveFloat64(b) => b.append_row(builder, metadata, index),
167            Self::Decimal32(b) => b.append_row(builder, metadata, index),
168            Self::Decimal64(b) => b.append_row(builder, metadata, index),
169            Self::Decimal128(b) => b.append_row(builder, metadata, index),
170            Self::PrimitiveDate32(b) => b.append_row(builder, metadata, index),
171            Self::PrimitiveTime64(b) => b.append_row(builder, metadata, index),
172            Self::TimestampMicrosecond(b) => b.append_row(builder, metadata, index),
173            Self::TimestampNanosecond(b) => b.append_row(builder, metadata, index),
174            Self::PrimitiveBoolean(b) => b.append_row(builder, metadata, index),
175            Self::PrimitiveString(b) => b.append_row(builder, metadata, index),
176            Self::PrimitiveStringView(b) => b.append_row(builder, metadata, index),
177            Self::PrimitiveLargeString(b) => b.append_row(builder, metadata, index),
178            Self::PrimitiveBinary(b) => b.append_row(builder, metadata, index),
179            Self::PrimitiveBinaryView(b) => b.append_row(builder, metadata, index),
180            Self::PrimitiveLargeBinary(b) => b.append_row(builder, metadata, index),
181            Self::PrimitiveUuid(b) => b.append_row(builder, metadata, index),
182            Self::List(b) => b.append_row(builder, metadata, index),
183            Self::LargeList(b) => b.append_row(builder, metadata, index),
184            Self::ListView(b) => b.append_row(builder, metadata, index),
185            Self::LargeListView(b) => b.append_row(builder, metadata, index),
186            Self::Struct(b) => b.append_row(builder, metadata, index),
187            Self::ValueOnly(b) => b.append_row(builder, metadata, index),
188            Self::Null(b) => b.append_row(builder, metadata, index),
189        }
190    }
191
192    /// Creates a new UnshredVariantRowBuilder from the `(value, typed_value)` pair of a shredded
193    /// variant struct. Returns None for the None/None case - caller decides how to handle based on
194    /// context.
195    fn try_new_opt(inner_struct: &'a StructArray) -> Result<Option<Self>> {
196        let value = if let Some(value_col) = inner_struct.column_by_name("value") {
197            validate_binary_array(value_col.as_ref(), "value")?;
198            Some(value_col)
199        } else {
200            None
201        };
202        let Some(typed_value) = inner_struct.column_by_name("typed_value") else {
203            // Copy the value across directly, if present. Else caller decides what to do.
204            return Ok(value.map(|v| Self::ValueOnly(ValueOnlyUnshredVariantBuilder::new(v))));
205        };
206
207        // Has typed_value -> determine type and create appropriate builder
208        macro_rules! primitive_builder {
209            ($enum_variant:ident, $cast_fn:ident) => {
210                Self::$enum_variant(UnshredPrimitiveRowBuilder::new(
211                    value,
212                    typed_value.$cast_fn(),
213                ))
214            };
215        }
216
217        let builder = match typed_value.data_type() {
218            DataType::Int8 => primitive_builder!(PrimitiveInt8, as_primitive),
219            DataType::Int16 => primitive_builder!(PrimitiveInt16, as_primitive),
220            DataType::Int32 => primitive_builder!(PrimitiveInt32, as_primitive),
221            DataType::Int64 => primitive_builder!(PrimitiveInt64, as_primitive),
222            DataType::Float32 => primitive_builder!(PrimitiveFloat32, as_primitive),
223            DataType::Float64 => primitive_builder!(PrimitiveFloat64, as_primitive),
224            DataType::Decimal32(p, s) if VariantDecimal4::is_valid_precision_and_scale(p, s) => {
225                Self::Decimal32(DecimalUnshredRowBuilder::new(value, typed_value, *s))
226            }
227            DataType::Decimal64(p, s) if VariantDecimal8::is_valid_precision_and_scale(p, s) => {
228                Self::Decimal64(DecimalUnshredRowBuilder::new(value, typed_value, *s))
229            }
230            DataType::Decimal128(p, s) if VariantDecimal16::is_valid_precision_and_scale(p, s) => {
231                Self::Decimal128(DecimalUnshredRowBuilder::new(value, typed_value, *s))
232            }
233            DataType::Decimal32(_, _)
234            | DataType::Decimal64(_, _)
235            | DataType::Decimal128(_, _)
236            | DataType::Decimal256(_, _) => {
237                return Err(ArrowError::InvalidArgumentError(format!(
238                    "{} is not a valid variant shredding type",
239                    typed_value.data_type()
240                )));
241            }
242            DataType::Date32 => primitive_builder!(PrimitiveDate32, as_primitive),
243            DataType::Time64(TimeUnit::Microsecond) => {
244                primitive_builder!(PrimitiveTime64, as_primitive)
245            }
246            DataType::Time64(time_unit) => {
247                return Err(ArrowError::InvalidArgumentError(format!(
248                    "Time64({time_unit}) is not a valid variant shredding type",
249                )));
250            }
251            DataType::Timestamp(TimeUnit::Microsecond, timezone) => Self::TimestampMicrosecond(
252                TimestampUnshredRowBuilder::new(value, typed_value, timezone.is_some()),
253            ),
254            DataType::Timestamp(TimeUnit::Nanosecond, timezone) => Self::TimestampNanosecond(
255                TimestampUnshredRowBuilder::new(value, typed_value, timezone.is_some()),
256            ),
257            DataType::Timestamp(time_unit, _) => {
258                return Err(ArrowError::InvalidArgumentError(format!(
259                    "Timestamp({time_unit}) is not a valid variant shredding type",
260                )));
261            }
262            DataType::Boolean => primitive_builder!(PrimitiveBoolean, as_boolean),
263            DataType::Utf8 => primitive_builder!(PrimitiveString, as_string),
264            DataType::Utf8View => primitive_builder!(PrimitiveStringView, as_string_view),
265            DataType::LargeUtf8 => primitive_builder!(PrimitiveLargeString, as_string),
266            DataType::Binary => primitive_builder!(PrimitiveBinary, as_binary),
267            DataType::BinaryView => primitive_builder!(PrimitiveBinaryView, as_binary_view),
268            DataType::LargeBinary => primitive_builder!(PrimitiveLargeBinary, as_binary),
269            DataType::FixedSizeBinary(16) => {
270                primitive_builder!(PrimitiveUuid, as_fixed_size_binary)
271            }
272            DataType::FixedSizeBinary(size) => {
273                return Err(ArrowError::InvalidArgumentError(format!(
274                    "FixedSizeBinary({size}) is not a valid variant shredding type",
275                )));
276            }
277            DataType::Struct(_) => Self::Struct(StructUnshredVariantBuilder::try_new(
278                value,
279                typed_value.as_struct(),
280            )?),
281            DataType::List(_) => Self::List(ListUnshredVariantBuilder::try_new(
282                value,
283                typed_value.as_list(),
284            )?),
285            DataType::LargeList(_) => Self::LargeList(ListUnshredVariantBuilder::try_new(
286                value,
287                typed_value.as_list(),
288            )?),
289            DataType::ListView(_) => Self::ListView(ListUnshredVariantBuilder::try_new(
290                value,
291                typed_value.as_list_view(),
292            )?),
293            DataType::LargeListView(_) => Self::LargeListView(ListUnshredVariantBuilder::try_new(
294                value,
295                typed_value.as_list_view(),
296            )?),
297            _ => {
298                return Err(ArrowError::NotYetImplemented(format!(
299                    "Unshredding not yet supported for type: {}",
300                    typed_value.data_type()
301                )));
302            }
303        };
304        Ok(Some(builder))
305    }
306}

323/// Builder for arrays that only have value column (already unshredded)
324struct ValueOnlyUnshredVariantBuilder<'a> {
325    value: &'a ArrayRef,
326}
327
328impl<'a> ValueOnlyUnshredVariantBuilder<'a> {
329    fn new(value: &'a ArrayRef) -> Self {
330        Self { value }
331    }
332
333    fn append_row(
334        &self,
335        builder: &mut impl VariantBuilderExt,
336        metadata: &VariantMetadata,
337        index: usize,
338    ) -> Result<()> {
339        if self.value.is_null(index) {
340            builder.append_null();
341        } else {
342            let value_bytes = binary_array_value(self.value.as_ref(), index).ok_or_else(|| {
343                ArrowError::InvalidArgumentError(
344                    "value field must be a binary-like array".to_string(),
345                )
346            })?;
347            let variant = Variant::try_new_with_metadata(metadata.clone(), value_bytes)?;
348            builder.append_value(variant);
349        }
350        Ok(())
351    }
352}

354/// Extension trait that directly adds row builder support for arrays that correspond to primitive
355/// variant types.
356trait AppendToVariantBuilder: Array {
357    fn append_to_variant_builder(
358        &self,
359        builder: &mut impl VariantBuilderExt,
360        index: usize,
361    ) -> Result<()>;
362}

364/// Macro that handles the unshredded case (typed_value is missing or NULL) and returns early if
365/// handled.  If not handled (shredded case), validates and returns the extracted value.
366macro_rules! handle_unshredded_case {
367    ($self:expr, $builder:expr, $metadata:expr, $index:expr, $partial_shredding:expr) => {{
368        let value = $self.value.as_ref().filter(|v| v.is_valid($index));
369        let value = value
370            .map(|v| {
371                let bytes = binary_array_value(v.as_ref(), $index).ok_or_else(|| {
372                    ArrowError::InvalidArgumentError(format!(
373                        "value field must be a binary-like array, instead got {}",
374                        v.data_type(),
375                    ))
376                })?;
377                Variant::try_new_with_metadata($metadata.clone(), bytes)
378            })
379            .transpose()?;
380
381        // If typed_value is null, handle unshredded case and return early
382        if $self.typed_value.is_null($index) {
383            match value {
384                Some(value) => $builder.append_value(value),
385                None => $builder.append_null(),
386            }
387            return Ok(());
388        }
389
390        // Only partial shredding allows value and typed_value to both be non-NULL
391        if !$partial_shredding && value.is_some() {
392            return Err(ArrowError::InvalidArgumentError(
393                "Invalid shredded variant: both value and typed_value are non-null".to_string(),
394            ));
395        }
396
397        // Return the extracted value for the partial shredded case
398        value
399    }};
400}

402/// Generic unshred builder that works with any Array implementing AppendToVariantBuilder
403struct UnshredPrimitiveRowBuilder<'a, T> {
404    value: Option<&'a ArrayRef>,
405    typed_value: &'a T,
406}
407
408impl<'a, T: AppendToVariantBuilder> UnshredPrimitiveRowBuilder<'a, T> {
409    fn new(value: Option<&'a ArrayRef>, typed_value: &'a T) -> Self {
410        Self { value, typed_value }
411    }
412
413    fn append_row(
414        &self,
415        builder: &mut impl VariantBuilderExt,
416        metadata: &VariantMetadata,
417        index: usize,
418    ) -> Result<()> {
419        handle_unshredded_case!(self, builder, metadata, index, false);
420
421        // If we get here, typed_value is valid and value is NULL
422        self.typed_value.append_to_variant_builder(builder, index)
423    }
424}
425
426// Macro to generate AppendToVariantBuilder implementations with optional value transformation
427macro_rules! impl_append_to_variant_builder {
428    ($array_type:ty $(, |$v:ident| $transform:expr)? ) => {
429        impl AppendToVariantBuilder for $array_type {
430            fn append_to_variant_builder(
431                &self,
432                builder: &mut impl VariantBuilderExt,
433                index: usize,
434            ) -> Result<()> {
435                let value = self.value(index);
436                $(
437                    let $v = value;
438                    let value = $transform;
439                )?
440                builder.append_value(value);
441                Ok(())
442            }
443        }
444    };
445}

484/// Trait for timestamp types to handle conversion to `DateTime<Utc>`

505/// Generic builder for timestamp types that handles timezone-aware conversion
506struct TimestampUnshredRowBuilder<'a, T: TimestampType> {
507    value: Option<&'a ArrayRef>,
508    typed_value: &'a PrimitiveArray<T>,
509    has_timezone: bool,
510}
511
512impl<'a, T: TimestampType> TimestampUnshredRowBuilder<'a, T> {
513    fn new(value: Option<&'a ArrayRef>, typed_value: &'a dyn Array, has_timezone: bool) -> Self {
514        Self {
515            value,
516            typed_value: typed_value.as_primitive(),
517            has_timezone,
518        }
519    }
520
521    fn append_row(
522        &self,
523        builder: &mut impl VariantBuilderExt,
524        metadata: &VariantMetadata,
525        index: usize,
526    ) -> Result<()> {
527        handle_unshredded_case!(self, builder, metadata, index, false);
528
529        // If we get here, typed_value is valid and value is NULL
530        let timestamp_value = self.typed_value.value(index);
531        let dt = T::to_datetime_utc(timestamp_value)?;
532        if self.has_timezone {
533            builder.append_value(dt);
534        } else {
535            builder.append_value(dt.naive_utc());
536        }
537        Ok(())
538    }
539}

541/// Generic builder for decimal unshredding
542struct DecimalUnshredRowBuilder<'a, A: DecimalType, V>
543where
544    V: VariantDecimalType<Native = A::Native>,
545{
546    value: Option<&'a ArrayRef>,
547    typed_value: &'a PrimitiveArray<A>,
548    scale: i8,
549    _phantom: PhantomData<V>,
550}
551
552impl<'a, A: DecimalType, V> DecimalUnshredRowBuilder<'a, A, V>
553where
554    V: VariantDecimalType<Native = A::Native>,
555{
556    fn new(value: Option<&'a ArrayRef>, typed_value: &'a dyn Array, scale: i8) -> Self {
557        Self {
558            value,
559            typed_value: typed_value.as_primitive(),
560            scale,
561            _phantom: PhantomData,
562        }
563    }
564
565    fn append_row(
566        &self,
567        builder: &mut impl VariantBuilderExt,
568        metadata: &VariantMetadata,
569        index: usize,
570    ) -> Result<()> {
571        handle_unshredded_case!(self, builder, metadata, index, false);
572
573        let raw = self.typed_value.value(index);
574        let variant = V::try_new_with_signed_scale(raw, self.scale)?;
575        builder.append_value(variant);
576        Ok(())
577    }
578}

580/// Builder for unshredding struct/object types with nested fields
581struct StructUnshredVariantBuilder<'a> {
582    value: Option<&'a ArrayRef>,
583    typed_value: &'a arrow::array::StructArray,
584    field_unshredders: IndexMap<&'a str, Option<UnshredVariantRowBuilder<'a>>>,
585}
586
587impl<'a> StructUnshredVariantBuilder<'a> {
588    fn try_new(value: Option<&'a ArrayRef>, typed_value: &'a StructArray) -> Result<Self> {
589        // Create unshredders for each field in constructor
590        let mut field_unshredders = IndexMap::new();
591        for (field, field_array) in typed_value.fields().iter().zip(typed_value.columns()) {
592            // Factory returns None for None/None case -- these are missing fields we should skip
593            let Some(field_array) = field_array.as_struct_opt() else {
594                return Err(ArrowError::InvalidArgumentError(format!(
595                    "Invalid shredded variant object field: expected Struct, got {}",
596                    field_array.data_type()
597                )));
598            };
599            let field_unshredder = UnshredVariantRowBuilder::try_new_opt(field_array)?;
600            field_unshredders.insert(field.name().as_ref(), field_unshredder);
601        }
602
603        Ok(Self {
604            value,
605            typed_value,
606            field_unshredders,
607        })
608    }
609
610    fn append_row(
611        &mut self,
612        builder: &mut impl VariantBuilderExt,
613        metadata: &VariantMetadata,
614        index: usize,
615    ) -> Result<()> {
616        let value = handle_unshredded_case!(self, builder, metadata, index, true);
617
618        // If we get here, typed_value is valid and value may or may not be valid
619        let mut object_builder = builder.try_new_object()?;
620
621        // Process typed fields (skip empty builders that indicate missing fields)
622        for (field_name, field_unshredder_opt) in &mut self.field_unshredders {
623            if let Some(field_unshredder) = field_unshredder_opt {
624                let mut field_builder = ObjectFieldBuilder::new(field_name, &mut object_builder);
625                field_unshredder.append_row(&mut field_builder, metadata, index)?;
626            }
627        }
628
629        // Process any unshredded fields (partial shredding)
630        if let Some(value) = value {
631            let Variant::Object(object) = value else {
632                return Err(ArrowError::InvalidArgumentError(
633                    "Expected object in value field for partially shredded struct".to_string(),
634                ));
635            };
636
637            for entry in object.iter_try() {
638                let (field_name, field_value) = entry?;
639                if self.field_unshredders.contains_key(field_name) {
640                    return Err(ArrowError::InvalidArgumentError(format!(
641                        "Field '{field_name}' appears in both typed_value and value",
642                    )));
643                }
644                object_builder.insert_bytes(field_name, field_value);
645            }
646        }
647
648        object_builder.finish();
649        Ok(())
650    }
651}

653/// Builder for unshredding list/array types with recursive element processing
654struct ListUnshredVariantBuilder<'a, L: ListLikeArray> {
655    value: Option<&'a ArrayRef>,
656    typed_value: &'a L,
657    element_unshredder: Box<UnshredVariantRowBuilder<'a>>,
658}
659
660impl<'a, L: ListLikeArray> ListUnshredVariantBuilder<'a, L> {
661    fn try_new(value: Option<&'a ArrayRef>, typed_value: &'a L) -> Result<Self> {
662        // Create a recursive unshredder for the list elements
663        // The element type comes from the values array of the list
664        let element_values = typed_value.values();
665
666        // For shredded lists, each element would be a ShreddedVariantFieldArray (struct)
667        // Extract value/typed_value from the element struct
668        let Some(element_values) = element_values.as_struct_opt() else {
669            return Err(ArrowError::InvalidArgumentError(format!(
670                "Invalid shredded variant array element: expected Struct, got {}",
671                element_values.data_type()
672            )));
673        };
674
675        // Create recursive unshredder for elements
676        //
677        // NOTE: A None/None array element is technically invalid, but the shredding spec
678        // requires us to emit `Variant::Null` when a required value is missing.
679        let element_unshredder = UnshredVariantRowBuilder::try_new_opt(element_values)?
680            .unwrap_or_else(UnshredVariantRowBuilder::null);
681
682        Ok(Self {
683            value,
684            typed_value,
685            element_unshredder: Box::new(element_unshredder),
686        })
687    }
688
689    fn append_row(
690        &mut self,
691        builder: &mut impl VariantBuilderExt,
692        metadata: &VariantMetadata,
693        index: usize,
694    ) -> Result<()> {
695        handle_unshredded_case!(self, builder, metadata, index, false);
696
697        // If we get here, typed_value is valid and value is NULL -- process the list elements
698        let mut list_builder = builder.try_new_list()?;
699        for element_index in self.typed_value.element_range(index) {
700            self.element_unshredder
701                .append_row(&mut list_builder, metadata, element_index)?;
702        }
703
704        list_builder.finish();
705        Ok(())
706    }
707}
