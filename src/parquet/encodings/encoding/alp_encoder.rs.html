18//! ALP (Adaptive Lossless floating-Point) encoder.
19//!
20//! Spec: <https://github.com/apache/parquet-format/blob/master/Encodings.md#adaptive-lossless-floating-point-alp--10>
21//!
22//! Values are buffered until the page is flushed, then encoded a vector at a
23//! time into the page layout that [`AlpDecoder`] reads back:
24//!
25//! ```text
26//! [AlpHeader][offsets][vector 0][vector 1]...[vector N-1]
27//! ```
28//!
29//! [`AlpDecoder`]: crate::encodings::decoding::alp_decoder::AlpDecoder

45/// Vectors are written at the spec's default size, the ALP paper's 1024.
46const VECTOR_SIZE: usize = 1 << ALP_DEFAULT_LOG_VECTOR_SIZE;
47
48/// Values sampled from a vector when estimating a candidate's encoded size.
49const SAMPLES_PER_VECTOR: usize = 256;
50
51/// Vectors sampled from the first page to build the column chunk's candidate set.
52const SAMPLE_VECTORS: usize = 8;
53
54/// Candidate `(exponent, factor)` pairs carried forward from the sampling pass.
55const MAX_COMBINATIONS: usize = 5;
56
57/// Consecutive non-improving candidates after which per-vector selection stops.
58const SAMPLING_EARLY_EXIT_THRESHOLD: usize = 4;
59
60/// Bits of overhead an exception costs: the value itself plus its `u16` position.
61fn exception_bits<F: AlpFloat>() -> u64 {
62    (F::Exact::WIDTH as u64 * 8) + 16
63}
64
65/// One ALP decimal-encoding candidate: `encoded = round(value * 10^e * 10^-f)`.
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67struct ExponentAndFactor {
68    exponent: u8,
69    factor: u8,
70}
71
72/// A candidate together with how often it won across the sampled vectors.
73#[derive(Debug, Clone, Copy)]
74struct Combination {
75    params: ExponentAndFactor,
76    num_appearances: u64,
77    estimated_size_bits: u64,
78}
79
80/// Sort key ranking candidates, larger is better: more appearances as a vector's
81/// best candidate, then smaller estimated size, then larger exponent, then
82/// larger factor.
83///
84/// The last two are tie-breaks taken from the ALP paper (Afroozeh et al., SIGMOD
85/// 2023, section 3.1.2), which prefers higher exponents and factors without
86/// justifying it further.
87fn rank(c: &Combination) -> (u64, Reverse<u64>, u8, u8) {
88    (
89        c.num_appearances,
90        Reverse(c.estimated_size_bits),
91        c.params.exponent,
92        c.params.factor,
93    )
94}
95
96fn is_better(c1: &Combination, c2: &Combination) -> bool {
97    rank(c1) > rank(c2)
98}

100/// Estimate, in bits, what `params` would cost on `sample`.
101///
102/// Encodes each value, checks whether it round-trips, and prices the result as
103/// `num_values * bit_width + num_exceptions * exception_bits`, where `bit_width`
104/// covers the FOR range of the values that did round-trip.
105///
106/// Returns `None` when `penalize_exceptions` is set and fewer than two values
107/// round-trip: such a candidate encodes almost everything as an exception, and
108/// its FOR range is meaningless.
109fn estimate_size_bits<F: AlpFloat>(
110    sample: &[F],
111    params: ExponentAndFactor,
112    penalize_exceptions: bool,
113) -> Option<u64> {
114    let encode_scale = F::encode_scale(params.exponent, params.factor);
115    let decode_scale = F::decode_scale(params.exponent, params.factor);
116
117    let mut num_exceptions = 0u64;
118    let mut min = None;
119    let mut max = None;
120
121    for &value in sample {
122        let encoded = value.encode_value(encode_scale);
123        if F::decode_value(encoded, decode_scale) == value {
124            min = Some(min.map_or(encoded, |m: <F::Exact as AlpExact>::Signed| m.min(encoded)));
125            max = Some(max.map_or(encoded, |m: <F::Exact as AlpExact>::Signed| m.max(encoded)));
126        } else {
127            num_exceptions += 1;
128        }
129    }
130
131    let num_values = sample.len() as u64;
132    let num_non_exceptions = num_values - num_exceptions;
133    if penalize_exceptions && num_non_exceptions < 2 {
134        return None;
135    }
136
137    // With nothing to frame, there is no packed section at all: every value is
138    // stored as an exception.
139    let (Some(min), Some(max)) = (min, max) else {
140        return Some(num_values * exception_bits::<F>());
141    };
142
143    let range =
144        F::Exact::reinterpret_from_signed(max).wrapping_sub(F::Exact::reinterpret_from_signed(min));
145    let bit_width = u64::from(num_required_bits(range.to_u64()));
146
147    Some(num_values * bit_width + num_exceptions * exception_bits::<F>())
148}

150/// Sample every `n`th value so the whole span is represented, capped at
151/// [`SAMPLES_PER_VECTOR`] values.
152fn sample_values<F: AlpFloat>(values: &[F], out: &mut Vec<F>) {
153    out.clear();
154    let stride = values.len().div_ceil(SAMPLES_PER_VECTOR).max(1);
155    out.extend(values.iter().step_by(stride).copied());
156}
157
158/// Build the column chunk's candidate set: the top [`MAX_COMBINATIONS`] pairs by
159/// how often they win across sampled vectors.
160///
161/// This is the first level of the ALP paper's two-level sampling. Each sampled
162/// vector is searched exhaustively (66 pairs for `f32`, 190 for `f64`); the
163/// winners are tallied, and the most frequent are carried forward so that each
164/// vector only has to choose among a handful of candidates.
165fn build_preset<F: AlpFloat>(values: &[F]) -> Vec<ExponentAndFactor> {
166    let num_vectors = values.len().div_ceil(VECTOR_SIZE);
167    let vector_stride = num_vectors.div_ceil(SAMPLE_VECTORS).max(1);
168
169    let mut sample = Vec::with_capacity(SAMPLES_PER_VECTOR);
170    let mut tally: Vec<Combination> = Vec::new();
171
172    for vector in values.chunks(VECTOR_SIZE).step_by(vector_stride) {
173        sample_values(vector, &mut sample);
174
175        // Start from the worst case - every value an exception, unpacked - so
176        // that any candidate that manages to encode anything beats it.
177        let mut best = Combination {
178            params: ExponentAndFactor {
179                exponent: F::MAX_EXPONENT,
180                factor: F::MAX_EXPONENT,
181            },
182            num_appearances: 0,
183            estimated_size_bits: sample.len() as u64
184                * (exception_bits::<F>() + F::Exact::WIDTH as u64 * 8),
185        };
186
187        for exponent in 0..=F::MAX_EXPONENT {
188            for factor in 0..=exponent {
189                let params = ExponentAndFactor { exponent, factor };
190                let Some(estimated_size_bits) = estimate_size_bits(&sample, params, true) else {
191                    continue;
192                };
193                let candidate = Combination {
194                    params,
195                    num_appearances: 0,
196                    estimated_size_bits,
197                };
198                if is_better(&candidate, &best) {
199                    best = candidate;
200                }
201            }
202        }
203
204        match tally.iter_mut().find(|c| c.params == best.params) {
205            Some(existing) => existing.num_appearances += 1,
206            None => tally.push(Combination {
207                num_appearances: 1,
208                ..best
209            }),
210        }
211    }
212
213    // Size estimates from different vectors are not comparable with each other,
214    // so only the win counts rank the candidate set.
215    for combination in &mut tally {
216        combination.estimated_size_bits = 0;
217    }
218    tally.sort_by_key(|c| Reverse(rank(c)));
219    tally.truncate(MAX_COMBINATIONS);
220
221    if tally.is_empty() {
222        // An empty page has nothing to sample; any valid pair will do.
223        return vec![ExponentAndFactor {
224            exponent: 0,
225            factor: 0,
226        }];
227    }
228    tally.into_iter().map(|c| c.params).collect()
229}

231/// Pick the candidate that encodes `vector` smallest.
232///
233/// This is the second level of the two-level sampling: only the chunk's
234/// candidates are tried, against a sample of the vector rather than all of it.
235fn select_params<F: AlpFloat>(
236    vector: &[F],
237    preset: &[ExponentAndFactor],
238    sample: &mut Vec<F>,
239) -> ExponentAndFactor {
240    if preset.len() == 1 {
241        return preset[0];
242    }
243
244    sample_values(vector, sample);
245
246    let mut best = preset[0];
247    let mut best_size_bits = u64::MAX;
248    let mut worse_in_a_row = 0;
249
250    for &params in preset {
251        let Some(size_bits) = estimate_size_bits(sample, params, false) else {
252            continue;
253        };
254        if size_bits >= best_size_bits {
255            worse_in_a_row += 1;
256            if worse_in_a_row == SAMPLING_EARLY_EXIT_THRESHOLD {
257                break;
258            }
259            continue;
260        }
261        best = params;
262        best_size_bits = size_bits;
263        worse_in_a_row = 0;
264    }
265    best
266}

268/// Reusable per-vector buffers, kept across vectors and pages so encoding a page
269/// does not allocate per vector.
270struct Scratch<F: AlpFloat> {
271    /// Decimal-encoded integers, overwritten in place with their FOR deltas.
272    encoded: Vec<<F::Exact as AlpExact>::Signed>,
273    /// Per-value exception flag (`1` = round-trip failed). A byte, not a bitset,
274    /// so the map loop's per-lane store stays vectorizable.
275    exc_mask: Vec<u8>,
276    exception_positions: Vec<u16>,
277    exception_values: Vec<F>,
278    sample: Vec<F>,
279}
280
281impl<F: AlpFloat> Scratch<F> {
282    fn new() -> Self {
283        Self {
284            encoded: Vec::new(),
285            exc_mask: Vec::new(),
286            exception_positions: Vec::new(),
287            exception_values: Vec::new(),
288            sample: Vec::new(),
289        }
290    }
291
292    fn estimated_memory_size(&self) -> usize {
293        self.encoded.capacity() * std::mem::size_of::<<F::Exact as AlpExact>::Signed>()
294            + self.exc_mask.capacity()
295            + self.exception_positions.capacity() * std::mem::size_of::<u16>()
296            + self.exception_values.capacity() * std::mem::size_of::<F>()
297            + self.sample.capacity() * std::mem::size_of::<F>()
298    }
299}

301/// Encode one vector and append it to `out`:
302/// `[AlpInfo][ForInfo][PackedValues][ExceptionPositions][ExceptionValues]`.
303fn encode_vector<F: AlpFloat>(
304    values: &[F],
305    params: ExponentAndFactor,
306    scratch: &mut Scratch<F>,
307    out: &mut Vec<u8>,
308) -> Result<()> {
309    let encode_scale = F::encode_scale(params.exponent, params.factor);
310    let decode_scale = F::decode_scale(params.exponent, params.factor);
311
312    let Scratch {
313        encoded,
314        exc_mask,
315        exception_positions,
316        exception_values,
317        ..
318    } = scratch;
319
320    let zero = F::Exact::default().reinterpret_as_signed();
321    let n_values = values.len();
322    encoded.resize(n_values, zero);
323    exc_mask.resize(n_values, 0);
324    exception_values.clear();
325
326    // Comparing the *decoded* value to the input (not the input to itself) flags
327    // -0.0, which encodes to +0.0.
328    for ((&value, enc_slot), mask_slot) in values
329        .iter()
330        .zip(encoded.iter_mut())
331        .zip(exc_mask.iter_mut())
332    {
333        let encoded_value = value.encode_value(encode_scale);
334        *enc_slot = encoded_value;
335        *mask_slot = u8::from(F::decode_value(encoded_value, decode_scale) != value);
336    }
337
338    // Branchless compaction: always write the index, advance only when flagged.
339    exception_positions.resize(n_values, 0);
340    let mut num_exceptions_usize = 0usize;
341    for (idx, &is_exception) in exc_mask.iter().enumerate() {
342        exception_positions[num_exceptions_usize] = idx as u16;
343        num_exceptions_usize += usize::from(is_exception);
344    }
345    exception_positions.truncate(num_exceptions_usize);
346
347    let num_exceptions = u16::try_from(exception_positions.len()).map_err(|_| {
348        general_err!(
349            "Invalid ALP vector: {} exceptions exceeds u16::MAX",
350            exception_positions.len()
351        )
352    })?;
353
354    // Fill each exception's packed slot with a real value (not the sentinel) so
355    // the FOR range - and thus the bit width - stays tight; the true value goes
356    // in the exception section.
357    let placeholder = first_non_exception_value::<F>(encoded, exception_positions);
358    for &position in exception_positions.iter() {
359        exception_values.push(values[position as usize]);
360        encoded[position as usize] = placeholder;
361    }
362
363    let min = encoded.iter().copied().min().unwrap_or(zero);
364    let max = encoded.iter().copied().max().unwrap_or(zero);
365
366    let frame_of_reference = F::Exact::reinterpret_from_signed(min);
367    let range = F::Exact::reinterpret_from_signed(max).wrapping_sub(frame_of_reference);
368    let bit_width = num_required_bits(range.to_u64());
369
370    let alp_info = AlpInfo {
371        exponent: params.exponent,
372        factor: params.factor,
373        num_exceptions,
374    };
375    let for_info = ForInfo::<F::Exact> {
376        frame_of_reference,
377        bit_width,
378    };
379    alp_info.extend_serialized(out);
380    for_info.extend_serialized(out);
381
382    // PackedValues: FOR deltas, LSB-first, as the spec requires. A zero bit
383    // width means every value equals the frame of reference, so nothing is
384    // stored.
385    if bit_width > 0 {
386        let mut writer = BitWriter::new_from_buf(std::mem::take(out));
387        for &encoded_value in encoded.iter() {
388            let delta =
389                F::Exact::reinterpret_from_signed(encoded_value).wrapping_sub(frame_of_reference);
390            writer.put_value(delta.to_u64(), bit_width as usize);
391        }
392        // Pads to a byte boundary, giving exactly the ceil(n * bit_width / 8)
393        // bytes the decoder derives from the metadata.
394        *out = writer.consume();
395    }
396
397    for &position in exception_positions.iter() {
398        out.extend_from_slice(&position.to_le_bytes());
399    }
400    for &value in exception_values.iter() {
401        value.to_exact_bits().extend_le_bytes(out);
402    }
403
404    Ok(())
405}

407/// The encoded integer of the first value that is not an exception, or zero if
408/// every value is one. `exception_positions` is ascending, so the first index it
409/// skips over is the first non-exception.
410fn first_non_exception_value<F: AlpFloat>(
411    encoded: &[<F::Exact as AlpExact>::Signed],
412    exception_positions: &[u16],
413) -> <F::Exact as AlpExact>::Signed {
414    let mut candidate = 0usize;
415    for &position in exception_positions {
416        if position as usize != candidate {
417            break;
418        }
419        candidate += 1;
420    }
421    encoded
422        .get(candidate)
423        .copied()
424        .unwrap_or_else(|| F::Exact::default().reinterpret_as_signed())
425}

427/// Encode `values` as one ALP page.
428fn encode_page<F: AlpFloat>(
429    values: &[F],
430    preset: &[ExponentAndFactor],
431    scratch: &mut Scratch<F>,
432) -> Result<Vec<u8>> {
433    let header = AlpHeader {
434        compression_mode: ALP_COMPRESSION_MODE,
435        integer_encoding: ALP_INTEGER_ENCODING_FOR_BIT_PACK,
436        vector_size: VECTOR_SIZE,
437        num_elements: values.len(),
438    };
439    let num_vectors = header.num_vectors();
440
441    let mut page = Vec::with_capacity(ALP_HEADER_SIZE + num_vectors * 4 + values.len() * 4);
442    page.extend_from_slice(&header.serialize()?);
443
444    // Offsets are only known once each vector is encoded, so leave room and
445    // backfill.
446    let offsets_start = page.len();
447    page.resize(offsets_start + num_vectors * std::mem::size_of::<u32>(), 0);
448
449    for (idx, vector) in values.chunks(VECTOR_SIZE).enumerate() {
450        // Offsets are relative to the start of the page body, which follows the
451        // fixed-size header.
452        let offset = u32::try_from(page.len() - ALP_HEADER_SIZE)
453            .map_err(|_| general_err!("Invalid ALP page: body exceeds u32 offset range"))?;
454        let offset_at = offsets_start + idx * std::mem::size_of::<u32>();
455        page[offset_at..offset_at + 4].copy_from_slice(&offset.to_le_bytes());
456
457        let params = select_params(vector, preset, &mut scratch.sample);
458        encode_vector(vector, params, scratch, &mut page)?;
459    }
460
461    Ok(page)
462}

464/// A page encoded incrementally, one vector at a time.
465///
466/// Once the column chunk's preset is known (after the first page), later pages do
467/// not need all their values at once: each vector is encoded and appended to
468/// `body` as soon as its [`VECTOR_SIZE`] values arrive, and the raw floats are
469/// dropped. Only a sub-vector remainder is held in `carry` between `put`s. The
470/// header and offset array - which depend on the final vector count - are
471/// prepended in [`StreamingPage::finish`].
472///
473/// The bytes produced are identical to encoding the same values in one pass with
474/// [`encode_page`]: same vectors, same per-vector parameters, same contiguous
475/// offsets. What changes is peak memory - the whole page of raw floats is never
476/// resident - and that the encode work is spread across `put`s.
477struct StreamingPage<F: AlpFloat> {
478    /// Encoded vectors, concatenated; becomes the page body after the offsets.
479    body: Vec<u8>,
480    /// Body-relative start offset of each completed vector.
481    vector_offsets: Vec<u32>,
482    /// Values not yet forming a complete vector, carried across `put`s.
483    carry: Vec<F>,
484    /// Total values appended to this page so far.
485    count: usize,
486}
487
488impl<F: AlpFloat> StreamingPage<F> {
489    fn new() -> Self {
490        Self {
491            body: Vec::new(),
492            vector_offsets: Vec::new(),
493            carry: Vec::new(),
494            count: 0,
495        }
496    }
497
498    fn estimated_memory_size(&self) -> usize {
499        self.body.capacity()
500            + self.vector_offsets.capacity() * std::mem::size_of::<u32>()
501            + self.carry.capacity() * std::mem::size_of::<F>()
502    }
503
504    /// Encode one complete vector and append it to `body`.
505    fn push_vector(
506        &mut self,
507        vector: &[F],
508        preset: &[ExponentAndFactor],
509        scratch: &mut Scratch<F>,
510    ) -> Result<()> {
511        let body_start = u32::try_from(self.body.len())
512            .map_err(|_| general_err!("Invalid ALP page: body exceeds u32 offset range"))?;
513        self.vector_offsets.push(body_start);
514        let params = select_params(vector, preset, &mut scratch.sample);
515        encode_vector(vector, params, scratch, &mut self.body)
516    }
517
518    /// Buffer `values`, encoding and dropping every complete vector they form.
519    fn put(
520        &mut self,
521        mut values: &[F],
522        preset: &[ExponentAndFactor],
523        scratch: &mut Scratch<F>,
524    ) -> Result<()> {
525        self.count += values.len();
526
527        // Finish a vector left partially filled by a previous `put`.
528        if !self.carry.is_empty() {
529            let need = VECTOR_SIZE - self.carry.len();
530            if values.len() < need {
531                self.carry.extend_from_slice(values);
532                return Ok(());
533            }
534            let (head, tail) = values.split_at(need);
535            self.carry.extend_from_slice(head);
536            // Move the carry out so the vector slice does not alias `self`; the
537            // allocation is handed back, cleared, for the next partial to reuse.
538            let mut vector = std::mem::take(&mut self.carry);
539            self.push_vector(&vector, preset, scratch)?;
540            vector.clear();
541            self.carry = vector;
542            values = tail;
543        }
544
545        // Encode whole vectors straight from the input, no copy.
546        let mut chunks = values.chunks_exact(VECTOR_SIZE);
547        for vector in chunks.by_ref() {
548            self.push_vector(vector, preset, scratch)?;
549        }
550        self.carry.extend_from_slice(chunks.remainder());
551        Ok(())
552    }
553
554    /// Encode the trailing partial vector, then assemble `[header][offsets][body]`
555    /// and reset for the next page.
556    fn finish(
557        &mut self,
558        preset: &[ExponentAndFactor],
559        scratch: &mut Scratch<F>,
560    ) -> Result<Vec<u8>> {
561        if !self.carry.is_empty() {
562            let mut vector = std::mem::take(&mut self.carry);
563            self.push_vector(&vector, preset, scratch)?;
564            vector.clear();
565            self.carry = vector;
566        }
567
568        let num_vectors = self.vector_offsets.len();
569        let offsets_section = num_vectors * std::mem::size_of::<u32>();
570        let header = AlpHeader {
571            compression_mode: ALP_COMPRESSION_MODE,
572            integer_encoding: ALP_INTEGER_ENCODING_FOR_BIT_PACK,
573            vector_size: VECTOR_SIZE,
574            num_elements: self.count,
575        };
576
577        let mut page = Vec::with_capacity(ALP_HEADER_SIZE + offsets_section + self.body.len());
578        page.extend_from_slice(&header.serialize()?);
579        for &body_start in &self.vector_offsets {
580            // Offsets are measured from the start of the offset array: the array's
581            // own size plus the vector's position within the body.
582            let page_offset = u32::try_from(offsets_section + body_start as usize)
583                .map_err(|_| general_err!("Invalid ALP page: body exceeds u32 offset range"))?;
584            page.extend_from_slice(&page_offset.to_le_bytes());
585        }
586        page.extend_from_slice(&self.body);
587
588        self.body.clear();
589        self.vector_offsets.clear();
590        self.count = 0;
591        Ok(page)
592    }
593}

595/// Encoder for ALP-encoded floating-point pages (`f32`/`f64`).
596///
597/// The first page is buffered whole: ALP samples it to choose the column chunk's
598/// candidate `(exponent, factor)` set, which needs all of the page's data. Once
599/// that preset is fixed, later pages are encoded incrementally, a vector at a
600/// time, without ever holding the whole page of raw floats (see [`StreamingPage`]).
601pub struct AlpEncoder<T: DataType>
602where
603    T::T: AlpFloat,
604{
605    /// Values buffered for the first page, until the preset is built. Empty once
606    /// streaming begins.
607    values: Vec<T::T>,
608    /// Candidate `(exponent, factor)` pairs for this column chunk, sampled once
609    /// from the first page and reused for the rest of the chunk. `None` until the
610    /// first page is flushed; its presence is what switches `put` to streaming.
611    preset: Option<Vec<ExponentAndFactor>>,
612    scratch: Scratch<T::T>,
613    /// The page built incrementally, used for every page after the first.
614    streaming: StreamingPage<T::T>,
615}
616
617impl<T: DataType> AlpEncoder<T>
618where
619    T::T: AlpFloat,
620{
621    pub(crate) fn new() -> Self {
622        Self {
623            values: Vec::new(),
624            preset: None,
625            scratch: Scratch::new(),
626            streaming: StreamingPage::new(),
627        }
628    }
629
630    /// Values buffered for the current page: the first page lives in `values`,
631    /// later pages in the streaming buffer. Used only for size estimates.
632    fn current_page_len(&self) -> usize {
633        if self.preset.is_none() {
634            self.values.len()
635        } else {
636            self.streaming.count
637        }
638    }
639}
640
641impl<T: DataType> Encoder<T> for AlpEncoder<T>
642where
643    T::T: AlpFloat,
644{
645    fn put(&mut self, values: &[T::T]) -> Result<()> {
646        let Self {
647            values: buffer,
648            preset,
649            scratch,
650            streaming,
651        } = self;
652        match preset.as_deref() {
653            // First page: buffer until the preset can be built on flush.
654            None => buffer.extend_from_slice(values),
655            // Later pages: encode incrementally against the fixed preset.
656            Some(preset) => streaming.put(values, preset, scratch)?,
657        }
658        Ok(())
659    }
660
661    fn encoding(&self) -> Encoding {
662        Encoding::ALP
663    }
664
665    fn estimated_data_encoded_size(&self) -> usize {
666        // Encoded size is not known until the parameters are chosen. In the
667        // worst case, almost every value is an exception while two encodable
668        // values force a full-width packed FOR range. Bound both the packed
669        // placeholder/value and the raw exception value by the exact type's
670        // width, plus the exception position. Bounding by value count (not the
671        // running encoded size) keeps the writer's page-boundary decisions
672        // identical whether the page is buffered or streamed.
673        let len = self.current_page_len();
674        let num_vectors = len.div_ceil(VECTOR_SIZE);
675        ALP_HEADER_SIZE
676            + num_vectors
677                * (std::mem::size_of::<u32>()
678                    + AlpInfo::STORED_SIZE
679                    + ForInfo::<<T::T as AlpFloat>::Exact>::stored_size())
680            + len * (2 * <T::T as AlpFloat>::Exact::WIDTH + std::mem::size_of::<u16>())
681    }
682
683    fn estimated_memory_size(&self) -> usize {
684        self.values.capacity() * std::mem::size_of::<T::T>()
685            + self.preset.as_ref().map_or(0, |p| {
686                p.capacity() * std::mem::size_of::<ExponentAndFactor>()
687            })
688            + self.scratch.estimated_memory_size()
689            + self.streaming.estimated_memory_size()
690    }
691
692    fn flush_buffer(&mut self) -> Result<Bytes> {
693        let Self {
694            values,
695            preset,
696            scratch,
697            streaming,
698        } = self;
699
700        // The first flush builds the preset from the whole buffered page and
701        // encodes it in one pass; that also arms streaming for later pages.
702        let page = match preset {
703            None => {
704                let built = build_preset(values);
705                let page = encode_page(values, &built, scratch)?;
706                values.clear();
707                *preset = Some(built);
708                page
709            }
710            Some(preset) => streaming.finish(preset.as_slice(), scratch)?,
711        };
712        Ok(page.into())
713    }
714}

848    /// bits an unencoded double takes.

850    /// The parameter to check is `exponent - factor`, not `exponent`: the encoded
851    /// integers depend only on the effective scale `10^(exponent - factor)`, so
852    /// (3, 1) and (2, 0) encode identically and tie on estimated size. The ALP

854    /// why the pair chosen here is not simply (2, 0).

857        let values: Vec<f64> = (0..1024).map(|i| (i as f64) * 0.01).collect();

862

864        let exponent = page[ALP_HEADER_SIZE + 4];
865        let factor = page[ALP_HEADER_SIZE + 5];
866        let num_exceptions =
867            u16::from_le_bytes([page[ALP_HEADER_SIZE + 6], page[ALP_HEADER_SIZE + 7]]);
868
869        assert_eq!(
870            exponent - factor,
871            2,
872            "two-decimal data should encode at an effective scale of 10^2, \
873             got exponent {exponent} factor {factor}"

876
877        let plain_size = values.len() * std::mem::size_of::<f64>();
878        assert!(

880            "expected at least 4x compression, got {} bytes vs {plain_size} plain",
881            page.len()
882        );

1047    /// Fill the preset to `MAX_COMBINATIONS` and include a vector no candidate
1048    /// can encode, exercising the all-exception size estimate and the early
1049    /// exit of the per-vector candidate search.
1050    #[test]
1051    fn test_full_preset_with_all_exception_vector() {
1052        // Nine vectors are sampled at stride two, so the even-indexed ones
1053        // decide the preset: four distinct decimal scales, plus the all-NaN
1054        // ninth vector, which no candidate can encode and which therefore
1055        // contributes the worst-case fallback pair as the fifth candidate.
1056        let scales = [0.01, 0.01, 0.001, 0.01, 0.0001, 0.01, 0.00001, 0.01];
1057        let mut values = Vec::with_capacity(9 * VECTOR_SIZE);
1058        for scale in scales {
1059            values.extend((0..VECTOR_SIZE).map(|i| (i as f64) * scale));
1060        }
1061        values.extend(std::iter::repeat_n(f64::NAN, VECTOR_SIZE));
1062
1063        let preset = build_preset(&values);
1064        assert_eq!(preset.len(), MAX_COMBINATIONS);
1065
1066        let nan_vector = &values[8 * VECTOR_SIZE..];
1067        let mut sample = Vec::new();
1068        sample_values(nan_vector, &mut sample);
1069        let candidate_costs: Vec<_> = preset
1070            .iter()
1071            .map(|&params| estimate_size_bits(&sample, params, false))
1072            .collect();
1073        assert!(candidate_costs.windows(2).all(|costs| costs[0] == costs[1]));
1074
1075        // Encoding the NaN vector prices all five candidates at the same
1076        // all-exception size, so the per-vector search stops early after
1077        // `SAMPLING_EARLY_EXIT_THRESHOLD` non-improving candidates. The
1078        // round-trip proves the page survives both paths losslessly.
1079        assert_bits_eq(&roundtrip::<DoubleType>(&values), &values);
