12/// Encoder for definition/repetition levels.
13/// Currently only supports Rle and BitPacked (dev/null) encoding, including v2.
14pub enum LevelEncoder {
15    Rle(RleEncoder),
16    RleV2(RleEncoder),
17}
18
19impl LevelEncoder {
20    /// Creates a new streaming level encoder for Data Page v1.
21    ///
22    /// This does not require knowing the number of values
23    /// upfront, making it suitable for incremental encoding where levels are fed in
24    /// as they arrive via [`put_with_observer`](Self::put_with_observer).
25    pub fn v1_streaming(max_level: i16) -> Self {
26        let bit_width = num_required_bits(max_level as u64);
27        // Reserve space for length header
28        let buffer = vec![0u8; 4];
29        LevelEncoder::Rle(RleEncoder::new_from_buf(bit_width, buffer))
30    }
31
32    /// Creates a new streaming RLE level encoder for Data Page v2.
33    ///
34    /// This does not require knowing the number of values
35    /// upfront, making it suitable for incremental encoding where levels are fed in
36    /// as they arrive via [`put_with_observer`](Self::put_with_observer).
37    pub fn v2_streaming(max_level: i16) -> Self {
38        let bit_width = num_required_bits(max_level as u64);
39        LevelEncoder::RleV2(RleEncoder::new_from_buf(bit_width, Vec::new()))
40    }
41
42    /// Put/encode levels vector into this level encoder and call
43    /// `observer(value, count)` for each run of identical values encountered
44    /// during encoding.
45    ///
46    /// Returns number of encoded values that are less than or equal to length
47    /// of the input buffer.
48    ///
49    /// This method does **not** flush the underlying encoder, so it can be called
50    /// incrementally across multiple batches without forcing run boundaries.
51    /// The encoder is flushed automatically when [`consume`](Self::consume) is called.
52    #[inline]
53    pub fn put_with_observer<F>(&mut self, buffer: &[i16], mut observer: F) -> usize
54    where
55        F: FnMut(i16, usize),
56    {
57        match *self {
58            LevelEncoder::Rle(ref mut encoder) | LevelEncoder::RleV2(ref mut encoder) => {
59                let mut remaining = buffer;
60                while let Some((&value, rest)) = remaining.split_first() {
61                    encoder.put(value as u64);
62                    // After put(), check if the encoder just entered RLE
63                    // accumulation mode. If so, scan ahead for the rest of
64                    // this run to batch the observer call and bulk-extend.
65                    if encoder.is_accumulating_rle(value as u64) {
66                        let run_len = rest.iter().take_while(|&&v| v == value).count();
67                        if run_len > 0 {
68                            encoder.extend_run(run_len);
69                        }
70                        observer(value, 1 + run_len);
71                        remaining = &rest[run_len..];
72                    } else {
73                        observer(value, 1);
74                        remaining = rest;
75                    }
76                }
77                buffer.len()
78            }
79        }
80    }
81
82    /// Encode `count` repetitions of a single level value, calling
83    /// `observer(value, count)` exactly once.
84    ///
85    /// This can be used to encode uniform runs without allocating a level
86    /// buffer. Like [`Self::put_with_observer`], but specialized for a single
87    /// repeated level value.
88    ///
89    /// This is O(1) amortized for RLE-based encoders (after a small warmup).
90    #[inline]
91    pub fn put_n_with_observer<F>(&mut self, value: i16, count: usize, mut observer: F)
92    where
93        F: FnMut(i16, usize),
94    {
95        let encoder = match *self {
96            LevelEncoder::Rle(ref mut encoder) | LevelEncoder::RleV2(ref mut encoder) => encoder,
97        };
98
99        // Feed values individually until the encoder enters RLE accumulation
100        // mode for this value, or until we've encoded everything.
101        let mut remaining = count;
102        while remaining > 0 && !encoder.is_accumulating_rle(value as u64) {
103            encoder.put(value as u64);
104            remaining -= 1;
105        }
106        // If we're now in accumulation mode, bulk-extend the rest.
107        if remaining > 0 {
108            encoder.extend_run(remaining);
109        }
110        observer(value, count);
111    }
112
113    /// Finalizes level encoder, flush all intermediate buffers and return resulting
114    /// encoded buffer. Returned buffer is already truncated to encoded bytes only.
115    #[inline]
116    #[cfg_attr(all(not(feature = "experimental"), not(test)), expect(unused))]
117    pub fn consume(self) -> Vec<u8> {
118        match self {
119            LevelEncoder::Rle(encoder) => {
120                let mut encoded_data = encoder.consume();
121                // Account for the buffer offset
122                let encoded_len = encoded_data.len() - mem::size_of::<i32>();
123                let len = (encoded_len as i32).to_le();
124                let len_bytes = len.as_bytes();
125                encoded_data[0..len_bytes.len()].copy_from_slice(len_bytes);
126                encoded_data
127            }
128            LevelEncoder::RleV2(encoder) => encoder.consume(),
129        }
130    }
131
132    /// Flushes all intermediate buffers, passes the encoded data to `f`, then
133    /// resets the encoder for reuse while retaining the buffer allocation.
134    #[inline]
135    pub fn flush_to<F, R>(&mut self, f: F) -> R
136    where
137        F: FnOnce(&[u8]) -> R,
138    {
139        let result = match self {
140            LevelEncoder::Rle(encoder) => {
141                let data = encoder.flush_buffer_mut();
142                // Patch the 4-byte length header reserved at the start of the buffer
143                let encoded_len = (data.len() - mem::size_of::<i32>()) as i32;
144                data[..4].copy_from_slice(&encoded_len.to_le_bytes());
145                f(data)
146            }
147            LevelEncoder::RleV2(encoder) => f(encoder.flush_buffer()),
148        };
149        match self {
150            LevelEncoder::Rle(encoder) => {
151                encoder.clear();
152                // Re-reserve the 4-byte length header for the next page
153                encoder.skip(mem::size_of::<i32>());
154            }
155            LevelEncoder::RleV2(encoder) => encoder.clear(),
156        }
157        result
158    }
159}

161#[cfg(test)]
162mod tests {
163    use super::*;

165    /// Encode `count` repetitions of `value` using `put_with_observer` and
166    /// return the raw encoded bytes.
167    fn reference_encode(max_level: i16, values: &[i16]) -> Vec<u8> {
168        let mut enc = LevelEncoder::v2_streaming(max_level);
169        enc.put_with_observer(values, |_, _| {});
170        enc.consume()
171    }
172
173    #[test]
174    fn test_put_n_with_observer_large_run() {
175        // Large count exercises the bulk extend_run path (past the 8-value warmup).
176        let max_level = 3;
177        let count = 10_000;
178        let mut enc = LevelEncoder::v2_streaming(max_level);
179        enc.put_n_with_observer(2, count, |_, _| {});
180        assert_eq!(enc.consume(), reference_encode(max_level, &vec![2; count]));
181    }
182
183    #[test]
184    fn test_put_n_with_observer_small_count() {
185        // Count smaller than the RLE warmup threshold — only the per-element loop runs.
186        let max_level = 3;
187        let mut enc = LevelEncoder::v2_streaming(max_level);
188        enc.put_n_with_observer(1, 5, |_, _| {});
189        assert_eq!(enc.consume(), reference_encode(max_level, &[1; 5]));
190    }

197        enc.put_n_with_observer(3, 8, |_, _| {});

233    #[test]
234    fn test_put_n_with_observer_followed_by_different_value() {
235        // Two consecutive put_n calls with different values — verifies that
236        // the encoder correctly transitions between runs.
237        let max_level = 3;
238        let mut enc = LevelEncoder::v2_streaming(max_level);
239        let mut calls: Vec<(i16, usize)> = Vec::new();
240        enc.put_n_with_observer(1, 100, |v, c| calls.push((v, c)));
241        enc.put_n_with_observer(3, 200, |v, c| calls.push((v, c)));
242        assert_eq!(calls, vec![(1, 100), (3, 200)]);
243
244        let reference = reference_encode(max_level, &[&[1i16; 100][..], &[3i16; 200]].concat());
245        assert_eq!(enc.consume(), reference);
246    }

248    #[test]
249    fn test_put_n_with_observer_interleaved_with_put_with_observer() {
250        // Mix put_n_with_observer and put_with_observer to verify they compose.
251        let max_level = 3;
252        let mut enc = LevelEncoder::v2_streaming(max_level);
253        enc.put_n_with_observer(2, 50, |_, _| {});
254        enc.put_with_observer(&[0, 0, 1, 1, 3], |_, _| {});
255        enc.put_n_with_observer(2, 50, |_, _| {});
256
257        let input = [&[2i16; 50][..], &[0, 0, 1, 1, 3], &[2i16; 50]].concat();
258        assert_eq!(enc.consume(), reference_encode(max_level, &input));
259    }
