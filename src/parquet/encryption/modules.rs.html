5use crate::errors::ParquetError;
6
7#[derive(PartialEq)]
8pub(crate) enum ModuleType {
9    Footer = 0,
10    ColumnMetaData = 1,
11    DataPage = 2,
12    DictionaryPage = 3,
13    DataPageHeader = 4,
14    DictionaryPageHeader = 5,
15    ColumnIndex = 6,
16    OffsetIndex = 7,
17    _BloomFilterHeader = 8,
18    _BloomFilterBitset = 9,
19}
20
21pub fn create_footer_aad(file_aad: &[u8]) -> crate::errors::Result<Vec<u8>> {
22    create_module_aad(file_aad, ModuleType::Footer, 0, 0, None)
23}
24
25pub(crate) fn create_module_aad(
26    file_aad: &[u8],
27    module_type: ModuleType,
28    row_group_idx: usize,
29    column_ordinal: usize,
30    page_ordinal: Option<usize>,
31) -> crate::errors::Result<Vec<u8>> {
32    let module_buf = [module_type as u8];
33
34    if module_buf[0] == (ModuleType::Footer as u8) {
35        let mut aad = Vec::with_capacity(file_aad.len() + 1);
36        aad.extend_from_slice(file_aad);
37        aad.extend_from_slice(module_buf.as_ref());
38        return Ok(aad);
39    }
40
41    if row_group_idx > i16::MAX as usize {
42        return Err(general_err!(
43            "Encrypted parquet files can't have more than {} row groups: {}",
44            i16::MAX,
45            row_group_idx
46        ));
47    }
48    if column_ordinal > i16::MAX as usize {
49        return Err(general_err!(
50            "Encrypted parquet files can't have more than {} columns: {}",
51            i16::MAX,
52            column_ordinal
53        ));
54    }
55
56    if module_buf[0] != (ModuleType::DataPageHeader as u8)
57        && module_buf[0] != (ModuleType::DataPage as u8)
58    {
59        let mut aad = Vec::with_capacity(file_aad.len() + 5);
60        aad.extend_from_slice(file_aad);
61        aad.extend_from_slice(module_buf.as_ref());
62        aad.extend_from_slice((row_group_idx as i16).to_le_bytes().as_ref());
63        aad.extend_from_slice((column_ordinal as i16).to_le_bytes().as_ref());
64        return Ok(aad);
65    }
66
67    let page_ordinal =
68        page_ordinal.ok_or_else(|| general_err!("Page ordinal must be set for data pages"))?;
69
70    if page_ordinal > i16::MAX as usize {
71        return Err(general_err!(
72            "Encrypted parquet files can't have more than {} pages per column chunk: {}",
73            i16::MAX,
74            page_ordinal
75        ));
76    }
77
78    let mut aad = Vec::with_capacity(file_aad.len() + 7);
79    aad.extend_from_slice(file_aad);
80    aad.extend_from_slice(module_buf.as_ref());
81    aad.extend_from_slice((row_group_idx as i16).to_le_bytes().as_ref());
82    aad.extend_from_slice((column_ordinal as i16).to_le_bytes().as_ref());
83    aad.extend_from_slice((page_ordinal as i16).to_le_bytes().as_ref());
84    Ok(aad)
85}
