5//! [ArrowReaderMetrics] for collecting metrics about the Arrow reader
6
7use std::sync::Arc;
8use std::sync::atomic::AtomicUsize;
9
10/// This enum represents the state of Arrow reader metrics collection.
11///
12/// The inner metrics are stored in an `Arc<ArrowReaderMetricsInner>`
13/// so cloning the `ArrowReaderMetrics` enum will not clone the inner metrics.
14///
15/// To access metrics, create an `ArrowReaderMetrics` via [`ArrowReaderMetrics::enabled()`]
16/// and configure the `ArrowReaderBuilder` with a clone.
17#[derive(Debug, Clone)]
18pub enum ArrowReaderMetrics {
19    /// Metrics are not collected (default)
20    Disabled,
21    /// Metrics are collected and stored in an `Arc`.
22    ///
23    /// Create this via [`ArrowReaderMetrics::enabled()`].
24    Enabled(Arc<ArrowReaderMetricsInner>),
25}
26
27impl ArrowReaderMetrics {
28    /// Creates a new instance of [`ArrowReaderMetrics::Disabled`]
29    pub fn disabled() -> Self {
30        Self::Disabled
31    }
32
33    /// Creates a new instance of [`ArrowReaderMetrics::Enabled`]
34    pub fn enabled() -> Self {
35        Self::Enabled(Arc::new(ArrowReaderMetricsInner::new()))
36    }
37
38    /// Predicate Cache: number of records read directly from the inner reader
39    ///
40    /// This is the total number of records read from the inner reader (that is
41    /// actually decoding). It measures the amount of work that could not be
42    /// avoided with caching.
43    ///
44    /// It returns the number of records read across all columns, so if you read
45    /// 2 columns each with 100 records, this will return 200.
46    ///
47    ///
48    /// Returns None if metrics are disabled.
49    pub fn records_read_from_inner(&self) -> Option<usize> {
50        match self {
51            Self::Disabled => None,
52            Self::Enabled(inner) => Some(
53                inner
54                    .records_read_from_inner
55                    .load(std::sync::atomic::Ordering::Relaxed),
56            ),
57        }
58    }
59
60    /// Predicate Cache: number of records read from the cache
61    ///
62    /// This is the total number of records read from the cache actually
63    /// decoding). It measures the amount of work that was avoided with caching.
64    ///
65    /// It returns the number of records read across all columns, so if you read
66    /// 2 columns each with 100 records from the cache, this will return 200.
67    ///
68    /// Returns None if metrics are disabled.
69    pub fn records_read_from_cache(&self) -> Option<usize> {
70        match self {
71            Self::Disabled => None,
72            Self::Enabled(inner) => Some(
73                inner
74                    .records_read_from_cache
75                    .load(std::sync::atomic::Ordering::Relaxed),
76            ),
77        }
78    }
79
80    /// Increments the count of records read from the inner reader
81    pub(crate) fn increment_inner_reads(&self, count: usize) {
82        let Self::Enabled(inner) = self else {
83            return;
84        };
85        inner
86            .records_read_from_inner
87            .fetch_add(count, std::sync::atomic::Ordering::Relaxed);
88    }
89
90    /// Increments the count of records read from the cache
91    pub(crate) fn increment_cache_reads(&self, count: usize) {
92        let Self::Enabled(inner) = self else {
93            return;
94        };
95
96        inner
97            .records_read_from_cache
98            .fetch_add(count, std::sync::atomic::Ordering::Relaxed);
99    }
100}
101
102/// Holds the actual metrics for the Arrow reader.
103///
104/// Please see [`ArrowReaderMetrics`] for the public interface.
105#[derive(Debug)]
106pub struct ArrowReaderMetricsInner {
107    // Metrics for Predicate Cache
108    /// Total number of records read from the inner reader (uncached)
109    records_read_from_inner: AtomicUsize,
110    /// Total number of records read from previously cached pages
111    records_read_from_cache: AtomicUsize,
112}
113
114impl ArrowReaderMetricsInner {
115    /// Creates a new instance of `ArrowReaderMetricsInner`
116    pub(crate) fn new() -> Self {
117        Self {
118            records_read_from_inner: AtomicUsize::new(0),
119            records_read_from_cache: AtomicUsize::new(0),
120        }
121    }
122}
