27/// A predicate operating on [`RecordBatch`]
28///
29/// See also:
30/// * [`RowFilter`] for more information  on applying filters during the
31///   Parquet decoding process.
32/// * [`ArrowPredicateFn`] for a concrete implementation based on a function
33pub trait ArrowPredicate: Send + 'static {
34    /// Returns the [`ProjectionMask`] that describes the columns required
35    /// to evaluate this predicate.
36    ///
37    /// All projected columns will be provided in the `batch` passed to
38    /// [`evaluate`](Self::evaluate). The projection mask should be as small as
39    /// possible because any columns needed for the overall projection mask are
40    /// decoded again after a predicate is applied.
41    fn projection(&self) -> &ProjectionMask;
42
43    /// Evaluate this predicate for the given [`RecordBatch`] containing the columns
44    /// identified by [`Self::projection`]
45    ///
46    /// Must return a [`BooleanArray`] that has the same length as the input
47    /// `batch` where each row indicates whether the row should be returned:
48    /// * `true`:the row should be returned
49    /// * `false` or `null`: the row should not be returned
50    fn evaluate(&mut self, batch: RecordBatch) -> Result<BooleanArray, ArrowError>;
51}

53/// An [`ArrowPredicate`] created from an [`FnMut`] and a [`ProjectionMask`]
54///
55/// See [`RowFilter`] for more information on applying filters during the
56/// Parquet decoding process.
57///
58/// The function is passed `RecordBatch`es with only the columns specified in
59/// the [`ProjectionMask`].
60///
61/// The function must return a [`BooleanArray`] that has the same length as the
62/// input `batch` where each row indicates whether the row should be returned:
63/// * `true`: the row should be returned
64/// * `false` or `null`: the row should not be returned
65///
66/// # Example:
67///
68/// Given an input schema: `"a:int64", "b:int64"`, you can create a predicate that
69/// evaluates `b > 0` like this:
70///
71/// ```
72/// # use std::sync::Arc;
73/// # use arrow::compute::kernels::cmp::gt;
74/// # use arrow_array::{BooleanArray, Int64Array, RecordBatch};
75/// # use arrow_array::cast::AsArray;
76/// # use arrow_array::types::Int64Type;
77/// # use parquet::arrow::arrow_reader::ArrowPredicateFn;
78/// # use parquet::arrow::ProjectionMask;
79/// # use parquet::schema::types::{SchemaDescriptor, Type};
80/// # use parquet::basic; // note there are two `Type`s that are different
81/// # // Schema for a table with one columns: "a" (int64) and "b" (int64)
82/// # let descriptor = SchemaDescriptor::new(
83/// #  Arc::new(
84/// #    Type::group_type_builder("my_schema")
85/// #      .with_fields(vec![
86/// #        Arc::new(
87/// #         Type::primitive_type_builder("a", basic::Type::INT64)
88/// #          .build().unwrap()
89/// #        ),
90/// #        Arc::new(
91/// #         Type::primitive_type_builder("b", basic::Type::INT64)
92/// #          .build().unwrap()
93/// #        ),
94/// #     ])
95/// #     .build().unwrap()
96/// #  )
97/// # );
98/// // Create a mask for selecting only the second column "b" (index 1)
99/// let projection_mask = ProjectionMask::leaves(&descriptor, [1]);
100/// // Closure that evaluates "b > 0"
101/// let predicate = |batch: RecordBatch| {
102///    let scalar_0 = Int64Array::new_scalar(0);
103///    let column = batch.column(0).as_primitive::<Int64Type>();
104///    // call the gt kernel to compute `>` which returns a BooleanArray
105///    gt(column, &scalar_0)
106///  };
107/// // Create ArrowPredicateFn that can be passed to RowFilter
108/// let arrow_predicate = ArrowPredicateFn::new(projection_mask, predicate);
109/// ```
110pub struct ArrowPredicateFn<F> {
111    f: F,
112    projection: ProjectionMask,
113}
114
115impl<F> ArrowPredicateFn<F>
116where
117    F: FnMut(RecordBatch) -> Result<BooleanArray, ArrowError> + Send + 'static,
118{
119    /// Create a new [`ArrowPredicateFn`] that invokes `f` on the columns
120    /// specified in `projection`.
121    pub fn new(projection: ProjectionMask, f: F) -> Self {
122        Self { f, projection }
123    }
124}
125
126impl<F> ArrowPredicate for ArrowPredicateFn<F>
127where
128    F: FnMut(RecordBatch) -> Result<BooleanArray, ArrowError> + Send + 'static,
129{
130    fn projection(&self) -> &ProjectionMask {
131        &self.projection
132    }
133
134    fn evaluate(&mut self, batch: RecordBatch) -> Result<BooleanArray, ArrowError> {
135        (self.f)(batch)
136    }
137}

139/// Filter applied *during* the parquet read process
140///
141/// See example on [`ArrowReaderBuilder::with_row_filter`]
142///
143/// [`RowFilter`] applies predicates in order, after decoding only the columns
144/// required. As predicates eliminate rows, fewer rows from subsequent columns
145/// may be required, thus potentially reducing IO and decode. This process is
146/// also known as *push down* filtering and  *late materialization*.
147///
148/// A `RowFilter` consists of a list of [`ArrowPredicate`]s. Only the rows for which
149/// all the predicates evaluate to `true` will be returned.
150/// Any [`RowSelection`] provided to the reader will be applied prior
151/// to the first predicate, and each predicate in turn will then be used to compute
152/// a more refined [`RowSelection`] used when evaluating the subsequent predicates.
153///
154/// Once all predicates have been evaluated, the final [`RowSelection`] is applied
155/// to the top-level [`ProjectionMask`] to produce the final output [`RecordBatch`].
156///
157/// This design has a couple of implications:
158///
159/// * [`RowFilter`] can be used to skip entire pages, and thus IO, in addition to CPU decode overheads
160/// * Columns may be decoded multiple times if they appear in multiple [`ProjectionMask`].
161///   Consecutive predicates whose projection is the same single top-level, non-repeated
162///   column are evaluated together on one decoded stream, so ordering such predicates
163///   next to each other avoids decoding that column again
164/// * IO will be deferred until needed by a [`ProjectionMask`]
165///
166/// As such there is a trade-off between a single large predicate, or multiple predicates,
167/// that will depend on the shape of the data. Whilst multiple smaller predicates may
168/// minimise the amount of data scanned/decoded, it may not be faster overall.
169///
170/// For example, if a predicate that needs a single column of data filters out all but
171/// 1% of the rows, applying it as one of the early `ArrowPredicateFn` will likely significantly
172/// improve performance.
173///
174/// As a counter example, if a predicate needs several columns of data to evaluate but
175/// leaves 99% of the rows, it may be better to not filter the data from parquet and
176/// apply the filter after the RecordBatch has been fully decoded.
177///
178/// Additionally, even if a predicate eliminates a moderate number of rows, it may still be faster
179/// to filter the data after the RecordBatch has been fully decoded, if the eliminated rows are
180/// not contiguous.
181///
182/// [`RowSelection`]: crate::arrow::arrow_reader::RowSelection
183/// [`ArrowReaderBuilder::with_row_filter`]: crate::arrow::arrow_reader::ArrowReaderBuilder::with_row_filter
184pub struct RowFilter {
185    /// A list of [`ArrowPredicate`]
186    pub(crate) predicates: Vec<Box<dyn ArrowPredicate>>,
187}
188
189impl Debug for RowFilter {
190    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
191        write!(f, "RowFilter {{ {} predicates: }}", self.predicates.len())
192    }
193}

195impl RowFilter {
196    /// Create a new [`RowFilter`] from an array of [`ArrowPredicate`]
197    pub fn new(predicates: Vec<Box<dyn ArrowPredicate>>) -> Self {
198        Self { predicates }
199    }
200    /// Returns the inner predicates
201    pub fn predicates(&self) -> &Vec<Box<dyn ArrowPredicate>> {
202        &self.predicates
203    }
204    /// Returns the inner predicates, consuming self
205    pub fn into_predicates(self) -> Vec<Box<dyn ArrowPredicate>> {
206        self.predicates
207    }

209    /// Fuse consecutive predicates on the same single top-level, non-repeated leaf.
210    /// This avoids repeated decoding or predicate-cache replay of that column.
211    pub(crate) fn fuse_same_projection(self, parquet_schema: &SchemaDescriptor) -> Self {
212        let mut predicates: Vec<Box<dyn ArrowPredicate>> =
213            Vec::with_capacity(self.predicates.len());
214        let mut group: Vec<Box<dyn ArrowPredicate>> = Vec::new();
215        let mut flush_group = |group: &mut Vec<Box<dyn ArrowPredicate>>| {
216            if group.len() > 1 && can_fuse_projection(group[0].projection(), parquet_schema) {
217                let group = std::mem::take(group);
218                predicates.push(Box::new(FusedPredicate::new(group)));
219            } else {
220                predicates.append(group);
221            }
222        };
223
224        for predicate in self.predicates {
225            if group
226                .last()
227                .is_some_and(|last| last.projection() != predicate.projection())
228            {
229                flush_group(&mut group);
230            }
231            group.push(predicate);
232        }
233        flush_group(&mut group);
234
235        Self { predicates }
236    }
237}

239/// Restrict fusion to one top-level, non-repeated leaf to limit compaction costs.
240fn can_fuse_projection(projection: &ProjectionMask, parquet_schema: &SchemaDescriptor) -> bool {
241    let mut leaf_indices =
242        (0..parquet_schema.num_columns()).filter(|idx| projection.leaf_included(*idx));
243    let Some(leaf_idx) = leaf_indices.next() else {
244        return false;
245    };
246    if leaf_indices.next().is_some() {
247        return false;
248    }
249
250    let column = parquet_schema.column(leaf_idx);
251    column.path().parts().len() == 1 && column.max_rep_level() == 0
252}

254/// Evaluate same-projection predicates in order on one decoded batch.
255/// Later predicates see only surviving rows, sliced when contiguous or compacted
256/// otherwise.
257struct FusedPredicate {
258    /// At least two predicates, all with the same projection.
259    predicates: Vec<Box<dyn ArrowPredicate>>,
260    /// Chooses how intermediate selections are composed. The output is always
261    /// a [`BooleanArray`], so this is independent of the reader's
262    /// [`RowSelectionPolicy`], which governs decoding.
263    composition_policy: RowSelectionPolicy,
264}
265
266impl FusedPredicate {
267    /// Create a fused predicate from at least two predicates that share one
268    /// projection.
269    fn new(predicates: Vec<Box<dyn ArrowPredicate>>) -> Self {
270        debug_assert!(predicates.len() > 1);
271        debug_assert!(
272            predicates
273                .windows(2)
274                .all(|pair| pair[0].projection() == pair[1].projection())
275        );
276        Self {
277            predicates,
278            composition_policy: RowSelectionPolicy::default(),
279        }
280    }
281}
282
283impl Debug for FusedPredicate {
284    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
285        write!(
286            f,
287            "FusedPredicate {{ {} predicates }}",
288            self.predicates.len()
289        )
290    }
291}
292
293impl ArrowPredicate for FusedPredicate {
294    fn projection(&self) -> &ProjectionMask {
295        self.predicates[0].projection()
296    }
297
298    fn evaluate(&mut self, batch: RecordBatch) -> Result<BooleanArray, ArrowError> {
299        let num_rows = batch.num_rows();
300        // Positions in the original batch; None until a predicate rejects rows.
301        let mut selection: Option<RowSelection> = None;
302        let mut filtered_batch = batch;
303        let last_predicate_idx = self.predicates.len() - 1;
304
305        for (idx, predicate) in self.predicates.iter_mut().enumerate() {
306            let filter = evaluate_predicate(predicate.as_mut(), filtered_batch.clone())?;
307            // No mapping is needed if all preceding predicates accepted every row.
308            if idx == last_predicate_idx && selection.is_none() {
309                return Ok(filter);
310            }
311            let true_count = filter.true_count();
312            if true_count == 0 {
313                return Ok(BooleanArray::new(BooleanBuffer::new_unset(num_rows), None));
314            }
315            if true_count == filter.len() {
316                continue;
317            }
318
319            let predicate_selection = RowSelection::from_boolean_buffer(filter.values().clone());
320            selection = Some(match selection.take() {
321                // Only the accumulated selection drives the composition
322                // algorithm, so adapt it once, right before it is used.
323                Some(prev) => self
324                    .composition_policy
325                    .apply(prev)
326                    .and_then(&predicate_selection),
327                None => predicate_selection,
328            });
329            if idx != last_predicate_idx {
330                filtered_batch = narrow_batch(&filtered_batch, &filter, true_count)?;
331            }
332        }
333
334        let mask = match selection {
335            Some(selection) => selection.into_boolean_buffer(),
336            None => BooleanBuffer::new_set(num_rows),
337        };
338        debug_assert_eq!(mask.len(), num_rows);
339        Ok(BooleanArray::new(mask, None))
340    }
341}
342
343/// Validate the predicate result length and treat nulls as false.
344fn evaluate_predicate(
345    predicate: &mut dyn ArrowPredicate,
346    batch: RecordBatch,
347) -> Result<BooleanArray, ArrowError> {
348    let input_rows = batch.num_rows();
349    let filter = predicate.evaluate(batch)?;
350    if filter.len() != input_rows {
351        return Err(ArrowError::InvalidArgumentError(format!(
352            "ArrowPredicate predicate returned {} rows, expected {input_rows}",
353            filter.len()
354        )));
355    }
356    Ok(match filter.null_count() {
357        0 => filter,
358        _ => prep_null_mask_filter(&filter),
359    })
360}

362/// Restrict `batch` to the rows `filter` accepts, slicing zero-copy when they
363/// form one contiguous range and compacting otherwise.
364///
365/// `filter` must be null-free and accept some but not all rows.
366fn narrow_batch(
367    batch: &RecordBatch,
368    filter: &BooleanArray,
369    true_count: usize,
370) -> Result<RecordBatch, ArrowError> {
371    let mut slices = SlicesIterator::new(filter);
372    let (start, end) = slices
373        .next()
374        .expect("a partially selected filter has a true slice");
375    if end - start == true_count {
376        Ok(batch.slice(start, end - start))
377    } else {
378        filter_record_batch(batch, filter)
379    }
380}

541    /// Evaluate `predicates` one after another, each on the rows kept by its
542    /// predecessors, the way `RowFilter` applies separate predicates.
543    fn sequential(batch: &RecordBatch, predicates: &mut [Box<dyn ArrowPredicate>]) -> Vec<bool> {
544        let mut kept = vec![true; batch.num_rows()];
545        for predicate in predicates {
546            // Like `RowFilter`, stop once every row has been rejected.
547            if !kept.contains(&true) {
548                break;
549            }
550            let survivors = filter_record_batch(batch, &BooleanArray::from(kept.clone())).unwrap();
551            let filter = predicate.evaluate(survivors).unwrap();
552            for (filter_idx, keep) in kept.iter_mut().filter(|keep| **keep).enumerate() {
553                *keep = filter.is_valid(filter_idx) && filter.value(filter_idx);
554            }
555        }
556        kept
557    }

626        assert_eq!(narrowed.column(0).to_data().buffers()[0].as_ptr(), unsafe {

628        });
629
630        let filter = BooleanArray::from(vec![false, true, false, true, false, false]);
631        let narrowed = narrow_batch(&batch, &filter, 2).unwrap();

633            narrowed.column(0).as_primitive::<Int32Type>().values(),

638    /// Fusing predicates into one `ReadPlanBuilder::with_predicate` call

647                DataType::Struct(Fields::from(vec![Field::new("c0", DataType::Int32, false)]));

651                0,
652                0,
653                false,
654                None,

656        };
657        let make_predicates = || vec![even(), divisible_by_three_after_even()];
