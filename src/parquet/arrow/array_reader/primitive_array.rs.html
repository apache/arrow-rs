28/// Provides conversion from `Vec<T>` to `Buffer`
29pub trait IntoBuffer {
30    fn into_buffer(self, target_type: &ArrowType) -> Buffer;
31}
32
33macro_rules! native_buffer {
34    ($($t:ty),*) => {
35        $(impl IntoBuffer for Vec<$t> {
36            fn into_buffer(self, _target_type: &ArrowType) -> Buffer {
37                Buffer::from_vec(self)
38            }
39        })*
40    };
41}
42native_buffer!(i8, i16, i32, i64, u8, u16, u32, u64, f32, f64);

44impl IntoBuffer for Vec<bool> {
45    fn into_buffer(self, _target_type: &ArrowType) -> Buffer {
46        BooleanBuffer::from(self.as_slice()).into_inner()
47    }
48}

50impl IntoBuffer for Vec<Int96> {
51    fn into_buffer(self, target_type: &ArrowType) -> Buffer {
52        let mut builder = Vec::with_capacity(self.len());
53        match target_type {
54            ArrowType::Timestamp(TimeUnit::Second, _) => {
55                builder.extend(self.iter().map(|x| x.to_seconds()));
56            }
57            ArrowType::Timestamp(TimeUnit::Millisecond, _) => {
58                builder.extend(self.iter().map(|x| x.to_millis()));
59            }
60            ArrowType::Timestamp(TimeUnit::Microsecond, _) => {
61                builder.extend(self.iter().map(|x| x.to_micros()));
62            }
63            ArrowType::Timestamp(TimeUnit::Nanosecond, _) => {
64                builder.extend(self.iter().map(|x| x.to_nanos()));
65            }
66            _ => unreachable!("Invalid target_type for Int96."),
67        }
68        Buffer::from_vec(builder)
69    }
70}

72/// Primitive array readers are leaves of array reader tree. They accept page iterator
73/// and read them into primitive arrays.
74pub struct PrimitiveArrayReader<T>
75where
76    T: DataType,
77    T::T: Copy + Default,
78    Vec<T::T>: IntoBuffer,
79{
80    data_type: ArrowType,
81    pages: Box<dyn PageIterator>,
82    def_levels_buffer: Option<Vec<i16>>,
83    rep_levels_buffer: Option<Vec<i16>>,
84    record_reader: RecordReader<T>,
85}
86
87impl<T> PrimitiveArrayReader<T>
88where
89    T: DataType,
90    T::T: Copy + Default,
91    Vec<T::T>: IntoBuffer,
92{
93    /// Construct primitive array reader.
94    ///
95    /// `padding_threshold` controls how null padding is applied. When
96    /// `None`, the reader pads all null positions (full padding). When
97    /// `Some(threshold)`, entries with `def < threshold` are excluded
98    /// from the value buffer (selective padding for list children).
99    pub fn new(
100        pages: Box<dyn PageIterator>,
101        column_desc: ColumnDescPtr,
102        arrow_type: Option<ArrowType>,
103        batch_size: usize,
104        padding_threshold: Option<i16>,
105    ) -> Result<Self> {
106        // Check if Arrow type is specified, else create it from Parquet type
107        let data_type = match arrow_type {
108            Some(t) => t,
109            None => parquet_to_arrow_field(column_desc.as_ref())?
110                .data_type()
111                .clone(),
112        };
113
114        let mut record_reader = RecordReader::<T>::new(column_desc, batch_size);
115        if let Some(threshold) = padding_threshold {
116            record_reader.set_padding_threshold(threshold);
117        }
118
119        Ok(Self {
120            data_type,
121            pages,
122            def_levels_buffer: None,
123            rep_levels_buffer: None,
124            record_reader,
125        })
126    }
127}

129/// Implementation of primitive array reader.
130impl<T> ArrayReader for PrimitiveArrayReader<T>
131where
132    T: DataType,
133    T::T: Copy + Default,
134    Vec<T::T>: IntoBuffer,
135{
136    fn as_any(&self) -> &dyn Any {
137        self
138    }
139
140    /// Returns data type of primitive array.
141    fn get_data_type(&self) -> &ArrowType {
142        &self.data_type
143    }
144
145    fn read_records(&mut self, batch_size: usize) -> Result<usize> {
146        read_records(&mut self.record_reader, self.pages.as_mut(), batch_size)
147    }
148
149    fn consume_batch(&mut self) -> Result<ArrayRef> {
150        let target_type = &self.data_type;
151
152        // Convert physical data to equivalent arrow type, and then perform
153        // coercion as needed
154        let len = self.record_reader.values_written();
155
156        let record_data = self
157            .record_reader
158            .consume_record_data()
159            .into_buffer(target_type);
160
161        let nulls = self
162            .record_reader
163            .consume_compact_bitmap()
164            .and_then(|b| NullBuffer::from_unsliced_buffer(b, len));
165
166        let array: ArrayRef = match T::get_physical_type() {
167            PhysicalType::BOOLEAN => Arc::new(BooleanArray::new(
168                BooleanBuffer::new(record_data, 0, len),
169                nulls,
170            )),
171            PhysicalType::INT32 => Arc::new(Int32Array::new(
172                ScalarBuffer::new(record_data, 0, len),
173                nulls,
174            )),
175            PhysicalType::INT64 => Arc::new(Int64Array::new(
176                ScalarBuffer::new(record_data, 0, len),
177                nulls,
178            )),
179            PhysicalType::FLOAT => Arc::new(Float32Array::new(
180                ScalarBuffer::new(record_data, 0, len),
181                nulls,
182            )),
183            PhysicalType::DOUBLE => Arc::new(Float64Array::new(
184                ScalarBuffer::new(record_data, 0, len),
185                nulls,
186            )),
187            PhysicalType::INT96 => Arc::new(Int64Array::new(
188                ScalarBuffer::new(record_data, 0, len),
189                nulls,
190            )),
191            PhysicalType::BYTE_ARRAY | PhysicalType::FIXED_LEN_BYTE_ARRAY => {
192                unreachable!("PrimitiveArrayReaders don't support complex physical types");
193            }
194        };
195
196        // Coerce the arrow type to the desired array type
197        let array = coerce_array(array, target_type)?;
198
199        // save definition and repetition buffers
200        self.def_levels_buffer = self.record_reader.consume_def_levels();
201        self.rep_levels_buffer = self.record_reader.consume_rep_levels();
202        self.record_reader.reset();
203        Ok(array)
204    }
205
206    fn skip_records(&mut self, num_records: usize) -> Result<usize> {
207        skip_records(&mut self.record_reader, self.pages.as_mut(), num_records)
208    }
209
210    fn get_def_levels(&self) -> Option<&[i16]> {
211        self.def_levels_buffer.as_deref()
212    }
213
214    fn get_rep_levels(&self) -> Option<&[i16]> {
215        self.rep_levels_buffer.as_deref()
216    }
217
218    fn max_def_level(&self) -> i16 {
219        self.record_reader.max_def_level()
220    }
221}

223/// Coerce the parquet physical type array to the target type
224///
225/// This should match the logic in schema::primitive::apply_hint
226fn coerce_array(array: ArrayRef, target_type: &ArrowType) -> Result<ArrayRef> {
227    if let ArrowType::Dictionary(key_type, value_type) = target_type {
228        let dictionary = pack_dictionary(key_type, array.as_ref())?;
229        let any_dictionary = dictionary.as_any_dictionary();
230
231        let coerced_values =
232            coerce_array(Arc::clone(any_dictionary.values()), value_type.as_ref())?;
233
234        return Ok(any_dictionary.with_values(coerced_values));
235    }
236
237    match array.data_type() {
238        ArrowType::Int32 => coerce_i32(array.as_primitive(), target_type),
239        ArrowType::Int64 => coerce_i64(array.as_primitive(), target_type),
240        ArrowType::Boolean | ArrowType::Float32 | ArrowType::Float64 => Ok(array),
241        _ => unreachable!("Cannot coerce array of type {}", array.data_type()),
242    }
243}

245fn coerce_i32(array: &Int32Array, target_type: &ArrowType) -> Result<ArrayRef> {
246    Ok(match target_type {
247        ArrowType::UInt8 => {
248            let array = array.unary(|i| i as u8) as UInt8Array;
249            Arc::new(array) as ArrayRef
250        }
251        ArrowType::Int8 => {
252            let array = array.unary(|i| i as i8) as Int8Array;
253            Arc::new(array) as ArrayRef
254        }
255        ArrowType::UInt16 => {
256            let array = array.unary(|i| i as u16) as UInt16Array;
257            Arc::new(array) as ArrayRef
258        }
259        ArrowType::Int16 => {
260            let array = array.unary(|i| i as i16) as Int16Array;
261            Arc::new(array) as ArrayRef
262        }
263        ArrowType::Int32 => Arc::new(array.clone()),
264        // follow C++ implementation and use overflow/reinterpret cast from  i32 to u32 which will map
265        // `i32::MIN..0` to `(i32::MAX as u32)..u32::MAX`
266        ArrowType::UInt32 => Arc::new(UInt32Array::new(
267            array.values().inner().clone().into(),
268            array.nulls().cloned(),
269        )) as ArrayRef,
270        ArrowType::Date32 => Arc::new(array.reinterpret_cast::<Date32Type>()) as _,
271        ArrowType::Date64 => {
272            let array: Date64Array = array.unary(|x| x as i64 * 86_400_000);
273            Arc::new(array) as ArrayRef
274        }
275        ArrowType::Time32(TimeUnit::Second) => {
276            Arc::new(array.reinterpret_cast::<Time32SecondType>()) as ArrayRef
277        }
278        ArrowType::Time32(TimeUnit::Millisecond) => {
279            Arc::new(array.reinterpret_cast::<Time32MillisecondType>()) as ArrayRef
280        }
281        ArrowType::Timestamp(time_unit, timezone) => match time_unit {
282            TimeUnit::Second => {
283                let array: TimestampSecondArray = array
284                    .unary(|x| x as i64)
285                    .with_timezone_opt(timezone.clone());
286                Arc::new(array) as _
287            }
288            TimeUnit::Millisecond => {
289                let array: TimestampMillisecondArray = array
290                    .unary(|x| x as i64)
291                    .with_timezone_opt(timezone.clone());
292                Arc::new(array) as _
293            }
294            TimeUnit::Microsecond => {
295                let array: TimestampMicrosecondArray = array
296                    .unary(|x| x as i64)
297                    .with_timezone_opt(timezone.clone());
298                Arc::new(array) as _
299            }
300            TimeUnit::Nanosecond => {
301                let array: TimestampNanosecondArray = array
302                    .unary(|x| x as i64)
303                    .with_timezone_opt(timezone.clone());
304                Arc::new(array) as _
305            }
306        },
307        ArrowType::Decimal32(p, s) => {
308            let array = array
309                .reinterpret_cast::<Decimal32Type>()
310                .with_precision_and_scale(*p, *s)?;
311            Arc::new(array) as ArrayRef
312        }
313        ArrowType::Decimal64(p, s) => {
314            let array: Decimal64Array =
315                array.unary(|i| i as i64).with_precision_and_scale(*p, *s)?;
316            Arc::new(array) as ArrayRef
317        }
318        ArrowType::Decimal128(p, s) => {
319            let array: Decimal128Array = array
320                .unary(|i| i as i128)
321                .with_precision_and_scale(*p, *s)?;
322            Arc::new(array) as ArrayRef
323        }
324        ArrowType::Decimal256(p, s) => {
325            let array: Decimal256Array = array
326                .unary(|i| i256::from_i128(i as i128))
327                .with_precision_and_scale(*p, *s)?;
328            Arc::new(array) as ArrayRef
329        }
330        _ => unreachable!("Cannot coerce i32 to {target_type}"),
331    })
332}

410macro_rules! pack_dictionary_helper {
411    ($t:ty, $values:ident) => {
412        match $values.data_type() {
413            ArrowType::Int32 => pack_dictionary_impl::<$t, Int32Type>($values.as_primitive()),
414            ArrowType::Int64 => pack_dictionary_impl::<$t, Int64Type>($values.as_primitive()),
415            ArrowType::Float32 => pack_dictionary_impl::<$t, Float32Type>($values.as_primitive()),
416            ArrowType::Float64 => pack_dictionary_impl::<$t, Float64Type>($values.as_primitive()),
417            _ => unreachable!("Invalid physical type"),
418        }
419    };
420}
421
422fn pack_dictionary(key: &ArrowType, values: &dyn Array) -> Result<ArrayRef> {
423    downcast_integer! {
424        key => (pack_dictionary_helper, values),
425        _ => unreachable!("Invalid key type"),
426    }
427}

429fn pack_dictionary_impl<K: ArrowDictionaryKeyType, V: ArrowPrimitiveType>(
430    values: &PrimitiveArray<V>,
431) -> Result<ArrayRef> {
432    let mut builder = PrimitiveDictionaryBuilder::<K, V>::with_capacity(1024, values.len());
433    builder.extend(values);
434    Ok(Arc::new(builder.finish()))
435}

456    #[expect(clippy::too_many_arguments)]
457    fn make_column_chunks<T: DataType>(
458        column_desc: ColumnDescPtr,
459        encoding: Encoding,
460        num_levels: usize,
461        min_value: T::T,
462        max_value: T::T,
463        def_levels: &mut Vec<i16>,
464        rep_levels: &mut Vec<i16>,
465        values: &mut Vec<T::T>,
466        page_lists: &mut Vec<Vec<Page>>,
467        use_v2: bool,
468        num_chunks: usize,
469    ) where
470        T::T: PartialOrd + SampleUniform + Copy,
471    {
472        for _i in 0..num_chunks {
473            let mut pages = VecDeque::new();
474            let mut data = Vec::new();
475            let mut page_def_levels = Vec::new();
476            let mut page_rep_levels = Vec::new();
477
478            make_pages::<T>(
479                column_desc.clone(),
480                encoding,
481                1,
482                num_levels,
483                min_value,
484                max_value,
485                &mut page_def_levels,
486                &mut page_rep_levels,
487                &mut data,
488                &mut pages,
489                use_v2,
490            );
491
492            def_levels.append(&mut page_def_levels);
493            rep_levels.append(&mut page_rep_levels);
494            values.append(&mut data);
495            page_lists.push(Vec::from(pages));
496        }
497    }

669                        .collect::<Vec<$result_primitive_type>>(),

672                let expected = arrow::compute::cast(&expected, &result_data_type)
673                    .expect("Unable to cast expected array");
674                assert_eq!(expected.data_type(), &result_data_type);
675                let expected = expected
676                    .as_any()
677                    .downcast_ref::<PrimitiveArray<$result_arrow_type>>()
678                    .expect(
679                        format!(
680                            "Unable to downcast expected {:?} to {:?}",
681                            expected.data_type(),
682                            result_data_type
683                        )
684                        .as_str(),
685                    )
686                    $(.clone().with_timezone($timezone))?

688                assert_eq!(expected, array);

784            .unwrap();
785
