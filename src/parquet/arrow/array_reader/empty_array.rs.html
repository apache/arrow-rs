5use crate::arrow::array_reader::ArrayReader;
6use crate::errors::Result;
7use arrow_array::{ArrayRef, StructArray};
8use arrow_data::ArrayDataBuilder;
9use arrow_schema::{DataType as ArrowType, Fields};
10use std::any::Any;
11use std::sync::Arc;
12
13/// Returns an [`ArrayReader`] that yields [`StructArray`] with no columns
14/// but with row counts that correspond to the amount of data in the file
15///
16/// This is useful for when projection eliminates all columns within a collection
17pub fn make_empty_array_reader(row_count: usize) -> Box<dyn ArrayReader> {
18    Box::new(EmptyArrayReader::new(row_count))
19}
20
21struct EmptyArrayReader {
22    data_type: ArrowType,
23    remaining_rows: usize,
24    need_consume_records: usize,
25}
26
27impl EmptyArrayReader {
28    pub fn new(row_count: usize) -> Self {
29        Self {
30            data_type: ArrowType::Struct(Fields::empty()),
31            remaining_rows: row_count,
32            need_consume_records: 0,
33        }
34    }
35}
36
37impl ArrayReader for EmptyArrayReader {
38    fn as_any(&self) -> &dyn Any {
39        self
40    }
41
42    fn get_data_type(&self) -> &ArrowType {
43        &self.data_type
44    }
45
46    fn read_records(&mut self, batch_size: usize) -> Result<usize> {
47        let len = self.remaining_rows.min(batch_size);
48        self.remaining_rows -= len;
49        self.need_consume_records += len;
50        Ok(len)
51    }
52
53    fn consume_batch(&mut self) -> Result<ArrayRef> {
54        let data = ArrayDataBuilder::new(self.data_type.clone())
55            .len(self.need_consume_records)
56            .build()
57            .unwrap();
58        self.need_consume_records = 0;
59        Ok(Arc::new(StructArray::from(data)))
60    }
61
62    fn skip_records(&mut self, num_records: usize) -> Result<usize> {
63        let skipped = self.remaining_rows.min(num_records);
64        self.remaining_rows -= skipped;
65        Ok(skipped)
66    }
67
68    fn get_def_levels(&self) -> Option<&[i16]> {
69        None
70    }
71
72    fn get_rep_levels(&self) -> Option<&[i16]> {
73        None
74    }
75}
