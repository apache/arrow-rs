44/// An array of variable length byte arrays that are potentially dictionary encoded
45/// and can be converted into a corresponding [`ArrayRef`]
46pub enum DictionaryBuffer<K: ArrowNativeType, V: OffsetSizeTrait> {
47    Dict {
48        keys: Vec<K>,
49        values: ArrayRef,
50        /// Exclusive upper bound for decoded keys.
51        key_upper_bound: Option<u64>,
52    },
53    Values {
54        values: OffsetBuffer<V>,
55    },
56}
57
58fn widen_key_bound(existing: &mut Option<u64>, incoming: Option<u64>) {
59    *existing = match (*existing, incoming) {
60        (Some(current), Some(new)) => Some(current.max(new)),
61        _ => None,
62    };
63}

65impl<K: ArrowNativeType + Ord, V: OffsetSizeTrait> DictionaryBuffer<K, V> {
66    #[cfg_attr(not(test), expect(unused))]
67    pub fn len(&self) -> usize {
68        match self {
69            Self::Dict { keys, .. } => keys.len(),
70            Self::Values { values } => values.len(),
71        }
72    }
73
74    /// Returns the key buffer or `None` if the dictionary must be rebuilt.
75    ///
76    /// Records an exclusive bound derived from `rle_bit_width`. A missing width
77    /// discards any previously recorded bound.
78    ///
79    /// # Panics
80    ///
81    /// Panics if the dictionary is too large for `K`.
82    pub fn as_keys_with_bit_width(
83        &mut self,
84        dictionary: &ArrayRef,
85        rle_bit_width: Option<u8>,
86    ) -> Option<&mut Vec<K>> {
87        assert!(K::from_usize(dictionary.len()).is_some());
88
89        // Dictionary RLE bit widths are validated to be at most 32.
90        let incoming_bound = rle_bit_width.map(|bit_width| 1u64 << bit_width);
91
92        match self {
93            Self::Dict {
94                keys,
95                values,
96                key_upper_bound,
97            } => {
98                // Need to discard fat pointer for equality check
99                // - https://stackoverflow.com/a/67114787
100                // - https://github.com/rust-lang/rust/issues/46139
101                let values_ptr = std::ptr::from_ref(values.as_ref()).cast::<()>();
102                let dict_ptr = std::ptr::from_ref(dictionary.as_ref()).cast::<()>();
103                if values_ptr == dict_ptr {
104                    widen_key_bound(key_upper_bound, incoming_bound);
105                    Some(keys)
106                } else if keys.is_empty() {
107                    *values = Arc::clone(dictionary);
108                    *key_upper_bound = incoming_bound;
109                    Some(keys)
110                } else {
111                    None
112                }
113            }
114            Self::Values { values } if values.is_empty() => {
115                *self = Self::Dict {
116                    keys: Default::default(),
117                    values: Arc::clone(dictionary),
118                    key_upper_bound: incoming_bound,
119                };
120                match self {
121                    Self::Dict { keys, .. } => Some(keys),
122                    Self::Values { .. } => unreachable!(),
123                }
124            }
125            Self::Values { .. } => None,
126        }
127    }
128
129    /// Returns a mutable reference to a values array
130    ///
131    /// If this is currently dictionary encoded, this will convert from the
132    /// dictionary encoded representation
133    pub fn spill_values(&mut self) -> Result<&mut OffsetBuffer<V>> {
134        match self {
135            Self::Values { values } => Ok(values),
136            Self::Dict { keys, values, .. } => {
137                let mut spilled = OffsetBuffer::with_capacity(0);
138                let data = values.to_data();
139                let dict_buffers = data.buffers();
140                let dict_offsets = dict_buffers[0].typed_data::<V>();
141                let dict_values = dict_buffers[1].as_slice();
142
143                if values.is_empty() {
144                    // If dictionary is empty, zero pad offsets
145                    spilled.offsets.resize(keys.len() + 1, V::default());
146                } else {
147                    // Note: at this point null positions will have arbitrary dictionary keys
148                    // and this will hydrate them to the corresponding byte array. This is
149                    // likely sub-optimal, as we would prefer zero length null "slots", but
150                    // spilling is already a degenerate case and so it is unclear if this is
151                    // worth optimising for, e.g. by keeping a null mask around
152                    spilled.extend_from_dictionary(keys.as_slice(), dict_offsets, dict_values)?;
153                }
154
155                *self = Self::Values { values: spilled };
156                match self {
157                    Self::Values { values } => Ok(values),
158                    Self::Dict { .. } => unreachable!(),
159                }
160            }
161        }
162    }

164    /// Converts this into an [`ArrayRef`] with the provided `data_type` and `null_buffer`
165    pub fn into_array(
166        self,
167        null_buffer: Option<Buffer>,
168        data_type: &ArrowType,
169        hash_scratch: &mut MutableBuffer,
170    ) -> Result<ArrayRef> {
171        assert!(matches!(data_type, ArrowType::Dictionary(_, _)));
172
173        match self {
174            Self::Dict {
175                keys,
176                values,
177                key_upper_bound,
178            } => {
179                let keys_provably_in_range = key_upper_bound
180                    .map(|bound| bound <= values.len() as u64)
181                    .unwrap_or(false);
182
183                if !values.is_empty() && !keys_provably_in_range {
184                    let min = K::from_usize(0).unwrap();
185                    let max = K::from_usize(values.len()).unwrap();
186
187                    let (observed_min, observed_max) = keys.as_slice().iter().copied().fold(
188                        (max, min),
189                        |(current_min, current_max), key| {
190                            (current_min.min(key), current_max.max(key))
191                        },
192                    );
193
194                    if observed_min < min || observed_max >= max {
195                        return Err(general_err!(
196                            "dictionary key beyond bounds of dictionary: 0..{}",
197                            values.len()
198                        ));
199                    }
200                }
201
202                let ArrowType::Dictionary(key_type, value_type) = data_type else {
203                    unreachable!()
204                };
205                let values = if let ArrowType::FixedSizeBinary(size) = **value_type {
206                    let binary = values.as_binary::<i32>();
207                    Arc::new(FixedSizeBinaryArray::new(
208                        size,
209                        binary.values().clone(),
210                        binary.nulls().cloned(),
211                    )) as _
212                } else {
213                    values
214                };
215
216                let num_keys = keys.len();
217                let keys_buffer = Buffer::from_vec(keys);
218                let null_buffer =
219                    null_buffer.map(|b| NullBuffer::new(BooleanBuffer::new(b, 0, num_keys)));
220
221                build_dictionary_array(key_type, keys_buffer, num_keys, null_buffer, values)
222            }
223            Self::Values { values } => {
224                let (key_type, value_type) = match data_type {
225                    ArrowType::Dictionary(k, v) => (k, v.as_ref().clone()),
226                    _ => unreachable!(),
227                };
228
229                // Plain byte-array pages are decoded into an offset buffer. Convert
230                // this representation explicitly before rebuilding an FSB dictionary,
231                // as FixedSizeBinary arrays have no offsets buffer.
232                if let ArrowType::FixedSizeBinary(size) = value_type {
233                    let binary = values.into_array(null_buffer, GenericBinaryArray::<V>::DATA_TYPE);
234                    let binary = binary.as_binary::<V>();
235                    let array =
236                        FixedSizeBinaryArray::try_from_sparse_iter_with_size(binary.iter(), size)?;
237                    return pack_fixed_values(key_type, &array);
238                }
239
240                hash_byte_slices(&values.offsets, &values.values, hash_scratch);
241                let hashes = hashes_as_u64(hash_scratch);
242
243                pack_values_from_offsets(key_type, &value_type, &values, hashes, null_buffer)
244            }
245        }
246    }
247}

249impl<K: ArrowNativeType, V: OffsetSizeTrait> ValuesBuffer for DictionaryBuffer<K, V> {
250    fn with_capacity(capacity: usize) -> Self {
251        Self::Values {
252            values: OffsetBuffer::with_capacity(capacity),
253        }
254    }
255
256    fn reserve_exact(&mut self, additional: usize) {
257        match self {
258            Self::Dict { keys, .. } => keys.reserve_exact(additional),
259            Self::Values { values, .. } => values.reserve_exact(additional),
260        }
261    }
262
263    fn pad_nulls(
264        &mut self,
265        read_offset: usize,
266        values_read: usize,
267        levels_read: usize,
268        valid_mask: &[u8],
269    ) -> Result<()> {
270        match self {
271            Self::Dict { keys, .. } => {
272                keys.resize(read_offset + levels_read, K::default());
273                keys.pad_nulls(read_offset, values_read, levels_read, valid_mask)
274            }
275            Self::Values { values, .. } => {
276                values.pad_nulls(read_offset, values_read, levels_read, valid_mask)
277            }
278        }
279    }
280}

282/// Build a dictionary array from a raw keys buffer and an already-decoded values
283/// `ArrayRef`, skipping the ArrayData round-trip.
284fn build_dictionary_array(
285    key_type: &ArrowType,
286    keys_buffer: Buffer,
287    num_keys: usize,
288    null_buffer: Option<NullBuffer>,
289    values: ArrayRef,
290) -> Result<ArrayRef> {
291    macro_rules! build {
292        ($kt:ty) => {{
293            let scalars =
294                ScalarBuffer::<<$kt as ArrowPrimitiveType>::Native>::new(keys_buffer, 0, num_keys);
295            let keys = PrimitiveArray::<$kt>::new(scalars, null_buffer);
296            // SAFETY: key bounds already validated by caller
297            let dict = unsafe { DictionaryArray::<$kt>::new_unchecked(keys, values) };
298            Ok(Arc::new(dict) as ArrayRef)
299        }};
300    }
301    match key_type {
302        ArrowType::Int8 => build!(Int8Type),
303        ArrowType::Int16 => build!(Int16Type),
304        ArrowType::Int32 => build!(Int32Type),
305        ArrowType::Int64 => build!(Int64Type),
306        ArrowType::UInt8 => build!(UInt8Type),
307        ArrowType::UInt16 => build!(UInt16Type),
308        ArrowType::UInt32 => build!(UInt32Type),
309        ArrowType::UInt64 => build!(UInt64Type),
310        _ => Err(general_err!("unsupported dictionary key type: {key_type}")),
311    }
312}

314macro_rules! fixed_dict_helper {
315    ($k:ty, $array:ident) => {
316        pack_fixed_values_impl::<$k>($array)
317    };
318}

320fn pack_fixed_values(key_type: &ArrowType, array: &FixedSizeBinaryArray) -> Result<ArrayRef> {
321    downcast_integer! {
322        key_type => (fixed_dict_helper, array),
323        _ => unreachable!(),
324    }
325}
326
327fn pack_fixed_values_impl<K: ArrowDictionaryKeyType>(
328    array: &FixedSizeBinaryArray,
329) -> Result<ArrayRef> {
330    let mut builder = FixedSizeBinaryDictionaryBuilder::<K>::with_capacity(
331        array.len(),
332        1024,
333        array.value_length(),
334    );
335    for x in array {
336        match x {
337            Some(x) => {
338                builder.append(x)?;
339            }
340            None => builder.append_null(),
341        }
342    }
343    Ok(Arc::new(builder.finish()))
344}

346macro_rules! offsets_dict_helper {
347    ($k:ty, $key_type:ident, $value_type:ident, $values:ident, $hashes:ident, $null_buffer:ident) => {
348        pack_values_from_offsets_impl::<$k, _>(
349            $values,
350            $hashes,
351            $null_buffer,
352            $key_type,
353            $value_type,
354        )
355    };

358fn pack_values_from_offsets<V: OffsetSizeTrait>(
359    key_type: &ArrowType,
360    value_type: &ArrowType,
361    values: &OffsetBuffer<V>,
362    hashes: &[u64],
363    null_buffer: Option<Buffer>,
364) -> Result<ArrayRef> {
365    downcast_integer! {
366        key_type => (offsets_dict_helper, key_type, value_type, values, hashes, null_buffer),
367        _ => unreachable!(),
368    }

371// Avoids double-hashing: keys are already high-quality u64 hashes from ahash,
372// so we pass them through directly rather than re-hashing inside the HashMap.
373struct PassthroughHasher(u64);
374impl std::hash::Hasher for PassthroughHasher {
375    fn finish(&self) -> u64 {
376        self.0
377    }
378    fn write(&mut self, _: &[u8]) {
379        unreachable!()
380    }
381    fn write_u64(&mut self, value: u64) {
382        self.0 = value;
383    }
384}
385#[derive(Default)]
386struct BuildPassthroughHasher;
387impl std::hash::BuildHasher for BuildPassthroughHasher {
388    type Hasher = PassthroughHasher;
389    fn build_hasher(&self) -> PassthroughHasher {
390        PassthroughHasher(0)
391    }
392}

394/// Builds a [`DictionaryArray`] directly from a flat [`OffsetBuffer`] using pre-computed
395/// hashes to deduplicate values in a single pass, avoiding the intermediate StringArray
396/// materialization
397fn pack_values_from_offsets_impl<K: ArrowDictionaryKeyType, V: OffsetSizeTrait>(
398    offset_buffer: &OffsetBuffer<V>,
399    hashes: &[u64],
400    null_buffer: Option<Buffer>,
401    key_type: &ArrowType,
402    value_type: &ArrowType,
403) -> Result<ArrayRef> {
404    let dict_type = ArrowType::Dictionary(Box::new(key_type.clone()), Box::new(value_type.clone()));
405    let num_values = offset_buffer.len();
406
407    let mut keys: Vec<K::Native> = Vec::with_capacity(num_values);
408    let mut unique_offsets: Vec<V> = Vec::with_capacity(num_values + 1);
409    unique_offsets.push(V::default());
410    let mut unique_bytes: Vec<u8> = Vec::with_capacity(offset_buffer.values.len());
411
412    let mut dedup: HbHashMap<u64, (usize, usize), BuildPassthroughHasher> =
413        HbHashMap::with_capacity_and_hasher(num_values, BuildPassthroughHasher);
414
415    // Tracks colliding values (same hash, different bytes) so repeated occurrences
416    // are deduplicated rather than inserted as new entries. Empty in the common case.
417    let mut collision_overflow: Vec<(u64, usize, usize)> = Vec::new();
418
419    for (input_idx, &hash) in hashes.iter().enumerate() {
420        let byte_start = offset_buffer.offsets[input_idx].as_usize();
421        let byte_end = offset_buffer.offsets[input_idx + 1].as_usize();
422        let bytes = &offset_buffer.values[byte_start..byte_end];
423
424        let output_idx = match dedup.entry(hash) {
425            Entry::Occupied(entry) => {
426                let (first_input_idx, existing_output_idx) = *entry.get();
427                let first_start = offset_buffer.offsets[first_input_idx].as_usize();
428                let first_end = offset_buffer.offsets[first_input_idx + 1].as_usize();
429                if &offset_buffer.values[first_start..first_end] == bytes {
430                    existing_output_idx
431                } else {
432                    // True hash collision — check overflow list before inserting.
433                    let existing = collision_overflow
434                        .iter()
435                        .find(|&&(entry_hash, collision_input_idx, _)| {
436                            if entry_hash != hash {
437                                return false;
438                            }
439                            let collision_start =
440                                offset_buffer.offsets[collision_input_idx].as_usize();
441                            let collision_end =
442                                offset_buffer.offsets[collision_input_idx + 1].as_usize();
443                            &offset_buffer.values[collision_start..collision_end] == bytes
444                        })
445                        .map(|&(_, _, collision_output_idx)| collision_output_idx);
446
447                    match existing {
448                        Some(collision_output_idx) => collision_output_idx,
449                        None => {
450                            let new_output_idx = unique_offsets.len() - 1;
451                            unique_bytes.extend_from_slice(bytes);
452                            let new_end = V::from_usize(unique_bytes.len()).ok_or_else(|| {
453                                general_err!("offset overflow building dictionary")
454                            })?;
455                            unique_offsets.push(new_end);
456                            collision_overflow.push((hash, input_idx, new_output_idx));
457                            new_output_idx
458                        }
459                    }
460                }
461            }
462            Entry::Vacant(entry) => {
463                let output_idx = unique_offsets.len() - 1;
464                unique_bytes.extend_from_slice(bytes);
465                let new_end = V::from_usize(unique_bytes.len())
466                    .ok_or_else(|| general_err!("offset overflow building dictionary"))?;
467                unique_offsets.push(new_end);
468                entry.insert((input_idx, output_idx));
469                output_idx
470            }
471        };
472
473        let key = K::Native::from_usize(output_idx)
474            .ok_or_else(|| general_err!("dictionary key overflow"))?;
475        keys.push(key);
476    }
477
478    let num_unique = unique_offsets.len() - 1;
479
480    // SAFETY: buffers are constructed directly from typed Vecs above; offsets are
481    // monotonically non-decreasing and bounded by unique_bytes.len(), and all
482    // key values are within 0..num_unique, so the invariants Arrow requires hold.
483    let value_data = unsafe {
484        arrow_data::ArrayData::builder(value_type.clone())
485            .len(num_unique)
486            .add_buffer(Buffer::from_vec(unique_offsets))
487            .add_buffer(Buffer::from_vec(unique_bytes))
488            .build_unchecked()
489    };
490
491    // SAFETY: keys are within 0..num_unique and value_data is valid.
492    let dict_array: DictionaryArray<K> = unsafe {
493        arrow_data::ArrayData::builder(dict_type)
494            .len(keys.len())
495            .add_buffer(Buffer::from_vec(keys))
496            .add_child_data(value_data)
497            .null_bit_buffer(null_buffer)
498            .build_unchecked()
499            .into()
500    };
501
502    Ok(Arc::new(dict_array))
503}

505fn hash_byte_slices<I: ArrowNativeType>(offsets: &[I], values: &[u8], scratch: &mut MutableBuffer) {
506    let count = offsets.len().saturating_sub(1);
507    scratch.clear();
508    scratch.resize(count * size_of::<u64>(), 0u8);
509
510    let state = RandomState::new();
511
512    // SAFETY: MutableBuffer is 64-byte aligned; scratch is sized to exactly count * size_of::<u64>()
513    #[expect(
514        clippy::cast_ptr_alignment,
515        reason = "MutableBuffer is 64-byte aligned"
516    )]
517    let hash_slots = unsafe { from_raw_parts_mut(scratch.as_mut_ptr().cast::<u64>(), count) };
518
519    for idx in 0..count {
520        let start = offsets[idx].as_usize();
521        let end = offsets[idx + 1].as_usize();
522        let mut hasher = state.build_hasher();
523        hasher.write(&values[start..end]);
524        // SAFETY: idx is within 0..count
525        unsafe { write_unaligned(hash_slots.as_mut_ptr().add(idx), hasher.finish()) };
526    }
527}

529#[inline]
530fn hashes_as_u64(scratch: &[u8]) -> &[u64] {
531    let n = scratch.len() / size_of::<u64>();
532    // SAFETY: scratch was written as u64s by hash_byte_slices, and MutableBuffer is 64-byte aligned
533    #[expect(
534        clippy::cast_ptr_alignment,
535        reason = "MutableBuffer is 64-byte aligned"
536    )]
537    unsafe {
538        from_raw_parts(scratch.as_ptr().cast::<u64>(), n)
539    }
540}

562            .extend_from_slice(values);

575        values.try_push(b"bongo", false).unwrap();

747    fn test_fixed_size_binary_key_overflow_boundary() {
748        for (key_type, capacity) in [(ArrowType::Int8, 128u32), (ArrowType::UInt8, 256)] {
749            for unique in [capacity, capacity + 1] {
750                let values = (0..unique)
751                    .map(|value| Some(value.to_le_bytes()))
752                    .chain([None, Some(0u32.to_le_bytes())]);
753                let values =
754                    FixedSizeBinaryArray::try_from_sparse_iter_with_size(values, 4).unwrap();
755                let result = pack_fixed_values(&key_type, &values);
756                if unique > capacity {
757                    let error = result.unwrap_err();
758                    let ParquetError::External(source) = &error else {
759                        panic!("expected an Arrow dictionary overflow error, got {error}");
760                    };
761                    assert!(
762                        matches!(
763                            source.downcast_ref::<arrow_schema::ArrowError>(),
764                            Some(arrow_schema::ArrowError::DictionaryKeyOverflowError)
765                        ),
766                        "{error}"
767                    );
768                } else {
769                    let array = result.unwrap();
770                    assert_eq!(array.len(), unique as usize + 2);
771                    assert_eq!(array.null_count(), 1);
772                    assert!(array.is_null(unique as usize));
773                    assert!(array.is_valid(unique as usize + 1));
774                    let dictionary = array.as_any_dictionary();
775                    assert_eq!(dictionary.values().len(), unique as usize);
776                    assert_eq!(
777                        dictionary.values().data_type(),
778                        &ArrowType::FixedSizeBinary(4)
779                    );
780                }
781            }
782        }
