5//! [`OffsetIndexMetaData`] structure holding decoded [`OffsetIndex`] information
6//!
7//! [`OffsetIndex`]: https://github.com/apache/parquet-format/blob/master/PageIndex.md
8
9use std::io::Write;
10
11use crate::parquet_thrift::{
12    ElementType, FieldType, ReadThrift, ThriftCompactInputProtocol, ThriftCompactOutputProtocol,
13    WriteThrift, WriteThriftField, read_thrift_vec, validate_list_type,
14};
15use crate::{
16    errors::{ParquetError, Result},
17    thrift_struct,
18};

20thrift_struct!(
21/// Page location information for [`OffsetIndexMetaData`]
22pub struct PageLocation {
23  /// Offset of the page in the file
24  1: required i64 offset
25  /// Size of the page, including header. Sum of compressed_page_size and header
26  2: required i32 compressed_page_size
27  /// Index within the RowGroup of the first row of the page. When an
28  /// OffsetIndex is present, pages must begin on row boundaries
29  /// (repetition_level = 0).
30  3: required i64 first_row_index
31}
32);
33
34thrift_struct!(
35/// [`OffsetIndex`] information for a column chunk. Contains offsets and sizes for each page
36/// in the chunk. Optionally stores fully decoded page sizes for BYTE_ARRAY columns.
37///
38/// See [`PageIndex`] for more information.
39///
40/// [`PageIndex`]: crate::file::metadata::page_index::PageIndex
41/// [`OffsetIndex`]: https://github.com/apache/parquet-format/blob/master/PageIndex.md
42pub struct OffsetIndexMetaData {
43  /// Vector of [`PageLocation`] objects, one per page in the chunk.
44  1: required list<PageLocation> page_locations
45  /// Optional vector of unencoded page sizes, one per page in the chunk.
46  /// Only defined for BYTE_ARRAY columns.
47  2: optional list<i64> unencoded_byte_array_data_bytes
48}
49);
50
51impl OffsetIndexMetaData {
52    /// Vector of [`PageLocation`] objects, one per page in the chunk.
53    pub fn page_locations(&self) -> &Vec<PageLocation> {
54        &self.page_locations
55    }
56
57    /// Optional vector of unencoded page sizes, one per page in the chunk. Only defined
58    /// for BYTE_ARRAY columns.
59    pub fn unencoded_byte_array_data_bytes(&self) -> Option<&Vec<i64>> {
60        self.unencoded_byte_array_data_bytes.as_ref()
61    }
62
63    // Fast-path read of offset index. This works because we expect all field deltas to be 1,
64    // and there's no nesting beyond PageLocation, so no need to save the last field id. Like
65    // read_page_locations(), this will fail if absolute field id's are used.
66    pub(super) fn try_from_fast<'a, R: ThriftCompactInputProtocol<'a>>(
67        prot: &mut R,
68    ) -> Result<Self> {
69        // Offset index is a struct with 2 fields. First field is an array of PageLocations,
70        // the second an optional array of i64.
71
72        // read field 1 header, then list header, then vec of PageLocations
73        let (field_type, delta) = prot.read_field_header()?;
74        if delta != 1 || field_type != FieldType::List as u8 {
75            return Err(general_err!("error reading OffsetIndex::page_locations"));
76        }
77
78        // we have to do this manually because we want to use the fast PageLocation decoder
79        let list_ident = prot.read_list_begin()?;
80        validate_list_type(ElementType::Struct, &list_ident)?;
81        let mut page_locations = Vec::with_capacity(list_ident.size as usize);
82        for _ in 0..list_ident.size {
83            page_locations.push(read_page_location(prot)?);
84        }
85
86        let mut unencoded_byte_array_data_bytes: Option<Vec<i64>> = None;
87
88        // read second field...if it's Stop we're done
89        let (mut field_type, delta) = prot.read_field_header()?;
90        if field_type == FieldType::List as u8 {
91            if delta != 1 {
92                return Err(general_err!(
93                    "encountered unknown field while reading OffsetIndex"
94                ));
95            }
96            let vec = read_thrift_vec::<i64, R>(&mut *prot)?;
97            unencoded_byte_array_data_bytes = Some(vec);
98
99            // this one should be Stop
100            (field_type, _) = prot.read_field_header()?;
101        }
102
103        if field_type != FieldType::Stop as u8 {
104            return Err(general_err!(
105                "encountered unknown field while reading OffsetIndex"
106            ));
107        }
108
109        Ok(Self {
110            page_locations,
111            unencoded_byte_array_data_bytes,
112        })
113    }
114}
115
116// hand coding this one because it is very time critical
117
118// Note: this will fail if the fields are either out of order, or if a suboptimal
119// encoder doesn't use field deltas.
120fn read_page_location<'a, R: ThriftCompactInputProtocol<'a>>(prot: &mut R) -> Result<PageLocation> {
121    // there are 3 fields, all mandatory, so all field deltas should be 1
122    let (field_type, delta) = prot.read_field_header()?;
123    if delta != 1 || field_type != FieldType::I64 as u8 {
124        return Err(general_err!("error reading PageLocation::offset"));
125    }
126    let offset = prot.read_i64()?;
127
128    let (field_type, delta) = prot.read_field_header()?;
129    if delta != 1 || field_type != FieldType::I32 as u8 {
130        return Err(general_err!(
131            "error reading PageLocation::compressed_page_size"
132        ));
133    }
134    let compressed_page_size = prot.read_i32()?;
135
136    let (field_type, delta) = prot.read_field_header()?;
137    if delta != 1 || field_type != FieldType::I64 as u8 {
138        return Err(general_err!("error reading PageLocation::first_row_index"));
139    }
140    let first_row_index = prot.read_i64()?;
141
142    // read end of struct...return error if there are unknown fields present
143    let (field_type, _) = prot.read_field_header()?;
144    if field_type != FieldType::Stop as u8 {
145        return Err(general_err!("unexpected field in PageLocation"));
146    }
147
148    Ok(PageLocation {
149        offset,
150        compressed_page_size,
151        first_row_index,
152    })
153}

155#[cfg(test)]
156mod tests {
157    use super::*;
158    use crate::parquet_thrift::tests::test_roundtrip;
159
160    #[test]
161    fn test_offset_idx_roundtrip() {
162        let page_locations = [
163            PageLocation {
164                offset: 0,
165                compressed_page_size: 10,
166                first_row_index: 0,
167            },
168            PageLocation {
169                offset: 10,
170                compressed_page_size: 20,
171                first_row_index: 100,
172            },
173        ]
174        .to_vec();
175        let unenc = [0i64, 100i64].to_vec();
176
177        test_roundtrip(OffsetIndexMetaData {
178            page_locations: page_locations.clone(),
179            unencoded_byte_array_data_bytes: Some(unenc),
180        });
181        test_roundtrip(OffsetIndexMetaData {
182            page_locations,
183            unencoded_byte_array_data_bytes: None,
184        });
185    }
