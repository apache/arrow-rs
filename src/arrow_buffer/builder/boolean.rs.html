9/// Builder for [`BooleanBuffer`]
10///
11/// Builds a packed buffer of bits representing boolean values. Each bit in the
12/// buffer corresponds to a boolean value,
13///
14/// # See Also
15///
16/// * [`NullBufferBuilder`] for building [`BooleanBuffer`]s for representing nulls
17/// * [`BufferBuilder`] for building [`Buffer`]s
18///
19/// # Example
20/// ```
21/// # use arrow_buffer::builder::BooleanBufferBuilder;
22/// let mut builder = BooleanBufferBuilder::new(10);
23/// builder.append(true);
24/// builder.append(false);
25/// builder.append_n(3, true); // append 3 trues
26/// let buffer = builder.build();
27/// assert_eq!(buffer.len(), 5); // 5 bits appended
28/// assert_eq!(buffer.values(), &[0b00011101_u8]); // packed bits
29///```
30///
31/// [`BufferBuilder`]: crate::builder::BufferBuilder
32/// [`NullBufferBuilder`]: crate::builder::NullBufferBuilder
33#[derive(Debug)]
34pub struct BooleanBufferBuilder {
35    buffer: MutableBuffer,
36    len: usize,
37}

40    /// Creates a new `BooleanBufferBuilder` with sufficient space for
41    /// `capacity` bits (not bytes).
42    ///
43    /// The capacity is rounded up to the nearest multiple of 8 for the
44    /// allocation.
45    #[inline]
46    pub fn new(capacity: usize) -> Self {
47        let byte_capacity = bit_util::ceil(capacity, 8);
48        let buffer = MutableBuffer::new(byte_capacity);
49        Self { buffer, len: 0 }
50    }

52    /// Creates a new `BooleanBufferBuilder` from [`MutableBuffer`] of `len`
53    ///
54    /// # Panics
55    ///
56    /// Panics if `len > buffer.len() * 8`
57    pub fn new_from_buffer(buffer: MutableBuffer, len: usize) -> Self {
58        assert!(len <= buffer.len() * 8);
59        let mut s = Self {
60            len: buffer.len() * 8,
61            buffer,
62        };
63        s.truncate(len);
64        s
65    }

103    /// Returns the capacity of the buffer, in bits (not bytes)
104    ///
105    /// Note this
106    ///
107    /// # Example
108    /// ```
109    /// # use arrow_buffer::builder::BooleanBufferBuilder;
110    /// // empty requires 0 bytes
111    /// let b = BooleanBufferBuilder::new(0);
112    /// assert_eq!(0, b.capacity());
113    /// // Creating space for 1 bit results in 64 bytes (space for 512 bits)
114    /// // (64 is the minimum allocation size for 64 bit architectures)
115    /// let mut b = BooleanBufferBuilder::new(1);
116    /// assert_eq!(512, b.capacity());
117    /// // 1000 bits requires 128 bytes (space for 1024 bits)
118    /// b.append_n(1000, true);
119    /// assert_eq!(1024, b.capacity());
120    /// ```
121    #[inline]
122    pub fn capacity(&self) -> usize {
123        self.buffer.capacity() * 8
124    }

126    /// Advances the buffer by `additional` bits
127    #[inline]
128    pub fn advance(&mut self, additional: usize) {
129        let new_len = self.len + additional;
130        let new_len_bytes = bit_util::ceil(new_len, 8);
131        if new_len_bytes > self.buffer.len() {
132            self.buffer.resize(new_len_bytes, 0);
133        }
134        self.len = new_len;
135    }

137    /// Truncates the builder to the given length
138    ///
139    /// If `len` is greater than the buffer's current length, this has no effect
140    #[inline]
141    pub fn truncate(&mut self, len: usize) {
142        if len > self.len {
143            return;
144        }
145
146        let new_len_bytes = bit_util::ceil(len, 8);
147        self.buffer.truncate(new_len_bytes);
148        self.len = len;
149
150        let remainder = self.len % 8;
151        if remainder != 0
152            && let Some(last) = self.buffer.as_mut().last_mut()
153        {
154            let mask = (1_u8 << remainder).wrapping_sub(1);
155            *last &= mask;
156        }
157    }

159    /// Reserve space to at least `additional` new bits.
160    /// Capacity will be `>= self.len() + additional`.
161    #[inline]
162    pub fn reserve(&mut self, additional: usize) {
163        let capacity = self.len + additional;
164        if capacity > self.capacity() {
165            // convert differential to bytes
166            let additional = bit_util::ceil(capacity, 8) - self.buffer.len();
167            self.buffer.reserve(additional);
168        }
169    }

171    /// Resizes the buffer, either truncating its contents (with no change in capacity), or
172    /// growing it (potentially reallocating it) and writing `false` in the newly available bits.
173    #[inline]
174    pub fn resize(&mut self, len: usize) {
175        match len.checked_sub(self.len) {
176            Some(delta) => self.advance(delta),
177            None => self.truncate(len),
178        }
179    }

181    /// Appends a boolean `v` into the buffer
182    #[inline]
183    pub fn append(&mut self, v: bool) {
184        self.advance(1);
185        if v {
186            unsafe { bit_util::set_bit_raw(self.buffer.as_mut_ptr(), self.len - 1) };
187        }
188    }

190    /// Appends the low `count` bits from `word` into the buffer.
191    ///
192    /// `word` is treated as a packed LSB-first bitmap. Only the lowest
193    /// `count` bits are appended; higher bits are ignored.
194    ///
195    /// This is significantly faster than calling [`Self::append`] in a
196    /// loop when the caller already has bits packed into a `u64`.
197    #[inline]
198    pub fn append_word(&mut self, word: u64, count: usize) {
199        debug_assert!(count <= 64);
200        let mask = (u64::MAX >> ((64 - count) & 63)) * ((count != 0) as u64);
201        let word = word & mask;
202
203        let new_len = self.len + count;
204        let new_len_bytes = bit_util::ceil(new_len, 8);
205        if new_len_bytes > self.buffer.len() {
206            self.buffer.resize(new_len_bytes, 0);
207        }
208
209        let bit_offset = self.len & 7;
210        let byte_start = self.len / 8;
211        let buf = self.buffer.as_slice_mut();
212
213        // Shift the word to align with the bit offset within the
214        // current byte, then OR it into the buffer. The shift merges
215        // correctly with any existing bits in the partial trailing byte.
216        let shifted = word << bit_offset;
217        let shifted_bytes = shifted.to_le_bytes();
218        let bytes_to_write = bit_util::ceil(count + bit_offset, 8).min(8);
219        for i in 0..bytes_to_write {
220            buf[byte_start + i] |= shifted_bytes[i];
221        }
222        // When bit_offset > 0 the shift can overflow into a 9th byte.
223        if bit_offset > 0 && count + bit_offset > 64 {
224            buf[byte_start + 8] |= (word >> (64 - bit_offset)) as u8;
225        }
226
227        self.len = new_len;
228    }

230    /// Appends n `additional` bits of value `v` into the buffer
231    #[inline]
232    pub fn append_n(&mut self, additional: usize, v: bool) {
233        match v {
234            true => {
235                let new_len = self.len + additional;
236                let new_len_bytes = bit_util::ceil(new_len, 8);
237                let cur_remainder = self.len % 8;
238                let new_remainder = new_len % 8;
239
240                if cur_remainder != 0
241                    && let Some(last) = self.buffer.as_slice_mut().last_mut()
242                {
243                    // Pad last byte with 1s
244                    *last |= !((1 << cur_remainder) - 1);
245                }
246                self.buffer.resize(new_len_bytes, 0xFF);
247                if new_remainder != 0
248                    && let Some(last) = self.buffer.as_slice_mut().last_mut()
249                {
250                    // Clear remaining bits
251                    *last &= (1 << new_remainder) - 1;
252                }
253                self.len = new_len;
254            }
255            false => self.advance(additional),
256        }
257    }

259    /// Appends a slice of booleans into the buffer
260    #[inline]
261    pub fn append_slice(&mut self, slice: &[bool]) {
262        let additional = slice.len();
263        self.advance(additional);
264
265        let offset = self.len() - additional;
266
267        // Up to 8 values touch at most two bytes, so pack them directly. This
268        // includes 8 values at an unaligned offset, which leave no whole byte
269        // for `pack_slice` to pack.
270        if additional <= 8 {
271            if additional == 0 {
272                return;
273            }
274            // At most 8 bits starting at bit 7 end at bit 14, so they span at most two bytes
275            let bits = (Self::pack_bits(slice) as u16) << (offset % 8);
276            let buffer = self.buffer.as_slice_mut();
277            buffer[offset / 8] |= bits as u8;
278            if let Some(next) = buffer.get_mut(offset / 8 + 1) {
279                *next |= (bits >> 8) as u8;
280            }
281        } else {
282            self.pack_slice(offset, slice)
283        }
284    }
285
286    /// Packs up to 8 bools into a byte, `slice[i]` at bit `i`,
287    /// without a branch per value
288    #[inline]
289    fn pack_bits(slice: &[bool]) -> u8 {
290        slice
291            .iter()
292            .enumerate()
293            .fold(0, |acc, (i, v)| acc | ((*v as u8) << i))
294    }
295
296    /// Writes `slice` into the already allocated bits starting at `offset`
297    ///
298    /// Kept out of line so that the short slice path of [`Self::append_slice`]
299    /// stays small when inlined
300    #[inline(never)]
301    fn pack_slice(&mut self, offset: usize, slice: &[bool]) {
302        let additional = slice.len();
303        let buffer = self.buffer.as_slice_mut();
304
305        // Fill the first byte up to the next byte boundary
306        let head = ((8 - offset % 8) % 8).min(additional);
307        let (head_slice, rest) = slice.split_at(head);
308        buffer[offset / 8] |= Self::pack_bits(head_slice) << (offset % 8);
309
310        // Pack 8 bools into each byte without branching: a `bool` is a byte
311        // holding 0 or 1, and the multiply moves byte `i` to bit `56 + i`
312        let byte_offset = (offset + head) / 8;
313        let (chunks, remainder) = rest.as_chunks::<8>();
314        for (dst, chunk) in buffer[byte_offset..].iter_mut().zip(chunks) {
315            let bytes: [u8; 8] = std::array::from_fn(|i| chunk[i] as u8);
316            let packed = u64::from_le_bytes(bytes).wrapping_mul(0x0102_0408_1020_4080) >> 56;
317            *dst = packed as u8;
318        }
319
320        // The remainder starts on a byte boundary
321        if !remainder.is_empty() {
322            buffer[(self.len - remainder.len()) / 8] |= Self::pack_bits(remainder);
323        }
324    }

326    /// Append `range` bits from `to_set`
327    ///
328    /// `to_set` is a slice of bits packed LSB-first into `[u8]`
329    ///
330    /// # Panics
331    ///
332    /// Panics if `to_set` does not contain `ceil(range.end / 8)` bytes
333    pub fn append_packed_range(&mut self, range: Range<usize>, to_set: &[u8]) {
334        let offset_write = self.len;
335        let len = range.end - range.start;
336        // allocate new bits as 0
337        self.advance(len);
338        // copy bits from to_set into self.buffer a word at a time
339        apply_bitwise_binary_op(
340            self.buffer.as_slice_mut(),
341            offset_write,
342            to_set,
343            range.start,
344            len,
345            |_a, b| b, // copy bits from to_set
346        );
347    }

349    /// Append [`BooleanBuffer`] to this [`BooleanBufferBuilder`]
350    pub fn append_buffer(&mut self, buffer: &BooleanBuffer) {
351        let range = buffer.offset()..buffer.offset() + buffer.len();
352        self.append_packed_range(range, buffer.values())
353    }

365    /// Resets this builder and returns a [`BooleanBuffer`].
366    ///
367    /// Use [`Self::build`] when you don't need to reuse this builder.
368    #[inline]
369    pub fn finish(&mut self) -> BooleanBuffer {
370        let buf = std::mem::replace(&mut self.buffer, MutableBuffer::new(0));
371        let len = std::mem::replace(&mut self.len, 0);
372        BooleanBuffer::new(buf.into(), 0, len)
373    }
374
375    /// Builds a [`BooleanBuffer`] without resetting the builder.
376    ///
377    /// This consumes the builder. Use [`Self::finish`] to reuse it.
378    #[inline]
379    pub fn build(self) -> BooleanBuffer {
380        BooleanBuffer::new(self.buffer.into(), 0, self.len)
381    }
382
383    /// Builds the [BooleanBuffer] without resetting the builder.
384    pub fn finish_cloned(&self) -> BooleanBuffer {
385        BooleanBuffer::new(Buffer::from_slice_ref(self.as_slice()), 0, self.len)
386    }

388    /// Extends the builder from a trusted length iterator of booleans.
389    /// # Safety
390    /// Callers must ensure that `iter` reports an exact size via `size_hint`.
391    ///
392    #[inline]
393    pub unsafe fn extend_trusted_len<I>(&mut self, iterator: I)
394    where
395        I: Iterator<Item = bool>,
396    {
397        let len = iterator.size_hint().0;
398        unsafe { self.buffer.extend_bool_trusted_len(iterator, self.len) };
399        self.len += len;
400    }

721    fn check_append_word(initial_bits: usize, word: u64, count: usize) {
722        let mut got = BooleanBufferBuilder::new(0);
723        let mut expected = BooleanBufferBuilder::new(0);
724        got.append_n(initial_bits, true);
725        expected.append_n(initial_bits, true);

729        }
730        assert_eq!(got.len(), expected.len());

836        for byte in 0..=u8::MAX {
837            // 8 values take the short path, 16 values are packed with the multiply
838            let values: Vec<bool> = (0..16).map(|i| byte & (1 << (i % 8)) != 0).collect();
839            let mut b = BooleanBufferBuilder::new(8);

843            let mut b = BooleanBufferBuilder::new(16);
844            b.append_slice(&values);
845            assert_eq!(b.as_slice(), &[byte, byte]);
846        }

849    #[test]
850    fn test_append_slice_offsets() {
851        // Exercise every combination of starting bit offset and slice length
852        // around the byte boundaries, including after a truncate.
853        let pattern: Vec<bool> = (0..100).map(|i| (i * 7 + i / 3) % 5 < 2).collect();
854        for start in 0..17 {
855            for len in 0..pattern.len() {
856                let mut b = BooleanBufferBuilder::new(0);
857                b.append_n(start + 3, true);
858                b.truncate(start);
859                b.append_slice(&pattern[..len]);
860                b.append(true);
861                let finished = b.finish();
862                assert_eq!(finished.len(), start + len + 1);
863                for i in 0..start {
864                    assert!(finished.value(i), "start {start} len {len} index {i}");
865                }
866                for (i, v) in pattern[..len].iter().enumerate() {
867                    assert_eq!(
868                        finished.value(start + i),
869                        *v,
870                        "start {start} len {len} index {i}"
871                    );
872                }
873                assert!(finished.value(start + len));
874            }
875        }
